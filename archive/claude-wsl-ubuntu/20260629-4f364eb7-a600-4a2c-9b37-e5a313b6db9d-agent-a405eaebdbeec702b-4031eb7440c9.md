#!/usr/bin/env python3
"""Broken internal markdown link scanner (stdlib only).

For every [text](target) link in tracked *.md files (root + docs/), verify:
  * relative file paths resolve (relative to the md file's directory),
  * #anchors resolve to an actual GitHub-slugified heading in the target file.

External (http/https/mailto), absolute (/...), and pure-anchor-to-self links to
non-existent headings are all checked. Prints `file:line -> target` for each
broken link plus a TOTAL count. Exit 1 if any broken link is found.
"""
from __future__ import annotations

import os
import re
import sys
from pathlib import Path

# Inline-link regex: [text](target).  Skips image-only refs are still links here.
LINK_RE = re.compile(r"(?<!\!)\[(?:[^\]]*)\]\(([^)]+)\)")
ATX_HEADING_RE = re.compile(r"^\s{0,3}(#{1,6})\s+(.*?)\s*#*\s*$")

# Roots to scan. Repo root *.md (top level only) + everything under docs/.
ROOT = Path(__file__).resolve().parent  # overridden by argv[1]


def slugify(heading: str) -> str:
    """GitHub-flavoured heading -> anchor slug (good-enough subset)."""
    # strip inline code/formatting markers and links, keep text
    text = re.sub(r"`+", "", heading)
    text = re.sub(r"\[([^\]]*)\]\([^)]*\)", r"\1", text)  # [t](u) -> t
    text = re.sub(r"[*_~]", "", text)
    text = text.strip().lower()
    # drop anything that's not word char, space or hyphen
    text = re.sub(r"[^\w\s-]", "", text, flags=re.UNICODE)
    text = text.replace(" ", "-")
    return text


def collect_anchors(md_path: Path) -> set[str]:
    anchors: set[str] = set()
    try:
        lines = md_path.read_text(encoding="utf-8", errors="replace").splitlines()
    except OSError:
        return anchors
    in_fence = False
    seen: dict[str, int] = {}
    for line in lines:
        stripped = line.strip()
        if stripped.startswith("```") or stripped.startswith("~~~"):
            in_fence = not in_fence
            continue
        if in_fence:
            continue
        m = ATX_HEADING_RE.match(line)
        if m:
            slug = slugify(m.group(2))
            if not slug:
                continue
            # GitHub disambiguates duplicate slugs with -1, -2, ...
            n = seen.get(slug, 0)
            anchors.add(slug if n == 0 else f"{slug}-{n}")
            seen[slug] = n + 1
        # explicit <a name="..."> / id="..." anchors
        for am in re.finditer(r'<a\s+(?:name|id)="([^"]+)"', line):
            anchors.add(am.group(1))
    return anchors


def iter_links(md_path: Path):
    lines = md_path.read_text(encoding="utf-8", errors="replace").splitlines()
    in_fence = False
    for i, line in enumerate(lines, start=1):
        stripped = line.strip()
        if stripped.startswith("```") or stripped.startswith("~~~"):
            in_fence = not in_fence
            continue
        if in_fence:
            continue
        for m in LINK_RE.finditer(line):
            yield i, m.group(1).strip()


def is_external(target: str) -> bool:
    return bool(re.match(r"^[a-zA-Z][a-zA-Z0-9+.-]*://", target)) or target.startswith(
        ("mailto:", "tel:", "#? ")
    ) or target.startswith("mailto:")


def main() -> int:
    repo = Path(sys.argv[1]).resolve() if len(sys.argv) > 1 else Path.cwd()
    md_files: list[Path] = []
    # top-level *.md only (root)
    md_files += sorted(p for p in repo.glob("*.md") if p.is_file())
    # all docs/**.md
    docs = repo / "docs"
    if docs.is_dir():
        md_files += sorted(docs.rglob("*.md"))

    anchor_cache: dict[Path, set[str]] = {}

    def anchors_for(p: Path) -> set[str]:
        if p not in anchor_cache:
            anchor_cache[p] = collect_anchors(p)
        return anchor_cache[p]

    broken: list[tuple[str, int, str, str]] = []  # file, line, target, reason

    for md in md_files:
        for lineno, target in iter_links(md):
            if not target or target.startswith("<"):
                continue
            if is_external(target):
                continue
            # split path#anchor
            path_part, _, anchor = target.partition("#")
            # pure in-page anchor
            if path_part == "":
                if anchor and anchor not in anchors_for(md):
                    broken.append(
                        (str(md.relative_to(repo)), lineno, target, "missing anchor (same file)")
                    )
                continue
            # strip query string if any
            path_part = path_part.split("?", 1)[0]
            if path_part.startswith("/"):
                resolved = (repo / path_part.lstrip("/")).resolve()
            else:
                resolved = (md.parent / path_part).resolve()
            if not resolved.exists():
                broken.append((str(md.relative_to(repo)), lineno, target, "missing path"))
                continue
            if anchor:
                if resolved.is_dir():
                    broken.append(
                        (str(md.relative_to(repo)), lineno, target, "anchor on a directory")
                    )
                elif resolved.suffix.lower() == ".md":
                    if anchor not in anchors_for(resolved):
                        broken.append(
                            (str(md.relative_to(repo)), lineno, target, "missing anchor")
                        )

    broken.sort(key=lambda b: (b[0], b[1]))
    for f, ln, tgt, reason in broken:
        print(f"{f}:{ln} -> {tgt}    [{reason}]")
    print(f"\nTOTAL broken internal links: {len(broken)}")
    print(f"(scanned {len(md_files)} markdown files)")
    return 1 if broken else 0


if __name__ == "__main__":
    raise SystemExit(main())
