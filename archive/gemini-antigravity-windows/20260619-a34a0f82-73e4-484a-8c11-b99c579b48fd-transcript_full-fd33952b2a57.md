194: def parse_cvat_boxes(xml_path: str) -> Tuple[Dict[str, Any], List[Dict[str, Any]]]:
195:     """Parse a CVAT XML export into (meta, boxes).
196: 
197:     ``boxes`` is a flat list of ``{"frame": int, "attrs": {name: value}}`` — one
198:     entry per visible box, from either image-mode or track-mode exports. Attr
199:     names are folded via ``_fold_attr_name`` (lowercase, separators unified,
200:     parenthesized hints dropped, plural-tolerant), not the raw vendor spelling.
201:     ``meta`` carries ``step`` (frame_filter) and ``stop_frame`` when present.
202:     """
203:     if not os.path.exists(xml_path):
204:         raise FileNotFoundError(f"CVAT export not found: {xml_path}")
205: 
206:     # Accept the standard CVAT delivery artifact (a .zip containing annotations.xml) as
207:     # well as a bare .xml. Corrupt/unparseable XML is surfaced as a ValueError (a clean
208:     # gate failure) rather than an uncaught ParseError traceback.
209:     try:
210:         if zipfile.is_zipfile(xml_path):
211:             with zipfile.ZipFile(xml_path) as zf:
212:                 names = ([n for n in zf.namelist() if n.endswith("annotations.xml")]
213:                          or [n for n in zf.namelist() if n.endswith(".xml")])
214:                 if not names:
215:                     raise ValueError(f"CVAT zip contains no annotations.xml: {xml_path}")
216:                 root = ET.fromstring(zf.read(names[0]))
217:         else:
218:             root = ET.parse(xml_path).getroot()
219:     except ET.ParseError as e:
220:         raise ValueError(f"CVAT XML is corrupt/unparseable ({xml_path}): {e}")
221:     meta: Dict[str, Any] = {}
222: 
223:     job = root.find(".//job")
224:     if job is not None:
225:         stop = job.findtext("stop_frame")
226:         if stop is not None and stop.strip().isdigit():
227:             meta["stop_frame"] = int(stop.strip())
228:         ff = job.findtext("frame_filter") or ""
229:         if "step=" in ff:
230:             try:
231:                 meta["step"] = int(ff.split("step=")[1].split()[0])
232:             except (ValueError, IndexError):
233:                 pass
234: 
235:     def _attrs(el: ET.Element) -> Dict[str, str]:
236:         out: Dict[str, str] = {}
237:         for a in el.findall("attribute"):
238:             out[_fold_attr_name(a.get("name"))] = (a.text or "").strip()
239:         return out
240: 
241:     boxes: List[Dict[str, Any]] = []
242: 
243:     # Image mode: <image id=frame> ... <box>...</box>
244:     for img in root.findall("image"):
245:         try:
246:             frame = int(img.get("id"))
247:         except (TypeError, ValueError):
248:             continue
249:         for box in img.findall("box"):
250:             boxes.append({"frame": frame, "attrs": _attrs(box)})
251: 
252:     # Track mode: <track> ... <box frame=N outside=0>...</box>
253:     for track in root.findall("track"):
254:         for box in track.findall("box"):
255:             if box.get("outside") == "1":
256:                 continue  # a track's terminating keyframe carries no real label
257:             try:
258:                 frame = int(box.get("frame"))
259:             except (TypeError, ValueError):
260:                 continue
261:             boxes.append({"frame": frame, "attrs": _attrs(box)})
262: 
263:     return meta, boxes

266: def _infer_step(frames: List[int]) -> int:
267:     """Most common positive gap between consecutive distinct sampled frames."""
268:     uniq = sorted(set(frames))
269:     gaps = [b - a for a, b in zip(uniq, uniq[1:]) if b - a > 0]
270:     if not gaps:
271:         return 1
272:     return Counter(gaps).most_common(1)[0][0]

314: def detect_issues(records: List[RallyRecord],
315:                   stop_frame: Optional[int] = None,
316:                   allowed_reasons: Optional[set] = None,
317:                   allowed_outcomes: Optional[set] = None,
318:                   max_gap_s: float = 15.0,
319:                   fps: Optional[float] = None,
320:                   boundary_tolerance_s: float = 0.3) -> List[Issue]:
321:     """Find the defects that block or degrade a clean ground-truth import.
322: 
323:     Sampling density INSIDE a rally is deliberately not flagged — sparse
324:     in-rally boxes are fine, and highlight compilation pads clips anyway
325:     (badminton-highlight-indexer compiler, config-driven 0.5s/0.5s). What IS
326:     flagged (``boundary_grid_snap``, when ``fps`` is known) is every rally
327:     BOUNDARY sitting on a common sampling grid coarser than the ±0.3s bar:
328:     that means first/last boxes were snapped to sampled frames instead of the
329:     exact serve-contact/dead frame, which structurally cannot meet the golden
330:     quality bar (Brief §4) no matter how careful the rater was — the defect
331:     that invalidated the pickleball/table-tennis pilot (2026-06-10 review).
332:     """
333:     allowed = allowed_reasons if allowed_reasons is not None else DEFAULT_ALLOWED_REASONS
334:     allowed_out = allowed_outcomes if allowed_outcomes is not None else DEFAULT_ALLOWED_OUTCOMES
335:     issues: List[Issue] = []
336: 
337:     ordered = sorted(records, key=lambda r: r.rally_number)
338:     for i, r in enumerate(ordered):
339:         if r.end_time <= r.start_time:
340:             issues.append(Issue(
341:                 "blocker", "zero_duration",
342:                 f"Rally {r.rally_number}: end <= start "
343:                 f"(frames {r.frame_min}-{r.frame_max}); single-sample or stray box.",
344:                 r.rally_number))
345: 
346:         if r.ending_reason is not None and r.ending_reason not in allowed:
347:             issues.append(Issue(
348:                 "medium", "ending_reason_vocab",
349:                 f"Rally {r.rally_number}: ending_reason '{r.ending_reason}' "
350:                 f"not in the agreed set {sorted(allowed)}.",
351:                 r.rally_number))
352: 
353:         if r.last_shot_outcome is not None and r.last_shot_outcome not in allowed_out:
354:             issues.append(Issue(
355:                 "low", "outcome_vocab",
356:                 f"Rally {r.rally_number}: last_shot_outcome '{r.last_shot_outcome}' "
357:                 f"not in {sorted(allowed_out)}.",
358:                 r.rally_number))
359: 
360:         if r.court is not None and r.court not in DEFAULT_ALLOWED_COURTS:
361:             issues.append(Issue(
362:                 "low", "court_vocab",
363:                 f"Rally {r.rally_number}: court '{r.court}' "
364:                 f"not in {sorted(DEFAULT_ALLOWED_COURTS)}.",
365:                 r.rally_number))
366: 
367:         if r.confidence is not None and r.confidence not in DEFAULT_ALLOWED_CONFIDENCE:
368:             issues.append(Issue(
369:                 "low", "confidence_vocab",
370:                 f"Rally {r.rally_number}: confidence '{r.confidence}' "
371:                 f"not in {sorted(DEFAULT_ALLOWED_CONFIDENCE)}.",
372:                 r.rally_number))
373:         elif r.confidence == "unsure":
374:             # Not an error — the annotator asked for a second look; surface it on
375:             # the review sheet so the human pass prioritises it.
376:             issues.append(Issue(
377:                 "low", "annotator_unsure",
378:                 f"Rally {r.rally_number}: annotator marked confidence=unsure — review.",
379:                 r.rally_number))
380: 
381:         # Inflated span: a long rally with implausibly few shots usually means a
382:         # stray/duplicated box stretched the frame range past the real ending.
383:         if r.shots_count and r.duration >= 5.0:
384:             sps = r.shots_count / r.duration
385:             if sps < 0.3:
386:                 issues.append(Issue(
387:                     "medium", "shot_density",
388:                     f"Rally {r.rally_number}: {r.shots_count} shots over "
389:                     f"{r.duration:.1f}s ({sps:.2f}/s) is implausibly low — the span "
390:                     f"may be inflated by a stray box.",
391:                     r.rally_number))
392: 
393:         if stop_frame is not None and r.frame_max > stop_frame:
394:             issues.append(Issue(
395:                 "high", "frame_out_of_range",
396:                 f"Rally {r.rally_number}: frame {r.frame_max} exceeds the video's "
397:                 f"last frame {stop_frame}.",
398:                 r.rally_number))
399: 
400:         if i > 0:
401:             prev = ordered[i - 1]
402:             if r.start_time < prev.end_time:
403:                 issues.append(Issue(
404:                     "blocker", "overlap",
405:                     f"Rally {prev.rally_number} (ends {prev.end_time:.2f}s) overlaps "
406:                     f"Rally {r.rally_number} (starts {r.start_time:.2f}s) by "
407:                     f"{prev.end_time - r.start_time:.2f}s — rallies must be disjoint.",
408:                     r.rally_number))
409:             elif r.start_time - prev.end_time > max_gap_s:
410:                 # A long unlabeled stretch between two rallies is a cheap, label-only
411:                 # second opinion for the worst error (a missed rally). It won't catch a
412:                 # rally MERGED into its neighbour's span (shot_density/inflated catches
413:                 # that), but it does catch an omitted rally that left a visible hole.
414:                 gap = r.start_time - prev.end_time
415:                 issues.append(Issue(
416:                     "medium", "possible_missed_rally",
417:                     f"{gap:.1f}s of unlabeled time between Rally {prev.rally_number} "
418:                     f"(ends {prev.end_time:.1f}s) and Rally {r.rally_number} "
419:                     f"(starts {r.start_time:.1f}s) — check for an unlabeled rally in the gap.",
420:                     r.rally_number))
421: 
422:     grid = _detect_boundary_grid(ordered, fps, boundary_tolerance_s)
423:     if grid is not None:
424:         issues.append(grid)
425:     return issues

428: def _detect_boundary_grid(records: List[RallyRecord], fps: Optional[float],
429:                           tolerance_s: float) -> Optional[Issue]:
430:     """Flag rally BOUNDARIES that all sit on one sampling grid (frame stride).
431: 
432:     If every first/last box frame is congruent modulo a stride g (the gcd of
433:     their pairwise differences), boundaries were snapped to sampled frames. A
434:     stride coarser than the boundary tolerance makes the ±tolerance bar
435:     structurally unmeetable, so the delivery must be reworked — a blocker.
436:     Needs frame info (CVAT path) and fps; CSV-loaded records (frame_min == -1)
437:     and small deliveries (< 3 rallies) are skipped to avoid false positives.
438:     """
439:     if not fps or fps <= 0:
440:         return None
441:     boundaries = [f for r in records for f in (r.frame_min, r.frame_max) if f >= 0]
442:     if len(boundaries) < 6:
443:         return None
444:     base = boundaries[0]
445:     g = 0
446:     for f in boundaries[1:]:
447:         g = math.gcd(g, abs(f - base))
448:     if g < 2:           # frame-accurate (or no common grid) — the healthy case
449:         return None
450:     stride_s = g / fps
451:     if stride_s <= tolerance_s:
452:         return None     # grid finer than the bar can still meet it
453:     return Issue(
454:         "blocker", "boundary_grid_snap",
455:         f"All {len(boundaries)} rally boundaries sit on a {g}-frame grid "
456:         f"(~{stride_s:.2f}s at {fps:.2f} fps) — first/last boxes were snapped to "
457:         f"sampled frames, so boundaries structurally cannot meet the "
458:         f"±{tolerance_s:g}s bar (Brief §4). The first/last box of every rally "
459:         f"must be placed on the exact serve-contact / dead frame (frame-step "
460:         f"to it); in-between boxes may stay sparse.")

### Briefly, here's how the pipeline detects rallies:
1. **Trajectory Tracking:** The video is processed (e.g., TrackNet) into a timeline of the shuttle's X, Y coordinates and visibility per frame.
2. **Velocity Thresholding:** A frame is flagged as "active" if the shuttle is moving fast enough across the screen.
3. **Temporal Grouping:** Active frames within a `merge_gap` (e.g. 2 seconds) are clustered into "candidate windows".
4. **Trimming & Splitting (W1, R4, R5 rules):** 
   - Windows that stretch too long are intelligently split at internal dead-time gaps.
   - A trailing trimmer shaves off the end of a window by looking at the density of "fast" frames, which cuts off the slow drift when a shuttle is picked up post-rally.
   - If a window continues infinitely without breaks (a tracking blob), it is forcefully chopped in half as a last resort.
5. **Net-Crossing Gate (Gate A):** To filter out players just tossing the shuttle back, it counts how many times the shuttle coordinate crosses the "median" of the play axis.
6. **AI Handoff:** The resulting clips are padded by 2 seconds and sent to an AI provider (like Gemini) to do final semantic boundary confirmation and to count the exact number of shuttle exchanges.

### Notable Issues Discovered:
* **`_trim_inactive_tail` Sensitivity:** The algorithm uses `win_count` as the number of *visible* frames when evaluating motion density at the end of a rally. Because it counts visible frames rather than total elapsed time, a very sparse tracking segment with just 1 fast visible frame will score a 100% density. This could artificially extend a window.
* **`_spread` Utility Math:** For extremely short sequences (`len < 20`), the 95th percentile indexing collapses gracefully, but for a 2-point sequence, it evaluates strictly to `0.0`. It currently doesn't break the pipeline because it's run over thousands of points, but would fail if repurposed on a tiny window.
* **Silent CSV Skips:** In `rally_cvat.py`, the `load_canonical_csv()` function silently continues if a `rally_number` is missing. This protects against minor typos but could mask a heavily malformed dataset without warning.

Overall, the codebase is defensively written and exceptionally well-thought-out, utilizing safe fallbacks (such as the recursive slice check `len(h) < 2` before splitting) to prevent infinite loops.
