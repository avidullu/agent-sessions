402	    await conn.execute(
403	        """
404	        CREATE TABLE telegram_sessions (
405	            account_id    INTEGER PRIMARY KEY REFERENCES accounts(id),
406	            phone         TEXT,
407	            session_path  TEXT NOT NULL,
408	            enabled       INTEGER NOT NULL DEFAULT 1,
409	            created_at    TEXT NOT NULL,
410	            last_login_at TEXT
411	        )
412	        """
413	    )
414	    # session_path is built from the *telegram_id* (= old users.id) to match
415	    # the convention telethon_client.build_client used pre-migration. The
416	    # sessions_dir prefix is prepended at read time (Settings.sessions_dir)
417	    # so this migration does not need to know the runtime path.
418	    await conn.execute(
419	        """
420	        INSERT INTO telegram_sessions (account_id, phone, session_path, enabled, created_at)
421	        SELECT id, phone, 'user_' || id || '.session', enabled, created_at
422	        FROM users
423	        """
424	    )
425	
426	    # Rebuild channels / videos / filter_config so their FK constraints point
427	    # at accounts(id) instead of the legacy users(id). SQLite's
428	    # ALTER TABLE RENAME rewrites FK references in *other* tables to follow
429	    # the rename — so a naive rename of users → users_deprecated would leave
430	    # these tables referencing users_deprecated(id), diverging from fresh DBs
431	    # (which reference accounts(id)) and breaking inserts for new accounts.
432	    # The rebuild: rename the data-holding legacy table out of the way, create
433	    # the new one with the correct FK, copy the data back, drop the renamed copy.
434	    for table, new_ddl, copy_cols in (
435	        (
436	            "channels",
437	            """
438	            CREATE TABLE channels (
439	                id            INTEGER PRIMARY KEY AUTOINCREMENT,
440	                user_id       INTEGER NOT NULL REFERENCES accounts(id),
441	                channel_id    INTEGER NOT NULL,
442	                title         TEXT,
443	                seen_first_at TEXT NOT NULL,
444	                UNIQUE (user_id, channel_id)
445	            )
446	            """,
447	            "id, user_id, channel_id, title, seen_first_at",
448	        ),
449	        (
450	            "videos",
451	            """
452	            CREATE TABLE videos (
453	                id          INTEGER PRIMARY KEY AUTOINCREMENT,
454	                user_id     INTEGER NOT NULL REFERENCES accounts(id),
455	                channel_id  INTEGER NOT NULL,
456	                message_id  INTEGER NOT NULL,
457	                document_id INTEGER,
458	                duration_s  INTEGER,
459	                width       INTEGER,
460	                height      INTEGER,
461	                size_bytes  INTEGER,
462	                mime_type   TEXT,
463	                matched     INTEGER NOT NULL DEFAULT 0,
464	                forwarded   INTEGER NOT NULL DEFAULT 0,
465	                seen_at     TEXT NOT NULL,
466	                UNIQUE (user_id, channel_id, message_id)
467	            )
468	            """,
469	            "id, user_id, channel_id, message_id, document_id, duration_s, width, "
470	            "height, size_bytes, mime_type, matched, forwarded, seen_at",
471	        ),
472	        (
473	            "filter_config",
474	            """
475	            CREATE TABLE filter_config (
476	                user_id     INTEGER PRIMARY KEY REFERENCES accounts(id),
477	                config_json TEXT NOT NULL
478	            )
479	            """,
480	            "user_id, config_json",
481	        ),
482	    ):
483	        await conn.execute(f"ALTER TABLE {table} RENAME TO {table}__legacy_data")
484	        await conn.execute(new_ddl)
485	        await conn.execute(
486	            f"INSERT INTO {table} ({copy_cols}) SELECT {copy_cols} FROM {table}__legacy_data"
487	        )
488	        await conn.execute(f"DROP TABLE {table}__legacy_data")
489	
490	    # The partial unique index for document_id dedup (created by migration 001
491	    # against the legacy videos table) was dropped along with the legacy copy
492	    # above. Recreate it on the new videos table.
493	    await conn.execute(
494	        "CREATE UNIQUE INDEX IF NOT EXISTS idx_videos_user_document "
495	        "ON videos(user_id, document_id) WHERE document_id IS NOT NULL"
496	    )
497	
498	    # Rename the legacy users table (do NOT drop — kept as a rollback safety
499	    # net for one release per ADR 0010). The data-holding channels/videos/
500	    # filter_config tables were already rebuilt above with FKs to accounts(id),
501	    # so renaming users no longer retargets any live FK.
502	    await conn.execute("ALTER TABLE users RENAME TO users_deprecated")

505	@_migration(3)
506	async def _migration_003_domain_schema(conn: aiosqlite.Connection) -> None:
507	    """Land the provider-neutral domain tables (ADR 0011, Track B).
508	
509	    Creates ``sources``, ``content_items``, ``policies``, ``destinations``,
510	    ``routing_rules``, ``delivery_attempts`` (all ``IF NOT EXISTS`` — on fresh
511	    DBs ``_SCHEMA`` already created them; on pre-ADR-0011 DBs this creates
512	    them), then backfills the three populated ones from the legacy
513	    ``channels`` / ``videos`` / ``filter_config`` tables and renames the
514	    legacy tables to ``*_deprecated`` (one-release rollback net, ADR 0010
515	    pattern). Idempotent: a no-op once the legacy tables have been renamed.
516	    """
517	    # Idempotency gate: if the legacy videos table is gone, this migration has
518	    # already run (or the DB is fresh and _SCHEMA's legacy tables were already
519	    # dropped by _apply_schema post-003). Nothing to backfill.
520	    if not await _table_exists(conn, "videos"):
521	        return
522	
523	    # 1a. Backfill sources from channels first (known title + watched=1). Order
524	    #    matters: channels-first so title/watched data wins; the videos-based
525	    #    backfill below fills gaps for orphan channels (review finding #1).
526	    await conn.execute(
527	        """
528	        INSERT OR IGNORE INTO sources
529	            (account_id, provider, provider_source_id, title, kind, watched,
530	             seen_first_at, last_seen_at)
531	        SELECT user_id, 'telegram', CAST(channel_id AS TEXT), title, 'channel',
532	               1, seen_first_at, seen_first_at
533	        FROM channels
534	        """
535	    )
536	
537	    # 1b. Backfill sources from videos for channels that have no channels row
538	    #     (e.g. backfill-only channels that were never seen via live traffic).
539	    #     INSERT OR IGNORE is a no-op on (account, provider, source_id) conflicts
540	    #     — the channels-based row above survives with its title + watched=1.
541	    await conn.execute(
542	        """
543	        INSERT OR IGNORE INTO sources
544	            (account_id, provider, provider_source_id, kind, watched, seen_first_at)
545	        SELECT v.user_id, 'telegram', CAST(v.channel_id AS TEXT),
546	               'channel', 0, MIN(v.seen_at)
547	        FROM videos v
548	        GROUP BY v.user_id, v.channel_id
549	        """
550	    )
551	
552	    # 2. Backfill content_items from videos in two phases (review findings #2 & #3):
553	    #    Phase A — insert every row with a guaranteed-unique tg:msg: Tier-0 key
554	    #    (no silent drops, even on fingerprint collisions).
555	    #    Phase B — promote to tg:doc: for items with a document_id (one per doc-id group).
556	    #    Phase C — promote remaining non-doc items to tg:fp: (one per fingerprint group).
557	    #    The non-promoted losers stay at tg:msg: (unique) — no silent data loss.
558	    await conn.execute(
559	        """
560	        INSERT INTO content_items
561	            (account_id, source_id, provider, provider_message_id, content_type,
562	             document_id, duration_s, width, height, size_bytes, mime_type,
563	             dedup_key, matched, seen_at)
564	        SELECT
565	            v.user_id,
566	            s.id,
567	            'telegram',
568	            CAST(v.message_id AS TEXT),
569	            'video',
570	            CASE WHEN v.document_id IS NOT NULL THEN CAST(v.document_id AS TEXT) END,
571	            v.duration_s, v.width, v.height, v.size_bytes, v.mime_type,
572	            'tg:msg:' || s.id || ':' || v.message_id,  -- Tier-0 (always unique)
573	            v.matched,
574	            v.seen_at
575	        FROM videos v
576	        JOIN sources s
577	          ON s.account_id = v.user_id
578	         AND s.provider = 'telegram'
579	         AND s.provider_source_id = CAST(v.channel_id AS TEXT)
580	        """
581	    )
582	
583	    # Phase B: promote one row per (account, document_id) to tg:doc:.
584	    await conn.execute(
585	        """
586	        UPDATE content_items SET dedup_key = 'tg:doc:' || document_id
587	        WHERE document_id IS NOT NULL
588	          AND dedup_key LIKE 'tg:msg:%'
589	          AND id IN (
590	            SELECT MIN(id) FROM content_items
591	            WHERE document_id IS NOT NULL AND dedup_key LIKE 'tg:msg:%'
592	            GROUP BY account_id, document_id
593	          )
594	        """
595	    )
596	
597	    # Phase C: promote one non-doc row per (account, fingerprint) to tg:fp:.
598	    await conn.execute(
599	        """
600	        UPDATE content_items SET dedup_key =
601	            'tg:fp:' || size_bytes || ':' || duration_s || ':'
602	            || height || ':' || width
603	        WHERE document_id IS NULL
604	          AND size_bytes IS NOT NULL AND duration_s IS NOT NULL
605	          AND height IS NOT NULL AND width IS NOT NULL
606	          AND dedup_key LIKE 'tg:msg:%'
607	          AND id IN (
608	            SELECT MIN(id) FROM content_items
609	            WHERE document_id IS NULL
610	              AND size_bytes IS NOT NULL AND duration_s IS NOT NULL
611	              AND height IS NOT NULL AND width IS NOT NULL
612	              AND dedup_key LIKE 'tg:msg:%'
613	            GROUP BY account_id, size_bytes, duration_s, height, width
614	          )
615	        """
616	    )
617	
618	    # 3. Backfill policies from filter_config: one 'default' policy per account
619	    #    (Option 2A). is_default=1 so the legacy load_filter_config shim reads
620	    #    this row. UNIQUE(account_id, name) dedups idempotent re-runs.
621	    await conn.execute(
622	        """
623	        INSERT OR IGNORE INTO policies (account_id, name, config_json, is_default,
624	                                        created_at, updated_at)
625	        SELECT user_id, 'default', config_json, 1,
626	               strftime('%Y-%m-%dT%H:%M:%SZ', 'now'),
627	               strftime('%Y-%m-%dT%H:%M:%SZ', 'now')
628	        FROM filter_config
629	        """
630	    )
631	
632	    # 4. Create the policies default-partial index (sig after backfill so the
633	    #    UNIQUE constraint only applies once rows are populated). On fresh DBs
634	    #    _SCHEMA creates it; on migrated DBs this is the first time.
635	    await conn.execute(
636	        "CREATE UNIQUE INDEX IF NOT EXISTS idx_policies_account_default "
637	        "ON policies(account_id) WHERE is_default = 1"
638	    )
639	
640	    # 5. Assert no silent video loss: on a legacy DB, every videos row MUST
641	    #    produce a content_items row. A mismatch signals fingerprint collisions
642	    #    where two distinct videos shared (size,dur,h,w) — the losers stay at
643	    #    tg:msg: keys and are still present. Log a warning so the operator can
644	    #    investigate; the data is preserved in *_deprecated tables.
645	    async with conn.execute("SELECT COUNT(*) FROM videos") as cur:
646	        vc = (await cur.fetchone())[0]
647	    async with conn.execute("SELECT COUNT(*) FROM content_items") as cur:
648	        cic = (await cur.fetchone())[0]
649	    if vc != cic:
650	        import logging
651	
652	        _logger = logging.getLogger(__name__)
653	        _logger.warning(
654	            "migration_003: %d videos rows produced %d content_items rows "
655	            "(%d gap). Colliding items remain in videos_deprecated.",
656	            vc,
657	            cic,
658	            vc - cic,
659	        )
660	
661	    # 6. Rename (don't drop) the legacy tables — kept for one release as a
662	    #    rollback safety net (ADR 0010 pattern).
663	    await conn.execute("ALTER TABLE channels RENAME TO channels_deprecated")
664	    await conn.execute("ALTER TABLE videos RENAME TO videos_deprecated")
665	    await conn.execute("ALTER TABLE filter_config RENAME TO filter_config_deprecated")

668	@_migration(4)
669	async def _migration_004_destinations_and_routing(conn: aiosqlite.Connection) -> None:
670	    """Backfill default destinations and delivery state (ADR 0011, Track C).
671	
672	    Creates a default destination row for each account that has a
673	    ``filter_config_deprecated.delivery_target`` value, then creates a default
674	    routing rule (source → default policy → destination) for each source.
675	    Also backfills ``delivery_attempts`` rows from the legacy ``videos_deprecated``
676	    table (one ``delivered`` attempt per previously-forwarded video to the
677	    account's default destination).
678	
679	    Idempotent: skips accounts whose default destination already exists.
680	    """
681	    # Gate: if destinations already have data, this migration has run.
682	    async with conn.execute("SELECT COUNT(*) FROM destinations") as cur:
683	        if (await cur.fetchone())[0] > 0:
684	            return
685	
686	    if not await _table_exists(conn, "filter_config_deprecated"):
687	        return
688	
689	    # 1. Backfill default destinations from filter_config_deprecated.
690	    #    Parse the JSON delivery_target, defaulting to "me".
691	    await conn.execute(
692	        """
693	        INSERT OR IGNORE INTO destinations (account_id, provider, target, kind, is_output, created_at)
694	        SELECT user_id, 'telegram',
695	               COALESCE(
696	                 CASE WHEN json_valid(config_json)
697	                      THEN json_extract(config_json, '$.delivery_target') END,
698	                 'me'
699	               ),
700	               CASE
701	                 WHEN COALESCE(
702	                   CASE WHEN json_valid(config_json)
703	                        THEN json_extract(config_json, '$.delivery_target') END,
704	                   'me'
705	                 ) = 'me'
706	                   THEN 'saved'
707	                 ELSE 'channel'
708	               END,
709	               0,
710	               strftime('%Y-%m-%dT%H:%M:%SZ', 'now')
711	        FROM filter_config_deprecated
712	        """
713	    )
714	
715	    # 2. Create a default routing rule per source (source → default policy → default destination).
716	    await conn.execute(
717	        """
718	        INSERT OR IGNORE INTO routing_rules
719	            (account_id, source_id, policy_id, destination_id, enabled, priority, created_at)
720	        SELECT s.account_id, s.id, p.id, d.id, 1, 0,
721	               strftime('%Y-%m-%dT%H:%M:%SZ', 'now')
722	        FROM sources s
723	        JOIN policies p ON p.account_id = s.account_id AND p.is_default = 1
724	        JOIN destinations d ON d.account_id = s.account_id
725	        WHERE d.target = (
726	            SELECT COALESCE(
727	                CASE WHEN json_valid(fcd.config_json)
728	                     THEN json_extract(fcd.config_json, '$.delivery_target') END,
729	                'me'
730	            )
731	            FROM filter_config_deprecated fcd
732	            WHERE fcd.user_id = s.account_id
733	            LIMIT 1
734	        )
735	        """
736	    )
737	
738	    # 3. Backfill delivery_attempts from previously-forwarded videos.
739	    await conn.execute(
740	        """
741	        INSERT OR IGNORE INTO delivery_attempts
742	            (content_item_id, destination_id, route_id, status, attempt_count,
743	             created_at, delivered_at)
744	        SELECT ci.id, d.id, rr.id, 'delivered', 1,
745	               vd.seen_at, vd.seen_at
746	        FROM videos_deprecated vd
747	        JOIN content_items ci
748	          ON ci.account_id = vd.user_id
749	         AND ci.provider = 'telegram'
750	         AND ci.provider_message_id = CAST(vd.message_id AS TEXT)
751	        JOIN sources sc
752	          ON sc.id = ci.source_id
753	         AND sc.provider_source_id = CAST(vd.channel_id AS TEXT)
754	        JOIN destinations d ON d.account_id = vd.user_id
755	        JOIN routing_rules rr
756	          ON rr.account_id = vd.user_id
757	         AND rr.source_id = sc.id
758	         AND rr.destination_id = d.id
759	        WHERE vd.forwarded = 1
760	        """
761	    )

781	    """Cast a provider_source_id / provider_message_id / document_id to int.
782	
783	    All existing data is Telegram — these ids are numeric strings. Non-Telegram

795	            f"non-Telegram providers (ADR 0011)."
796	        ) from exc
