377	+class NvdecResidentFrontend:
378	+    """On-GPU frame source for the ``nvdec_resident`` detector pass (#506).
379	+
380	+    PyNvVideoCodec ``SimpleDecoder(output_color_type=RGB)`` decodes on the GPU's
381	+    NVDEC engine; frames arrive as CUDA tensors via DLPack and never touch the host.
382	+    ``read_chw`` then applies the cv2-faithful affine warp (``build_affine_grid``)
383	+    and ImageNet-normalizes on-device, yielding the same ``[3, inp_h, inp_w]``
384	+    float32 tensor per frame that the ImageDataset path assembles on the CPU.
385	+    """
386	+
387	+    def __init__(self, video_path: str, inp_wh, gpu_id: int = 0):
388	+        import PyNvVideoCodec as nvc
389	+        import torch
390	+
391	+        self._torch = torch
392	+        self._device = torch.device("cuda", gpu_id)
393	+        inp_w, inp_h = int(inp_wh[0]), int(inp_wh[1])
394	+        try:
395	+            self._dec = nvc.SimpleDecoder(
396	+                video_path, gpu_id=gpu_id, output_color_type=nvc.OutputColorType.RGB
397	+            )
398	+        except Exception as e:
399	+            raise RuntimeError(
400	+                f"PyNvVideoCodec RGB SimpleDecoder failed for {video_path}: {e!r} — "
401	+                f"decode_backend=nvdec_resident needs PyNvVideoCodec with RGB output "
402	+                f"and libnvcuvid/libnvidia-encode on the box (deploy/cloudrun/Dockerfile "
403	+                f"/ gpu_setup.sh stage them). Fall back to decode_backend=cpu."
404	+            ) from e
405	+        self.count = len(self._dec)
406	+        probe = self._decode_rgb_hwc(0)
407	+        self.orig_h, self.orig_w = int(probe.shape[0]), int(probe.shape[1])
408	+        trans_inp_inv = _wasb_inverse_affines(
409	+            self.orig_h, self.orig_w, (inp_w, inp_h), [0]
410	+        )[0]
411	+        self._grid = build_affine_grid(
412	+            trans_inp_inv, inp_w, inp_h, self.orig_w, self.orig_h
413	+        ).to(self._device)
414	+        self._mean = torch.tensor(IMAGENET_MEAN, device=self._device).view(3, 1, 1)
415	+        self._std = torch.tensor(IMAGENET_STD, device=self._device).view(3, 1, 1)
416	+
417	+    def _decode_rgb_hwc(self, index: int):
418	+        """Frame ``index`` as an on-GPU HWC RGB uint8 tensor (zero-copy via DLPack)."""
419	+        torch = self._torch
420	+        t = torch.from_dlpack(self._dec[index])
421	+        if t.dim() == 3 and t.shape[0] == 3 and t.shape[-1] != 3:
422	+            t = t.permute(1, 2, 0)  # CHW layout variant → HWC
423	+        if t.dim() != 3 or t.shape[-1] != 3:
424	+            raise RuntimeError(
425	+                f"unexpected NVDEC frame shape {tuple(t.shape)} for frame {index} — "
426	+                f"expected 3-channel RGB (is output_color_type=RGB supported here?)"
427	+            )
428	+        return t
429	+
430	+    def read_chw(self, index: int):
431	+        """Decoded frame ``index`` → the normalized ``[3, inp_h, inp_w]`` float32 CUDA
432	+        tensor the detector expects (warpAffine-faithful resize + ImageNet normalize)."""
433	+        import torch.nn.functional as F
434	+
435	+        hwc = self._decode_rgb_hwc(index)
436	+        chw = hwc.permute(2, 0, 1).unsqueeze(0).float() / 255.0
437	+        warped = F.grid_sample(
438	+            chw, self._grid, mode="bilinear", align_corners=False, padding_mode="border"
439	+        ).squeeze(0)
440	+        return (warped - self._mean) / self._std
441	+
442	+    def release(self):
443	+        self._dec = None

337	class SequentialFrameStore:

339	    detector's sliding-window access pattern (peak O(window) frames, not O(video)).
340	
341	    The detector DataLoader runs with ``shuffle=False`` and (in streaming mode)
342	    ``num_workers=0``. ``ImageDataset.__getitem__(i)`` reads the frames of window ``i``
343	    in order — ``i, i+1, ..., i+window-1`` — and samples are requested ``i = start,
344	    start+1, ...``. So the global read sequence dips back by ``window-1`` at each window
345	    boundary (``0,1,2, 1,2,3, 2,3,4, ...`` for ``window=3``); the highest index seen so
346	    far never decreases. We keep only frames at or above ``max_seen - (window-1)`` (the
347	    earliest index any not-yet-finished window can still ask for) and decode each source
348	    frame exactly once via the forward-only ``reader(index)``.

352	        self._reader = reader

354	        self._window = max(1, int(window))
355	        self._buf: dict = {}
356	        # On a resumed run the detector's first needed frame is `start_index`; begin
357	        # pulling there so the reader can skip (seek past) the already-cached prefix
358	        # instead of decoding 0..start_index-1 just to throw them away.
359	        self._next = int(start_index)  # next index not yet pulled from the reader
360	        self._max_seen = int(start_index) - 1
361	        self._floor = int(start_index)  # lowest index we still keep (never evict below)
362	
363	    def get(self, index: int):
364	        index = int(index)
365	        if index < self._floor:
366	            raise KeyError(
367	                f"frame {index} already evicted (below floor {self._floor}); the access "
368	                f"pattern must stay within a window of the highest frame seen"
369	            )
370	        if index >= self._total:
371	            raise KeyError(f"frame {index} out of range (total {self._total})")
372	        # Pull forward from the reader until the requested index is buffered.
373	        while self._next <= index:
374	            self._buf[self._next] = self._reader(self._next)
375	            self._next += 1
376	        if index > self._max_seen:
377	            self._max_seen = index
378	        # Evict frames older than the earliest a still-running window could re-request:
379	        # once we've seen index `m`, no future window starts before `m - (window-1)`.
380	        new_floor = max(self._floor, self._max_seen - (self._window - 1))
381	        for k in range(self._floor, new_floor):
382	            self._buf.pop(k, None)
383	        self._floor = new_floor
384	        return self._buf[index]

429	@contextmanager
430	def _streaming_read_image_patch(
431	    frame_loader, total: int, window: int = 1, start_index: int = 0
432	):
433	    """Scope a shim over WASB's ``read_image`` so ``ImageDataset`` is fed in-memory decoded
434	    frames during the detector pass, byte-for-byte as it would over real PNGs.
435	
436	    WASB's ``ImageDataset.__getitem__`` reads each frame via ``read_image(path)`` — NOT
437	    ``cv2.imread``. ``read_image`` (``utils.utils``) does ``Image.open(path).convert('RGB')``,
438	    returning a PIL **RGB** image, after an ``osp.exists(path)`` guard. We therefore feed it a
439	    PIL RGB image built from the streamed BGR frame (see ``_bgr_to_pil_rgb``), which is
440	    byte-identical to the disk round-trip, and the shim never touches the filesystem so the
441	    ``osp.exists`` guard is sidestepped for synthetic stream paths.
442	
443	    We patch ``dataloaders.dataset_loader.read_image`` — the binding the consumer actually
444	    calls (``dataset_loader`` does ``from utils import read_image``, so the name lives in that
445	    module's namespace; patching ``utils.read_image`` would NOT rebind the already-imported
446	    reference).
447	
448	    ``frame_loader(index) -> BGR uint8 ndarray`` is a forward-only sequential decoder wrapped
449	    in a ``SequentialFrameStore`` (sliding buffer sized to ``window`` = ``frames_in``); on a
450	    resume we start at ``start_index`` so the decoder seeks past already-cached windows. The
451	    original reader is restored on exit.
452	    """
453	    from dataloaders import dataset_loader as _dl
454	
455	    store = SequentialFrameStore(
456	        frame_loader, total, window=window, start_index=start_index
457	    )
458	    real_read_image = _dl.read_image
459	    # Rebind read_image in the consuming module for the duration of the detector pass;
460	    # restore on exit. (Plain assignment, not setattr-with-constant, to stay ruff B010-clean.)
461	    _dl.read_image = _make_streamed_read_image(store, real_read_image)  # type: ignore[assignment]
462	    try:
463	        yield store
464	    finally:
465	        _dl.read_image = real_read_image  # type: ignore[assignment]

471	class _PrefetchIter:
472	    """Bounded producer-thread wrapper over a batch iterator (the num_workers=0 DataLoader).
473	
474	    In streaming mode the DataLoader must stay single-process (the in-memory frame store +
475	    the cv2.imread shim live in this process and cannot be pickled to workers), so the
476	    CPU-side batch assembly (decode + PIL + ToTensor) SERIALIZES with GPU inference —
477	    measured on the first real L4 serving run (2026-07-03) this left the GPU <1% utilized.
478	    A single producer THREAD fixes the overlap without multiprocessing: cv2/PIL/numpy
479	    release the GIL for the heavy parts, the store's forward-only access pattern is
480	    preserved (one consumer thread iterating sequentially), and the bounded queue caps
481	    memory at ``depth`` batches. A producer exception is re-raised in the consumer."""
482	
483	    _END = object()
484	
485	    def __init__(self, src, depth: int = 2):
486	        import queue
487	        import threading
488	
489	        self._q: "queue.Queue" = queue.Queue(maxsize=max(1, int(depth)))
490	        self._exc: "BaseException | None" = None
491	
492	        def _fill() -> None:
493	            try:
494	                for item in src:
495	                    self._q.put(item)
496	            except BaseException as e:  # noqa: BLE001 - surfaced in __next__
497	                self._exc = e
498	            finally:
499	                self._q.put(self._END)
500	
501	        self._t = threading.Thread(target=_fill, name="wasb-prefetch", daemon=True)
502	        self._t.start()
503	
504	    def __iter__(self):
505	        return self
506	
507	    def __next__(self):
508	        item = self._q.get()
509	        if item is self._END:
510	            if self._exc is not None:
511	                raise self._exc
512	            raise StopIteration
513	        return item

783	def extract_frames(video_path: str, frames_dir: str) -> str:
784	    """Decode a video to PNG frames named by source frame index (00000000.png ...).
785	
786	    Idempotent: if the dir already holds the expected number of frames (matching the
787	    container's reported frame count) we skip re-decoding — re-extraction after a
788	    reboot is wasteful. NOTE: cv2 PNG extraction is slow; for big clips prefer
789	    extracting with ffmpeg on the host and passing --frames_dir.
790	    """
791	    import cv2
792	
793	    os.makedirs(frames_dir, exist_ok=True)
794	    cap = cv2.VideoCapture(video_path)
795	    if not cap.isOpened():
796	        raise SystemExit(f"cannot open video: {video_path}")
797	    expected = int(cap.get(cv2.CAP_PROP_FRAME_COUNT) or 0)
798	    existing = len(glob.glob(osp.join(frames_dir, "*.png")))
799	    if expected > 0 and existing >= expected:
800	        cap.release()
801	        logger.info(f"reusing {existing} already-extracted frames -> {frames_dir}")
802	        return frames_dir
803	    i = 0
804	    while True:
805	        ok, frame = cap.read()
806	        if not ok:
807	            break
808	        cv2.imwrite(osp.join(frames_dir, f"{i:08d}.png"), frame)
809	        i += 1
810	    cap.release()
811	    logger.info(f"extracted {i} frames -> {frames_dir}")
812	    return frames_dir

827	def make_video_frame_loader(video_path: str):
828	    """Return ``(frame_loader, count_hint, release)`` for output-preserving streaming.
829	
830	    ``frame_loader(index) -> BGR uint8 ndarray`` decodes the video forward-only with one
831	    long-lived ``cv2.VideoCapture``. It MUST be called with non-decreasing indices (the
832	    detector's sliding-window order); it reads sequentially and only uses a frame-seek to
833	    skip forward when a *resume* starts past the current position. Each ``cap.read()``
834	    returns exactly the BGR uint8 array that ``cv2.imwrite``/``cv2.imread`` of a PNG would
835	    yield (PNG is lossless), so the detector sees identical pixels to the disk path.
836	
837	    Returns the count hint from the container so the caller can build the frame list;
838	    ``release`` closes the capture.
839	    """
840	    import cv2
841	
842	    cap = cv2.VideoCapture(video_path)
843	    if not cap.isOpened():
844	        raise SystemExit(f"cannot open video: {video_path}")
845	    count_hint = int(cap.get(cv2.CAP_PROP_FRAME_COUNT) or 0)
846	    state = {"pos": 0}  # next index cap.read() will return
847	
848	    def frame_loader(index: int):
849	        index = int(index)
850	        if index < state["pos"]:
851	            raise RuntimeError(
852	                f"streaming decode is forward-only; got index {index} after {state['pos']}"
853	            )
854	        if index > state["pos"]:
855	            # Forward skip only happens when resuming past cached windows. Seek once.
856	            cap.set(cv2.CAP_PROP_POS_FRAMES, index)
857	            state["pos"] = index
858	        ok, frame = cap.read()
859	        if not ok:
860	            raise RuntimeError(f"failed to decode frame {index} from {video_path}")
861	        state["pos"] += 1
862	        return frame
863	
864	    def release():
865	        cap.release()
866	
867	    return frame_loader, count_hint, release
