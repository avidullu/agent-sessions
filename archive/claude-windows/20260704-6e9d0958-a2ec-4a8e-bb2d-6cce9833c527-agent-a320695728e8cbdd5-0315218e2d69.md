def is_full_fusion_feature_file(path: str) -> bool:
    """True when an existing golden feature JSON already has shuttle+person cue dicts.

    Corpus-growth runs can leave leaner windowing-only JSONs with the same filename pattern.
    Those are useful for gts/windowing analysis, but they are not the replayable fusion
    substrate that ``fusion_compare`` and ``ablation`` need.
    """

    try:
        with open(path, encoding="utf-8") as f:
            data = json.load(f)
    except (OSError, json.JSONDecodeError, TypeError):
        return False
    candidates = data.get("candidates")
    if not isinstance(candidates, list):
        return False
    if not candidates:
        return True
    return all(
        isinstance(candidate, dict)
        and "shuttle" in candidate
        and "person" in candidate
        for candidate in candidates
    )


def load_source_video_paths(source_manifest_path: Optional[str]) -> Dict[str, str]:
    """Load ``video_id -> local_path`` from the collector golden-source manifest.

    The collector records source/proxy ownership and full-file MD5s. Only entries marked
    ``present_local`` with a concrete ``local_path`` can drive local GPU extraction.
    """

    if not source_manifest_path:
        return {}
    with open(source_manifest_path, encoding="utf-8") as f:
        data = json.load(f)
    entries = data.get("videos", data) if isinstance(data, dict) else data
    if not isinstance(entries, list):
        raise ValueError(f"source manifest must contain a videos list: {source_manifest_path}")

    paths: Dict[str, str] = {}
    for entry in entries:
        if not isinstance(entry, dict):
            continue
        video_id = entry.get("video_id") or entry.get("name")
        local_path = entry.get("local_path")
        if not video_id or not local_path:
            continue
        if entry.get("source_status", "present_local") != "present_local":
            continue
        paths[str(video_id)] = str(local_path)
    return paths


def attach_source_video_paths(
    specs: Sequence[Dict[str, Any]], source_manifest_path: Optional[str]
) -> int:
    """Attach explicit ``video_path`` fields to manifest specs from the source manifest.

    Returns the number of specs updated. The collector source manifest is authoritative
    for local source/proxy video location and overrides derived/stale manifest paths.
    """

    source_paths = load_source_video_paths(source_manifest_path)
    updated = 0
    for spec in specs:
        name = str(spec.get("name", ""))
        source_path = source_paths.get(name)
        if not source_path:
            continue
        if spec.get("video_path") == source_path:
            continue
        spec["video_path"] = source_path
        updated += 1
    return updated


def _filter_specs(
    specs: Sequence[Dict[str, Any]], only: Sequence[str]
) -> List[Dict[str, Any]]:
    if not only:
        return list(specs)
    wanted = list(dict.fromkeys(only))
    wanted_set = set(wanted)
    filtered = [spec for spec in specs if str(spec.get("name")) in wanted_set]
    found = {str(spec.get("name")) for spec in filtered}
    missing = [name for name in wanted if name not in found]
    if missing:
        raise ValueError(f"--only requested unknown video(s): {', '.join(missing)}")
    return filtered


def _derive_video_path(spec: Dict[str, Any]) -> str:
    """Original MP4 for a manifest entry — explicit `video_path`, else derived from the
    golden labels path (``X.rallies.csv`` -> ``X.MP4``, the golden-set naming convention)."""
    if spec.get("video_path"):
        return str(spec["video_path"])
    labels = str(spec["labels"])
    if labels.endswith(".rallies.csv"):
        return labels[: -len(".rallies.csv")] + ".MP4"
    raise ValueError(
        f"{spec.get('name')}: no video_path and labels {labels!r} isn't *.rallies.csv"
    )
