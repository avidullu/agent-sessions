197	  const startedAt = Date.now();

199	  for (;;) {

201	    if (job.status === "done") {
202	      const result = (job.result ?? {}) as CompileResult;
203	      if (result.status === "failed" || !result.download_url) {

212	      const msg = job.error || "Compile job failed.";
