428	    def export_eval_set(self, args):
429	        """Export curated annotations as indexer-ready eval/training sets.
430	
431	        Emits, per qualifying video, a `<out_dir>/<sport>/<video_id>.csv` in the

433	        `<out_dir>/manifest.json` that pairs each CSV with its video file. The
434	        manifest is the contract the downstream eval/training driver reads: the
435	        indexer keys videos by *file MD5*, not by our `video_id`, so it must run
436	        against the exact `local_path`/`video_url` recorded here.
437	
438	        By default only CURATED + commercially-safe videos are exported, mirroring
439	        the license gate already enforced elsewhere in the toolchain.
440	        """
441	        out_dir = args.out_dir

443	        statuses = [CurationStatus.CURATED]

445	            statuses.append(CurationStatus.STAGING)

447	        supported = self.config.get("supported_sports", ["badminton"])
448	        sport_strs = [args.sport] if args.sport else supported

503	            "total_videos": len(manifest),

506	        }
507	        manifest_path = os.path.join(out_dir, "manifest.json")
508	        with open(manifest_path, "w", encoding="utf-8") as f:
509	            json.dump(manifest_doc, f, indent=2)

511	        print(f"\n=== Exported eval/training set to '{out_dir}' ===")
512	        if manifest:
513	            summary = [[m["sport"], m["video_id"], m["num_segments"],
514	                        "YES" if m["is_commercial"] else "NO",
515	                        os.path.basename(m["local_path"]) if m["local_path"] else "(no local file)"]
516	                       for m in manifest]
517	            print(tabulate(summary, headers=["Sport", "Video ID", "Segments", "Comm?", "Video file"], tablefmt="grid"))
518	        print(f"\nVideos exported : {len(manifest)}")
519	        print(f"Total segments  : {manifest_doc['total_segments']}")
520	        if skipped_noncomm:
521	            print(f"Skipped (non-commercial) : {skipped_noncomm}  (pass --include-noncommercial to include)")
522	        if skipped_empty:

543	        except (FileNotFoundError, ValueError) as e:
544	            print(f"Error: {e}")
545	            sys.exit(1)
546	
547	        write_canonical_csv(result.records, args.out)
548	        report = render_issue_report(result, os.path.basename(args.input))
549	        if args.issues:

551	                f.write(report)
552	        print(report)
553	        print(f"Canonical CSV written: {args.out}  ({len(result.records)} rallies)")
554	        if result.n_blockers:
555	            print(f"\n  {result.n_blockers} BLOCKER issue(s) — import-local will reject "
556	                  f"this file until they are resolved (by the vendor or manual review).")
557	        else:
558	            print("\n  No blockers. Next:")
559	            print(f"    python -m src.cli.curate import-local --sport <sport> "

563	        """Run the full validation suite on a rally delivery and emit a report.
564	
565	        Accepts a CVAT export (--input + --video-path/--fps) or an already-canonical
566	        CSV (--csv). Writes, into --out-dir:
567	          <id>_report.md    human report (scoreboard + issues + rallies to review)
568	          <id>_report.json  machine summary (for CI / a bulk-ingest loop)
569	          <id>_review.csv   the annotatable manual-verification sheet
570	        Exits non-zero if any blocker issue exists, so it can gate a bulk pipeline.
571	        """
572	        import json
573	        from src.parsers.cvat.rally_cvat import (
574	            convert, load_canonical_csv, detect_issues,
575	            build_summary, render_validation_md, write_review_csv,
576	        )
577	        vid = args.video_id or "delivery"
578	        fps = None
579	        if args.input:
580	            try:
581	                result = convert(args.input, fps=args.fps, video_path=args.video_path)
582	            except (FileNotFoundError, ValueError) as e:
583	                print(f"Error: {e}")
584	                sys.exit(1)
585	            records, issues, fps = result.records, result.issues, result.fps
586	        elif args.csv:
587	            try:
588	                records = load_canonical_csv(args.csv)

609	            f.write(md)
610	        write_review_csv(records, issues, review_path)
611	
612	        print(md)
613	        print(f"\nWrote:\n  {md_path}\n  {json_path}\n  {review_path}")
614	        if summary["n_blockers"]:
615	            print(f"\n  GATE: BLOCKED — {summary['n_blockers']} blocker(s) must be resolved.")
616	            sys.exit(2)

  def _mmss(seconds: float) -> str:
      m = int(seconds // 60)
      s = seconds - m * 60
