def parse_size_mb(value_str: str, unit: str) -> int:
    """Parse a size value + unit string into mebibytes.

    >>> parse_size_mb("10", "MB")
    10
    >>> parse_size_mb("1", "GB")
    1024
    >>> parse_size_mb("500", "KB")
    1

    Note: returns the raw arithmetic result; callers must enforce
    ``FilterConfig`` domain constraints (e.g. ``max_size_mb >= 1``).
    """
    value = int(value_str)
    unit = unit.lower().rstrip("b")  # "MB" → "m", "KB" → "k", "GB" → "g"
    multipliers: dict[str, int] = {"": 1, "m": 1, "k": 1, "g": 1024, "t": 1024 * 1024}
    if unit in multipliers:
        raw = value * multipliers[unit]
        if unit == "k":
            raw = max(1, raw // 1024)
        return raw
    raise ValueError(f"Unknown size unit: {unit}")


def parse_duration_s(value_str: str, unit: str) -> int:
    """Parse a duration value + unit string into seconds.

    >>> parse_duration_s("15", "m")
    900
    >>> parse_duration_s("2", "h")
    7200
    >>> parse_duration_s("90", "s")
    90

    Note: returns the raw arithmetic result; callers must enforce
    ``FilterConfig`` domain constraints (e.g. max >= min).
    """
    value = int(value_str)
    unit = unit.lower()
    multipliers: dict[str, int] = {"": 1, "s": 1, "m": 60, "h": 3600}
    if unit in multipliers:
        return value * multipliers[unit]
    raise ValueError(f"Unknown time unit: {unit}")

def _format_duration(seconds: int | None) -> str:
    """Format seconds as a human-readable duration string."""
    if seconds is None:
        return "unlimited"
    if seconds == 0:
        return "0s"
    if seconds < 60:
        return f"{seconds}s"
    if seconds < 3600:
        m, s = divmod(seconds, 60)
        return f"{m}m{s}s" if s else f"{m}m"
    h, rem = divmod(seconds, 3600)
    m, s = divmod(rem, 60)
    parts = f"{h}h"
    if m:
        parts += f"{m}m"
    if s:
        parts += f"{s}s"
    return parts


def _format_size(mb: int | None) -> str:
    """Format mebibytes as a human-readable size string."""
    if mb is None:
        return "unlimited"
    if mb == 0:
        return "0MB"
    if mb >= 1024:
        gb = mb / 1024
        if gb == int(gb):
            return f"{int(gb)}GB"
        return f"{gb:.1f}GB"
    return f"{mb}MB"
