145	    """Run one statement through the full parse pipeline into an in-memory DB.
146	
147	    Returns ``(outcome, row_count, parser_name)`` where ``outcome`` is the document status
148	    (``reconciled`` / ``unreconciled``) or a failure tag (``locked`` / ``no-parser`` / ``error``).
149	    Nothing is persisted and no statement value is surfaced — only the verdict and counts.
150	    """
151	    from muneem import db as dbmod
152	    from muneem.extract import extract
153	    from muneem.parsers import get_parser
154	    from muneem.statement_unlock import resolve_password
155	    from muneem.unlock import is_encrypted
156	
157	    password = ""
158	    if is_encrypted(pdf):
159	        password = resolve_password(pdf, interactive=interactive) or ""
160	        if not password:
161	            return ("locked", 0, None)
162	    try:
163	        ex = extract(pdf, password=password)
164	        parser = get_parser(ex.text)
165	    except Exception as exc:  # eval must not crash on one bad file; record the redacted reason
166	        _log.warning(
167	            "eval: extract/route failed for %s (%s)", redact_filename(pdf), type(exc).__name__
168	        )
169	        return ("error", 0, None)
170	    if parser is None:
171	        return ("no-parser", 0, None)
172	    name = parser.__class__.__name__
173	    try:
174	        with dbmod.connect(":memory:") as conn:
175	            dbmod.init_db(conn)
176	            doc_id = dbmod.insert_document(
177	                conn, source="file", external_id=pdf.name, sha256=ex.sha256, status="parsed"
178	            )

180	            doc = dbmod.get_document_by_sha(conn, ex.sha256)
