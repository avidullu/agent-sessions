194	    """Check the proxy carries the source's exact frame↔time identity.

199	    still catches any real trim.

202	
203	    if src.nb_frames is None or proxy.nb_frames is None:
204	        errors.append("frame count unavailable on source and/or proxy — cannot prove parity")
205	    elif src.nb_frames != proxy.nb_frames:
206	        errors.append(f"frame count changed: source {src.nb_frames} -> proxy {proxy.nb_frames}")

209	        errors.append("fps unavailable on source and/or proxy — cannot prove parity")
210	    elif abs(proxy.avg_fps - src.avg_fps) / src.avg_fps > 0.001:
211	        errors.append(f"fps changed: source {src.avg_fps:.3f} -> proxy {proxy.avg_fps:.3f}")

214	        errors.append("duration unavailable on source and/or proxy — cannot prove parity")
