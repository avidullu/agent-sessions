1	"""Redact secret-shaped values before they leave the process (PACKAGING_PLAN.md P0b).
2	
3	``GET /api/config`` returns the whole ``AppConfig`` unauthenticated. Today AppConfig stores NO
4	secrets by design (the Gemini key lives in the environment; ``cf_access_aud`` is a public
5	identifier), so this is DEFENSIVE: if a credential-shaped field is ever added to the config, this
6	masks it in the response and the accompanying test fails loudly rather than leaking it to any
7	localhost process/browser tab.
8	
9	The match list is deliberately NARROW so it never masks legitimate non-secret fields such as
10	``edge_auth_enabled`` (a boolean), ``cf_access_aud`` (a public id), or an ML ``max_tokens`` knob.
11	"""
12	
13	from __future__ import annotations
14	
15	import re
16	from typing import Any
17	
18	#: Field-name patterns whose VALUE is masked. Narrow on purpose (see module docstring).
19	SECRET_KEY_RE = re.compile(
20	    r"(password|passwd|secret|credential|api[_-]?key|access[_-]?token|auth[_-]?token|refresh[_-]?token)",
21	    re.IGNORECASE,
22	)
23	
24	REDACTED = "***REDACTED***"
25	
26	
27	def redact_secrets(obj: Any) -> Any:
28	    """Return a deep copy of ``obj`` with any value under a secret-shaped KEY replaced by
29	    :data:`REDACTED`. Recurses into dicts and lists; scalars and non-secret keys pass through.
30	
31	    A secret-named key masks its WHOLE value (even a nested dict/list — e.g. a ``credentials``
32	    block), so a structured secret can't slip through via recursion.
33	    """
34	    if isinstance(obj, dict):
35	        out: dict[Any, Any] = {}
36	        for key, value in obj.items():
37	            if isinstance(key, str) and SECRET_KEY_RE.search(key):
38	                out[key] = REDACTED
39	            else:
40	                out[key] = redact_secrets(value)
41	        return out
42	    if isinstance(obj, (list, tuple)):
43	        return [redact_secrets(item) for item in obj]
44	    return obj
