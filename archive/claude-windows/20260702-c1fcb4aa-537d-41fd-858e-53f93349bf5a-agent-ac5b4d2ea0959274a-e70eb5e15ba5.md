become a CLEAN validation failure (→ the job fails with a reason), never a hang or a

import subprocess
from unittest.mock import patch

from backend.config.models import AppConfig
from backend.pipeline.validators.format import FormatValidator
from backend.pipeline.validators.pts_continuity import PTSContinuityValidator
from backend.pipeline.validators.resolution_fps import (
    ResolutionFPSValidator,
    _probe_meta,
)
from backend.utils.ffmpeg import FFPROBE_TIMEOUT_SEC


def _timeout_run(cmd, *a, **k):
    raise subprocess.TimeoutExpired(cmd, k.get("timeout", FFPROBE_TIMEOUT_SEC))


def test_format_validator_probe_carries_timeout(tmp_path):
    video = tmp_path / "v.mp4"
    video.write_bytes(b"x")
    captured = {}

    def fake_run(cmd, *a, **k):
        captured["timeout"] = k.get("timeout")
        raise subprocess.CalledProcessError(1, cmd, stderr="bad")

    with patch.object(subprocess, "run", side_effect=fake_run):
        FormatValidator().validate(str(video), AppConfig(), {})
    assert captured["timeout"] == FFPROBE_TIMEOUT_SEC


def test_format_validator_timeout_is_clean_distinct_failure(tmp_path):
    video = tmp_path / "v.mp4"
    video.write_bytes(b"x")
    with patch.object(subprocess, "run", side_effect=_timeout_run):
        res = FormatValidator().validate(str(video), AppConfig(), {})
    assert res.passed is False
    assert "timed out" in res.message
