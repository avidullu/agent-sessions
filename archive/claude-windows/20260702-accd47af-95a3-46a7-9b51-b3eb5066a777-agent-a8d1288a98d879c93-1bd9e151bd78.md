624	def _resolve_compile(
625	    db, cfg, video_id: str, segment_ids: List[int], output_filename: str
626	):
627	    """Shared compile resolution — run at SUBMIT (fast-fail 4xx) AND in the WORKER (re-resolve, since a
628	    row could change between submit and run). Verifies the video exists, the requested segments match
629	    and survive the ``stitching.min_shot_count`` floor (segments WITHOUT a known shot_count are exempt
630	    so manual/legacy picks can't silently vanish), and the output name is traversal-safe. Raises
631	    ``_CompileError(404/400)``. Returns ``(video_row, kept_segments, safe_out_name)``."""
632	    video = db.get_video(video_id)
633	    if not video:
634	        raise _CompileError(404, "Indexed video record not found.")
635	    # Reject path traversal / absolute names (os.path.join would otherwise write anywhere on disk).
636	    try:
637	        out_name = safe_output_filename(output_filename)
638	    except ValueError as e:
639	        raise _CompileError(400, str(e)) from e
640	
641	    all_segments = db.query_play_segments(video_id, {})
642	    matched = [s for s in all_segments if s["id"] in segment_ids]
643	    if not matched:
644	        raise _CompileError(400, "No matching play segments found to stitch.")
645	
646	    stitching = getattr(cfg, "stitching", None) or StitchingConfig()
647	    kept = []
648	    for s in matched:
649	        meta = s.get("metadata") or {}
650	        sc = meta.get("shot_count") if isinstance(meta, dict) else None
651	        try:
652	            if sc is not None and int(sc) < stitching.min_shot_count:
653	                continue
654	        except (TypeError, ValueError):
655	            pass  # unparseable count -> treat as unknown, keep
656	        kept.append(s)
657	    if len(kept) < len(matched):
658	        logger.info(
659	            "[compile] stitching.min_shot_count=%s: dropped %d/%d segments below the floor.",
660	            stitching.min_shot_count,
661	            len(matched) - len(kept),
662	            len(matched),
663	        )
664	    if not kept:
665	        raise _CompileError(
666	            400,
667	            f"All {len(matched)} matching segments fall below "
668	            f"stitching.min_shot_count={stitching.min_shot_count}.",
669	        )
