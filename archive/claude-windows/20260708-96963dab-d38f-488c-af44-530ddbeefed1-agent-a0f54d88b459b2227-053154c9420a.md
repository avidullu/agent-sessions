57	    """Parse a storage URI (or a bare filesystem path) into a ``StorageRef``.
58	
59	    A string with no ``scheme://`` prefix (including Windows ``C:\\...`` paths, which
60	    have no ``//``) is treated as a local filesystem path — preserving today's behaviour.
61	    """
62	    s = str(uri)
63	    m = _SCHEME_RE.match(s)
64	    if not m:
65	        return StorageRef("local", "", s, f"local://{s}")
66	    scheme = m.group(1).lower()
67	    rest = s[m.end() :]
68	    if scheme == "gs":
69	        scheme = "gcs"
70	    if scheme in ("local", "gdrive", "gdrive-api"):
71	        # Path-keyed, no bucket: local FS, a path under the mounted Drive, or a path under a Drive-API
72	        # root folder (gdrive-api resolves the path → a Drive file id).
73	        return StorageRef(scheme, "", rest, s)
74	    if scheme in _KNOWN:
75	        bucket, _, key = rest.partition("/")
76	        return StorageRef(scheme, bucket, key, s)

80	def resolve_input_uri(

82	) -> str:

84	
85	    Precedence (first match wins):
86	      1. An explicit ``scheme://`` on ``raw`` (``gs://``, ``s3://``, ``local://``, ``gdrive://``) → as-is.
87	      2. An ABSOLUTE local path (``/data/x.mp4``, ``C:\\…``, ``C:/…``) → as-is (always local).
88	      3. A bare/relative name + a non-empty ``input_root`` → joined under it
89	         (``"gs://bucket/videos"`` + ``"x.mp4"`` → ``"gs://bucket/videos/x.mp4"``).
90	      4. A bare/relative name + ``input_backend != "local"`` → prefixed (``"gcs"`` + ``"bucket/x"`` →
91	         ``"gcs://bucket/x"``).
92	      5. Otherwise → ``raw`` unchanged (a relative/bare LOCAL path — today's behaviour).
93	
94	    So the default (``input_backend="local"``, ``input_root=""``) is an identity passthrough for every
95	    path used today, preserving byte-for-byte behaviour.
96	    """
97	    s = str(raw)
98	    if _SCHEME_RE.match(s):  # 1. explicit scheme wins
99	        return s
100	    # 2. an anchored LOCAL path is always local, never joined. Detect this OS-INDEPENDENTLY —
101	    #    os.path.isabs is platform- AND Python-version-dependent (on Windows + py3.13 a POSIX-style
102	    #    "/data/x" is NOT absolute). Anchored == a leading "/" or "\\", or a Windows drive prefix "X:".
103	    if s.startswith(("/", "\\")) or (len(s) >= 2 and s[1] == ":" and s[0].isalpha()):
104	        return s
105	    if input_root:  # 3. join a relative name under the configured root
106	        return input_root.rstrip("/") + "/" + s.lstrip("/")
107	    if (
108	        input_backend and input_backend != "local"
109	    ):  # 4. prefix a bucket/key with the declared backend
110	        return f"{input_backend}://{s}"
111	    return s  # 5. bare/relative local path — unchanged
