def _parse_time(val) -> float:
    """Converts a timestamp value to total float seconds.
    Handles float/int pass-through, plain numeric strings, and
    colon-separated formats (MM:SS, HH:MM:SS). Logs warnings
    when unexpected formats are encountered."""
    if isinstance(val, (int, float)):
        return float(val)
    if isinstance(val, str):
        val = val.strip()
        if ":" in val:
            # Gemini returned MM:SS or HH:MM:SS despite being asked for decimal seconds.
            # Convert it, but warn so the issue is visible in logs.
            parts = val.split(":")
            parts.reverse()
            total = 0.0
            for i, p in enumerate(parts):
                try:
                    total += float(p) * (60 ** i)
                except ValueError:
                    pass
            logger.warning(f"Timestamp '{val}' was in colon-separated format (MM:SS/HH:MM:SS) instead of decimal seconds. Converted to {total:.2f}s.")
            return total
        try:
            return float(val)
        except ValueError:
            logger.warning(f"Could not parse timestamp string '{val}' as a number. Defaulting to 0.0.")
            return 0.0
    logger.warning(f"Unexpected timestamp type {type(val).__name__} for value '{val}'. Defaulting to 0.0.")
    return 0.0

            "=" * 60,

    def _validate_and_dedupe(self, ai_segments: List[Dict[str, Any]], video_duration: float) -> List[Dict[str, Any]]:
        """Parse timestamps, validate/clamp candidates, and resolve overlaps (keep longer)."""
        logger.info(f"Processing {len(ai_segments)} play segments through validation...")

        # 7. Parse timestamps and build candidate segment list
        idx_cfg = self.config.get("indexing", {})
        min_segment_duration = idx_cfg.get("min_segment_duration", 3.0)
        max_segment_duration = 90.0  # Singles play segments almost never exceed this

        candidates = []
        for idx, segment in enumerate(ai_segments):
            start = _parse_time(segment.get("start_time", 0.0))
            end = _parse_time(segment.get("end_time", 0.0))
            if start >= end:
                logger.info(f"Skipping segment #{idx+1}: start_time ({start:.2f}s) >= end_time ({end:.2f}s)")
                continue
            candidates.append({
                "idx": idx,
                "start": start,
                "end": end,
                "raw": segment
            })

        # 8. Post-hoc validation pass
        logger.info(f"Running validation on {len(candidates)} candidate play segments...")
        validated = []
        rejected = []

        for c in candidates:
            label = f"Segment #{c['idx']+1} [{c['start']:.2f}s - {c['end']:.2f}s]"

            # A: Reject segments that start beyond the video
            if video_duration > 0 and c["start"] >= video_duration:
                rejected.append((label, f"start_time ({c['start']:.2f}s) is beyond video duration ({video_duration:.1f}s)"))
                continue

            # B: Clamp end_time to video duration
            if video_duration > 0 and c["end"] > video_duration:
                logger.info(f"{label}: end_time ({c['end']:.2f}s) exceeds video duration ({video_duration:.1f}s). Clamping.")
                c["end"] = video_duration

            dur = c["end"] - c["start"]

            # C: Reject segments shorter than min_segment_duration
            if dur < min_segment_duration:
                rejected.append((label, f"duration ({dur:.2f}s) is below minimum ({min_segment_duration}s)"))
                continue

            # D: Flag suspiciously long segments
            if dur > max_segment_duration:
                logger.warning(
                    f"{label}: duration ({dur:.1f}s) is unusually long (>{max_segment_duration}s). "
                    f"This may be a detection error. Keeping but flagging."
                )

            validated.append(c)

        # E: Resolve overlapping segments (keep the longer one)
        if len(validated) > 1:
            # Sort by start time
            validated.sort(key=lambda r: r["start"])
            non_overlapping = [validated[0]]
            for curr in validated[1:]:
                prev = non_overlapping[-1]
                # Check if current segment overlaps with the previous one
                if curr["start"] < prev["end"]:
                    prev_dur = prev["end"] - prev["start"]
                    curr_dur = curr["end"] - curr["start"]
                    kept, dropped = (prev, curr) if prev_dur >= curr_dur else (curr, prev)
                    kept_label = f"Segment #{kept['idx']+1} [{kept['start']:.2f}s - {kept['end']:.2f}s]"
                    dropped_label = f"Segment #{dropped['idx']+1} [{dropped['start']:.2f}s - {dropped['end']:.2f}s]"
                    rejected.append((dropped_label, f"overlaps with {kept_label} — kept the longer segment"))
                    if kept == curr:
                        non_overlapping[-1] = curr  # Replace previous with current
                else:
                    non_overlapping.append(curr)
            validated = non_overlapping

        # Log validation summary
        if rejected:
            reject_lines = [f"  ✗ {label}: {reason}" for label, reason in rejected]
            logger.info(f"{len(rejected)} play segments were REJECTED:\n" + "\n".join(reject_lines))
        logger.info(f"{len(validated)} play segments passed validation (out of {len(ai_segments)} from Gemini).")

        stitch_cfg = self.config.get("stitching", {})
        begin_pad = stitch_cfg.get("begin_padding", 0.5)
        end_pad = stitch_cfg.get("end_padding", 0.5)

        total_highlight_dur = 0.0
        for c in validated:
            total_highlight_dur += (c["end"] - c["start"]) + begin_pad + end_pad

        mins = int(total_highlight_dur // 60)
        secs = int(total_highlight_dur % 60)
        logger.info(
            f"Estimated final highlight video duration: {mins:02d}:{secs:02d} "
            f"({total_highlight_dur:.1f}s total; includes {begin_pad}s begin + {end_pad}s end padding per segment)"
        )

        return validated

    def _populate_db(self, video_id: str, validated: List[Dict[str, Any]], model_name: str) -> List[Dict[str, Any]]:
        """9. Populate SQLite DB with validated play segments; returns segments_data."""
        segments_data = []
        for c in validated:
            start = c["start"]
            end = c["end"]
            segment = c["raw"]
            dur = end - start

            # Extract shot_count from Gemini or compute estimate if not present
            shot_count = segment.get("shuttle_exchange_count")
            if shot_count is None:
                shot_count = max(3, int(dur / 0.8))  # 1 shot every 0.8 seconds as fallback estimate
            else:
                try:
                    shot_count = int(shot_count)
                except Exception:
                    shot_count = max(3, int(dur / 0.8))

            # Only store metrics we actually know — no fabricated data
            metadata = {
                "shot_count": shot_count,
                "shot_count_confidence": segment.get("shot_count_confidence", "Unknown"),
                "detector": model_name
            }

            segment_id = self.db.add_play_segment(video_id, start, end, None, None, metadata)

            if segment_id:
                segments_data.append({
                    "id": segment_id,
                    "start_time": start,
                    "end_time": end,
                    "duration": dur,
                    "metadata": metadata
                })

        return segments_data

        # 2. Decide whether to use chunked processing

        chunk_duration = idx_cfg.get("chunk_duration", 300)    # 5 minutes default

        use_chunking = video_duration > 0 and video_duration > chunk_duration

        segments_data = self._populate_db(video_id, validated, model_name)

        return segments_data
