55	    """Parse a size value + unit string into mebibytes."""
56	    value = int(value_str)
57	    unit = unit.lower().rstrip("b")
58	    multipliers: dict[str, int] = {"": 1, "m": 1, "k": 1, "g": 1024, "t": 1024 * 1024}
59	    if unit in multipliers:
