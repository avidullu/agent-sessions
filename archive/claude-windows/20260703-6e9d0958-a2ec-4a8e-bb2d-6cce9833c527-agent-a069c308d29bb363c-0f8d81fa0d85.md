293	def _preset_comparable(base_preset: Any, cur_preset: Any) -> bool:
294	    """Are the two stored preset dicts the same windowing definition? (name is cosmetic)."""
295	    if not isinstance(base_preset, dict) or not isinstance(cur_preset, dict):
296	        return base_preset == cur_preset

298	    return all(base_preset.get(k) == cur_preset.get(k) for k in keys if k != "name")

301	def compare_config(
302	    name: str, base_sum: Dict[str, Any], cur_sum: Dict[str, Any], tols: Tolerances
303	) -> ConfigComparison:
304	    """Compare one config's current summary to its baseline summary.
305	
306	    * If the preset definition or the net-crossings gate changed, the whole config is STALE
307	      (numbers aren't comparable) — flag it, skip gating.
308	    * If the corpus (video set) changed, the aggregate is not comparable (different videos) →
309	      skip the aggregate gate + flag added/removed, but STILL run the per-video gate on the
310	      INTERSECTION (a code regression on a shared video must still trip the wire).
311	    """
312	    cmp = ConfigComparison(config=name)
313	
314	    # 1) Config-definition drift → stale, not gateable.
315	    if not _preset_comparable(
316	        base_sum.get("preset"), cur_sum.get("preset")
317	    ) or base_sum.get("min_crossings") != cur_sum.get("min_crossings"):
318	        cmp.status = "stale"
319	        cmp.aggregate_gated = False
320	        cmp.findings.append(
321	            Finding(config=name, scope="config", metric="preset", kind="stale")
322	        )
323	        return cmp
324	
325	    base_pv = base_sum.get("per_video", {})
326	    cur_pv = cur_sum.get("per_video", {})
327	    base_names, cur_names = set(base_pv), set(cur_pv)
328	    cmp.added_videos = sorted(cur_names - base_names)
329	    cmp.removed_videos = sorted(base_names - cur_names)
330	    corpus_changed = bool(cmp.added_videos or cmp.removed_videos)
331	
332	    # 2) Aggregate gate (only when the corpus matches — means over different sets aren't comparable).
333	    if corpus_changed:
334	        cmp.aggregate_gated = False
335	        cmp.status = "stale"
336	        cmp.findings.append(
337	            Finding(config=name, scope="config", metric="corpus", kind="stale")
338	        )
339	    else:
340	        base_agg = base_sum.get("aggregate", {})
341	        cur_agg = cur_sum.get("aggregate", {})
342	        for m in GATED_HIGHER:
343	            b, c = base_agg.get(m), cur_agg.get(m)
344	            if b is None or c is None:
345	                continue
346	            if c < b - tols.metric_tol:
347	                cmp.findings.append(Finding(name, "aggregate", m, "regression", b, c))
348	            elif c > b + tols.metric_tol:
349	                cmp.findings.append(Finding(name, "aggregate", m, "improvement", b, c))
350	        for m in GATED_LOWER:
351	            b, c = base_agg.get(m), cur_agg.get(m)
352	            if b is None or c is None:
353	                continue
354	            if c > b + tols.rate_tol:
355	                cmp.findings.append(Finding(name, "aggregate", m, "regression", b, c))
356	            elif c < b - tols.rate_tol:
357	                cmp.findings.append(Finding(name, "aggregate", m, "improvement", b, c))
358	
359	    # 3) Per-video over-seg guard on the intersection (always — catches per-video code regressions).
360	    for vid in sorted(base_names & cur_names):
361	        for m in PER_VIDEO_GUARD:
362	            b, c = base_pv[vid].get(m), cur_pv[vid].get(m)
363	            if b is None or c is None:
364	                continue
365	            if c > b + tols.over_seg_tol:
366	                cmp.findings.append(
367	                    Finding(
368	                        name,
369	                        "per_video",
370	                        m,
371	                        "regression",
372	                        float(b),
373	                        float(c),
374	                        video=vid,
375	                    )
376	                )
377	
378	    if cmp.regressions:
379	        cmp.status = "regression"
380	    elif cmp.status != "stale":
381	        cmp.status = "ok"
382	    return cmp
