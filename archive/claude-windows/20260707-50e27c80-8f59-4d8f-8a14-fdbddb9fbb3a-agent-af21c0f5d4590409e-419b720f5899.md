377	+class NvdecResidentFrontend:
378	+    """On-GPU frame source for the ``nvdec_resident`` detector pass (#506).
379	+
380	+    PyNvVideoCodec ``SimpleDecoder(output_color_type=RGB)`` decodes on the GPU's
381	+    NVDEC engine; frames arrive as CUDA tensors via DLPack and never touch the host.
382	+    ``read_chw`` then applies the cv2-faithful affine warp (``build_affine_grid``)
383	+    and ImageNet-normalizes on-device, yielding the same ``[3, inp_h, inp_w]``
384	+    float32 tensor per frame that the ImageDataset path assembles on the CPU.
385	+    """
386	+
387	+    def __init__(self, video_path: str, inp_wh, gpu_id: int = 0):
388	+        import PyNvVideoCodec as nvc
389	+        import torch
390	+
391	+        self._torch = torch
392	+        self._device = torch.device("cuda", gpu_id)
393	+        inp_w, inp_h = int(inp_wh[0]), int(inp_wh[1])
394	+        try:
395	+            self._dec = nvc.SimpleDecoder(
396	+                video_path, gpu_id=gpu_id, output_color_type=nvc.OutputColorType.RGB
397	+            )
398	+        except Exception as e:
399	+            raise RuntimeError(
400	+                f"PyNvVideoCodec RGB SimpleDecoder failed for {video_path}: {e!r} — "
401	+                f"decode_backend=nvdec_resident needs PyNvVideoCodec with RGB output "
402	+                f"and libnvcuvid/libnvidia-encode on the box (deploy/cloudrun/Dockerfile "
403	+                f"/ gpu_setup.sh stage them). Fall back to decode_backend=cpu."
404	+            ) from e
405	+        self.count = len(self._dec)
406	+        probe = self._decode_rgb_hwc(0)
407	+        self.orig_h, self.orig_w = int(probe.shape[0]), int(probe.shape[1])
408	+        trans_inp_inv = _wasb_inverse_affines(
409	+            self.orig_h, self.orig_w, (inp_w, inp_h), [0]
410	+        )[0]
411	+        self._grid = build_affine_grid(
412	+            trans_inp_inv, inp_w, inp_h, self.orig_w, self.orig_h
413	+        ).to(self._device)
414	+        self._mean = torch.tensor(IMAGENET_MEAN, device=self._device).view(3, 1, 1)
415	+        self._std = torch.tensor(IMAGENET_STD, device=self._device).view(3, 1, 1)
416	+
417	+    def _decode_rgb_hwc(self, index: int):
418	+        """Frame ``index`` as an on-GPU HWC RGB uint8 tensor (zero-copy via DLPack)."""
419	+        torch = self._torch
420	+        t = torch.from_dlpack(self._dec[index])
421	+        if t.dim() == 3 and t.shape[0] == 3 and t.shape[-1] != 3:
422	+            t = t.permute(1, 2, 0)  # CHW layout variant → HWC
423	+        if t.dim() != 3 or t.shape[-1] != 3:
424	+            raise RuntimeError(
425	+                f"unexpected NVDEC frame shape {tuple(t.shape)} for frame {index} — "
426	+                f"expected 3-channel RGB (is output_color_type=RGB supported here?)"
427	+            )
428	+        return t
429	+
430	+    def read_chw(self, index: int):
431	+        """Decoded frame ``index`` → the normalized ``[3, inp_h, inp_w]`` float32 CUDA
432	+        tensor the detector expects (warpAffine-faithful resize + ImageNet normalize)."""
433	+        import torch.nn.functional as F
434	+
435	+        hwc = self._decode_rgb_hwc(index)
436	+        chw = hwc.permute(2, 0, 1).unsqueeze(0).float() / 255.0
437	+        warped = F.grid_sample(
438	+            chw, self._grid, mode="bilinear", align_corners=False, padding_mode="border"
439	+        ).squeeze(0)
440	+        return (warped - self._mean) / self._std
441	+
442	+    def release(self):
443	+        self._dec = None
