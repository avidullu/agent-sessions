344	@app.post("/api/process", status_code=202)
345	def process_video(req: ProcessRequest, request: Request):
346	    """Submit a processing job (202 + job_id). Poll GET /api/jobs/{job_id} for state.
347	
348	    Fast-fail checks (bad path, missing file, unknown segmenter) happen here so the
349	    client gets an immediate 4xx; everything slow — including MD5 hashing of the file —
350	    runs on the worker (DEFERRED_HARDENING_PLAN #16). The row is created 'queued'; the
351	    worker drains it via `claim_due_job` (EXECUTION_POLICY.md P3 self-scheduling loop).
352	    """
353	    # M9 (D9): edge-auth tenant resolution. DEFAULT-OFF → owner_id None (the local single-user,
354	    # unchanged). When edge_auth_enabled, the Cf-Access JWT is required + verified (a missing or
355	    # spoofed identity header fails here with 401, before any work) and tags the job's owner_id.
356	    try:
357	        owner_id = edge_auth.resolve_tenant(request.headers, global_config)
358	    except edge_auth.EdgeAuthError as e:
359	        raise HTTPException(status_code=401, detail=f"edge auth: {e}") from e
360	
361	    allowed_root = getattr(getattr(global_config, "security", None), "allowed_video_root", None)
362	    try:
363	        validate_input_path(req.video_path, allowed_root=allowed_root)
364	        # M2: resolve the input once (path or storage URI) — raises ValueError for an unsupported
365	        # scheme, which we map to a clean 400 (an unknown scheme must never 500).
366	        input_ref = storage.resolve_input_ref(req.video_path, getattr(global_config, "storage", None))
367	    except ValueError as e:
368	        raise HTTPException(status_code=400, detail=str(e)) from e
369	    # The local-existence fast-fail applies only to a LOCAL input, and stats the RESOLVED local path
370	    # (input_ref.key — e.g. local://… stripped to the bare path), not the raw URI string. A
371	    # bucket/Drive URI can't be cheaply os.path.exists()'d — its existence is verified when the worker
372	    # fetches it (a fetch miss fails the job loudly). validate_input_path above still runs for every
373	    # input: its null-byte guard always applies, and a configured allowed_video_root (hosted mode)
374	    # rejects a non-local URI as out-of-root.
375	    if input_ref.backend == "local" and not os.path.exists(input_ref.key):
376	        raise HTTPException(status_code=404, detail=f"Video file not found at '{req.video_path}'")
377	    if req.segmenter and not segmenter_registry.get(req.segmenter):
378	        available = list(segmenter_registry.list_available().keys())
379	        raise HTTPException(
380	            status_code=400,
381	            detail=f"Segmenter '{req.segmenter}' not found. Available segmenters: {available}"
382	        )
383	
384	    job_id = uuid.uuid4().hex
385	    if not db_instance.create_job(job_id, req.video_path, req.segmenter,
386	                                  req.player_pool, req.max_frames, owner_id=owner_id):
387	        raise HTTPException(status_code=500, detail="Failed to persist job record.")
388	    _get_executor().submit(_drain_due_jobs)
389	    return JSONResponse(
390	        status_code=202,
391	        content={"job_id": job_id, "status": "queued", "poll": f"/api/jobs/{job_id}"},
392	    )

489	@app.post("/api/compile")
490	def compile_highlights(req: CompileRequest):
491	    video = db_instance.get_video(req.video_id)
492	    if not video:
493	        raise HTTPException(status_code=404, detail="Indexed video record not found.")
494	
495	    # Retrieve matching play segments from db
496	    all_segments = db_instance.query_play_segments(req.video_id, {})
497	    matched_segments = [s for s in all_segments if s["id"] in req.segment_ids]
498	
499	    if not matched_segments:
500	        raise HTTPException(status_code=400, detail="No matching play segments found to stitch.")
501	
502	    # Reject path traversal / absolute output names (os.path.join would otherwise let an
503	    # absolute or '..'-laden output_filename write anywhere on disk).
504	    try:
505	        out_name = safe_output_filename(req.output_filename)
506	    except ValueError as e:
507	        raise HTTPException(status_code=400, detail=str(e)) from e
508	
509	    # The configured shot-count floor (stitching.min_shot_count) applies on the API
510	    # path too — parity with scripts/run_process_and_stitch.py. Segments WITHOUT shot_count
511	    # metadata are exempt: the floor filters known counts only, so explicitly
512	    # requested manual/legacy segments can't silently vanish under the default floor.
513	    stitching = getattr(global_config, "stitching", None) or StitchingConfig()
514	    kept = []
515	    for s in matched_segments:
516	        meta = s.get("metadata") or {}
517	        sc = meta.get("shot_count") if isinstance(meta, dict) else None
518	        try:
519	            if sc is not None and int(sc) < stitching.min_shot_count:
520	                continue
521	        except (TypeError, ValueError):
522	            pass  # unparseable count -> treat as unknown, keep
523	        kept.append(s)
524	    if len(kept) < len(matched_segments):
525	        logger.info(f"[compile] stitching.min_shot_count={stitching.min_shot_count}: dropped "
526	                    f"{len(matched_segments) - len(kept)}/{len(matched_segments)} segments below the floor.")
527	    if not kept:
528	        raise HTTPException(status_code=400,
529	                            detail=f"All {len(matched_segments)} matching segments fall below "
530	                                   f"stitching.min_shot_count={stitching.min_shot_count}.")
531	
532	    # Extract start/end time pairs
533	    time_pairs = [(s["start_time"], s["end_time"]) for s in kept]
534	
535	    # Run compiler with the configured stitching paddings + optional branding watermark
536	    output_dir = getattr(getattr(global_config, "output", None), "output_dir", "output") or "output"
537	    compiler = VideoCompiler(output_dir, begin_padding=stitching.begin_padding,
538	                             end_padding=stitching.end_padding, watermark=stitching.watermark)
539	    try:
540	        # The stored `filepath` is the SOURCE ref — a bucket/Drive URI (gs://, s3://, gdrive://) for a
541	        # video ingested via /api/process from a URI, or a plain local path. Resolve it to a LOCAL file
542	        # the stitcher can read: identity for a local path (byte-identical to before), fetch-to-scratch
543	        # for a bucket URI (the same seam the process path uses, _execute_processing main.py:159).
544	        # Without this, compiling a bucket-ingested video 500s — ffmpeg can't open a gs:// path.
545	        local_src = storage.acquire_local_input(video["filepath"], getattr(global_config, "storage", None))
546	        out_path = compiler.compile_highlights(local_src, time_pairs, out_name)
547	        # `filepath` is server-local (not reachable by the browser); also return the ready-to-use
548	        # download URL so the SPA needn't hand-build it (P6 / PER_RALLY_SELECTION.md §4.3).
549	        download_url = f"/api/highlights/{quote(req.video_id)}/{quote(out_name)}"
550	        return {"status": "success", "filepath": out_path, "download_url": download_url}
551	    except Exception as e:
552	        raise HTTPException(status_code=500, detail=f"Highlight stitching failed: {str(e)}") from e

555	def run_cli_diagnose_clip(video_path: str, timestamp: float):
556	    """CLI debugging utility to examine segment properties around a timestamp."""
557	    print("=" * 60)
558	    print(f"DEBUGGING VIDEO CLIP: {video_path} at {timestamp}s")
559	    print("=" * 60)
560	
561	    cfg = load_config()  # legacy wrapper still works, returns dict for now
562	
563	    # 1. Validation check diagnostics
564	    print("[DIAGNOSTIC] Running validation framework...")
565	    results = registry.run_all(video_path, cfg)
566	    for r in results:
567	        status_symbol = "[PASS]" if r.passed else "[FAIL]"
568	        print(f"  {status_symbol} {r.validator_name}: {r.message}")
569	        if r.details:
570	            print(f"         Details: {r.details}")
571	
572	    # 2. Local segment analysis around the timestamp (sampling +/- 5 seconds)
573	    print("-" * 60)
574	    print("[DIAGNOSTIC] Analyzing local motion around the timestamp...")
575	    import cv2
576	    cap = cv2.VideoCapture(video_path)
577	    if not cap.isOpened():
578	        print("[FAIL] OpenCV could not open video.")
579	        return
580	
581	    fps = cap.get(cv2.CAP_PROP_FPS)
582	    total_frames = cap.get(cv2.CAP_PROP_FRAME_COUNT)
583	
584	    start_frame = max(0, int((timestamp - 3.0) * fps))
585	    end_frame = min(int(total_frames), int((timestamp + 3.0) * fps))
586	
587	    # Measure frame-by-frame changes in sample region
588	    cap.set(cv2.CAP_PROP_POS_FRAMES, start_frame)
589	    prev_gray = None
590	    motions = []
591	
592	    for _ in range(start_frame, end_frame):
593	        ret, frame = cap.read()
594	        if not ret:
595	            break
596	        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
597	        gray = cv2.GaussianBlur(gray, (21, 21), 0)
598	
599	        if prev_gray is not None:
600	            diff = cv2.absdiff(prev_gray, gray)
601	            _, thresh = cv2.threshold(diff, 25, 255, cv2.THRESH_BINARY)
602	            motion_ratio = cv2.countNonZero(thresh) / (thresh.shape[0] * thresh.shape[1])
603	            motions.append(motion_ratio)
604	        prev_gray = gray
605	
606	    cap.release()
607	
608	    if motions:
609	        avg_motion = sum(motions) / len(motions)
610	        # Support both legacy dict (from wrapper) and future direct model
611	        if hasattr(cfg, "indexing"):
612	            threshold = getattr(cfg.indexing, "motion_threshold", 0.08)
613	        else:
614	            threshold = cfg.get("indexing", {}).get("motion_threshold", 0.08)  # type: ignore[attr-defined]
615	        print(f"  Sample Frame Count: {len(motions)}")
616	        print(f"  Average motion index around timestamp: {round(avg_motion, 4)} (threshold: {threshold})")
617	        if avg_motion > threshold:
618	            print("  Result: [ACTIVE PLAY] High motion detected. Segment classifies as active play.")
619	        else:
620	            print("  Result: [DEAD TIME] Idle movement. Segment classified as dead time.")
621	    else:
622	        print("  [ERROR] No visual frames were extracted in the sample range.")
623	
624	    print("=" * 60)

    def build_middleware_stack(self) -> ASGIApp:
        debug = self.debug
        error_handler = None
        exception_handlers: dict[Any, ExceptionHandler] = {}

        for key, value in self.exception_handlers.items():
            if key in (500, Exception):
                error_handler = value
            else:
                exception_handlers[key] = value

        middleware = (
            [Middleware(ServerErrorMiddleware, handler=error_handler, debug=debug)]
            + self.user_middleware
            + [Middleware(ExceptionMiddleware, handlers=exception_handlers, debug=debug)]
        )

        app = self.router
        for cls, args, kwargs in reversed(middleware):
            app = cls(app, *args, **kwargs)
        return app
