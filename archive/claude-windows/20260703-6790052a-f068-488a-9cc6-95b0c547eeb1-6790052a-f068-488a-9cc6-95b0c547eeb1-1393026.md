"""The ``muneem`` command-line interface.

``ingest`` parses a PDF into the local store (promo, or ``--doc-type bank_statement`` with
automatic password unlock); ``offers list`` / ``txns list <bank>|all`` read back what was
found (listings tag anything not cleanly reconciled); ``eval <folder>`` reports reconcile
rates; ``db …`` covers the encryption lifecycle (status/encrypt/rotate-key/backup/
setup-recovery/recover). ``--ocr`` adds local image OCR for codes rasterized into banners
(approximate; flagged for verification). ``fetch`` (Gmail pull) lands with Phase F.
"""

from __future__ import annotations

from contextlib import contextmanager
from pathlib import Path

import typer

from muneem import __version__
from muneem import db as dbmod
from muneem.config import load_config
from muneem.log import get_logger
from muneem.redaction import redact_filename

_log = get_logger(__name__)

app = typer.Typer(
    add_completion=False,
    help="muneem — a local-first personal-finance bookkeeper.",
    no_args_is_help=True,
)
offers_app = typer.Typer(no_args_is_help=True, help="Work with extracted offers.")
app.add_typer(offers_app, name="offers")

txns_app = typer.Typer(no_args_is_help=True, help="Work with parsed transactions.")
app.add_typer(txns_app, name="txns")

db_app = typer.Typer(no_args_is_help=True, help="Manage the local encrypted ledger.")
app.add_typer(db_app, name="db")


@contextmanager
def _ledger(db_path: Path):
    """Open the encrypted ledger, turning an un-migrated *plaintext* file into a clean CLI
    error (pointing at ``muneem db encrypt``) rather than a raw traceback."""
    try:
        with dbmod.connect_encrypted(db_path) as conn:
            yield conn
    except dbmod.EncryptedDatabaseError as exc:
        typer.echo(str(exc), err=True)
        raise typer.Exit(1) from exc


# Dispatch table for supported banks (extensible as new parsers land).
BANK_DISPATCH = {
    "axis": (dbmod.list_axis_transactions, "debit", "credit"),
    "axis_cc": (dbmod.list_axis_cc_transactions, "amount", "amount"),  # direction from "type"
    "au": (dbmod.list_au_transactions, "debit", "credit"),
    "hdfc": (dbmod.list_hdfc_transactions, "withdrawal", "deposit"),
}


def _version_callback(value: bool) -> None:
    if value:
        typer.echo(f"muneem {__version__}")
        raise typer.Exit()


@app.callback()
def main(
    version: bool = typer.Option(
        False,
        "--version",
        callback=_version_callback,
        is_eager=True,
        help="Show the muneem version and exit.",
    ),
) -> None:
    """muneem command-line interface."""


@app.command()
def info() -> None:
    """Show resolved local paths (no secrets, no sensitive data)."""
    cfg = load_config()
    present = "present" if cfg.config_file.is_file() else "absent"
    typer.echo(f"muneem {__version__}")
    typer.echo(f"config dir : {cfg.config_dir}")
    typer.echo(f"data dir   : {cfg.data_dir}")
    typer.echo(f"db path    : {cfg.db_path}")
    typer.echo(f"staging    : {cfg.staging_dir}")
    typer.echo(f"config file: {cfg.config_file} ({present})")


def _ingest_bank_statement(pdf: Path, db_path: Path, doc_type: str, *, interactive: bool) -> None:
    """Unlock (if encrypted) → extract → select parser → store reconciled transactions.

    The hybrid password unlock (cached/configured candidates → prompt) runs *before* extraction,
    since the text needed to pick a parser is itself encrypted. The resolved password is forwarded
    to extraction and the parser, used in-memory only — never logged.
    """
    from muneem import db as dbmod
    from muneem.extract import extract
    from muneem.parsers import get_parser
    from muneem.statement_unlock import resolve_password
    from muneem.unlock import is_encrypted

    password: str | None = ""
    if is_encrypted(pdf):
        password = resolve_password(pdf, interactive=interactive)
        if password is None:
            hint = " (run without --no-prompt to enter it)" if not interactive else ""
            typer.echo(f"Could not unlock {pdf.name}: no working password{hint}.", err=True)
            raise typer.Exit(1)

    ex = extract(pdf, password=password or "")  # resolved above; `or ""` keeps the type str
    parser = get_parser(ex.text)
    if parser is None:
        typer.echo(f"No parser recognizes {pdf.name} as a supported bank statement.", err=True)
        raise typer.Exit(1)

    with _ledger(db_path) as conn:
        dbmod.init_db(conn)
        existing = dbmod.get_document_by_sha(conn, ex.sha256)
        if existing is not None and existing["status"] == "reconciled":
            typer.echo(f"already ingested (doc {existing['id']}): {pdf.name}")
            return
        if existing is not None:
            # sha256 dedup must not entomb a failure: a document the oracle did NOT endorse
            # gets a fresh parse (a parser fix deserves a second chance). Same document id,
            # clean row slate; provenance upserts on the re-parse.
            doc_id = int(existing["id"])
            dbmod.clear_document_rows(conn, doc_id)
            typer.echo(
                f"re-parsing previously {existing['status']} document (doc {doc_id}): {pdf.name}"
            )
        else:
            doc_id = dbmod.insert_document(
                conn,
                source="file",
                external_id=pdf.name,
                doc_type=doc_type,
                sha256=ex.sha256,
                path=str(pdf.resolve()),
                status="parsed",
            )
        count = parser.parse_and_insert(pdf, conn, doc_id, password=password)
        doc = dbmod.get_document_by_sha(conn, ex.sha256)
        status = doc["status"] if doc else "parsed"

    typer.echo(
        f"ingested {pdf.name} (doc {doc_id}); parsed {count} transaction(s) "
        f"via {parser.__class__.__name__} [{status}]"
    )


def _eval_statement(pdf: Path, *, interactive: bool) -> tuple[str, int, str | None]:
    """Run one statement through the full parse pipeline into an in-memory DB.

    Returns ``(outcome, row_count, parser_name)`` where ``outcome`` is the document status
    (``reconciled`` / ``unreconciled``) or a failure tag (``locked`` / ``no-parser`` / ``error``).
    Nothing is persisted and no statement value is surfaced — only the verdict and counts.
    """
    from muneem import db as dbmod
    from muneem.extract import extract
    from muneem.parsers import get_parser
    from muneem.statement_unlock import resolve_password
    from muneem.unlock import is_encrypted

    password = ""
    if is_encrypted(pdf):
        password = resolve_password(pdf, interactive=interactive) or ""
        if not password:
            return ("locked", 0, None)
    try:
        ex = extract(pdf, password=password)
        parser = get_parser(ex.text)
    except Exception as exc:  # eval must not crash on one bad file; record the redacted reason
        _log.warning(
            "eval: extract/route failed for %s (%s)", redact_filename(pdf), type(exc).__name__
        )
        return ("error", 0, None)
    if parser is None:
        return ("no-parser", 0, None)
    name = parser.__class__.__name__
    try:
        with dbmod.connect(":memory:") as conn:
            dbmod.init_db(conn)
            doc_id = dbmod.insert_document(
                conn, source="file", external_id=pdf.name, sha256=ex.sha256, status="parsed"
            )
            count = parser.parse_and_insert(pdf, conn, doc_id, password=password)
            doc = dbmod.get_document_by_sha(conn, ex.sha256)
            status = doc["status"] if doc else "parsed"
    except Exception as exc:
        _log.warning(
            "eval: parse/store failed for %s (%s)", redact_filename(pdf), type(exc).__name__
        )
        return ("error", 0, name)
    return (status, count, name)


@app.command("eval")
def eval_statements(
    folder: Path = typer.Argument(
        ...,
        exists=True,
        file_okay=False,
        readable=True,
        help="A LOCAL folder of statement PDFs to evaluate (e.g. your seed or golden set).",
    ),
    prompt: bool = typer.Option(
        False,
        "--prompt/--no-prompt",
        help="Prompt for a password when cached/configured candidates don't unlock a file.",
    ),
) -> None:
    """Evaluate the parsers against a folder of statements and report the reconcile rate.

    The seed→golden onboarding loop (see docs/onboarding-a-bank.md): point it at a few *seed*
    statements while tuning a profile, then at a *golden* set to confirm. Everything runs locally in
    an in-memory DB; only counts / statuses / parser names print — never statement values.
    """
    pdfs = sorted(folder.glob("*.pdf"))
    if not pdfs:
        typer.echo(f"no .pdf files in {folder}")
        return
    reconciled = 0
    for pdf in pdfs:
        outcome, count, name = _eval_statement(pdf, interactive=prompt)
        reconciled += outcome == "reconciled"
        typer.echo(f"  {outcome:12} {count:4d} row(s)  {name or '—':20} {pdf.name}")
    typer.echo(f"\n{reconciled}/{len(pdfs)} reconciled")


@app.command()
def ingest(
    pdf: Path = typer.Argument(
        ..., exists=True, dir_okay=False, readable=True, help="Path to a PDF to ingest."
    ),
    doc_type: str = typer.Option("promo", help="Document category."),
    db: Path | None = typer.Option(None, help="SQLite path (defaults to the local data dir)."),
    ocr: bool = typer.Option(
        False, "--ocr", help="Also OCR images locally (Tesseract) for codes in banners."
    ),
    no_prompt: bool = typer.Option(
        False,
        "--no-prompt",
        help="Never prompt for a statement password; use only cached/configured candidates.",
    ),
) -> None:
    """Store a PDF's offers (promo) or transactions (bank_statement) in the local DB."""
    from muneem.extract import extract
    from muneem.offers import extract_offer

    db_path = db or load_config().db_path

    if doc_type == "bank_statement":
        _ingest_bank_statement(pdf, db_path, doc_type, interactive=not no_prompt)
        return

    from muneem import db as dbmod

    ex = extract(pdf)
    text_fields = extract_offer(ex.text)
    # (code, source) pairs; text-layer codes are exact, OCR codes are approximate.
    found: list[tuple[str, str]] = [(c, "text") for c in text_fields.codes]
    expiry_raw, expiry_type = text_fields.expiry_raw, text_fields.expiry_type

    if ocr:
        from muneem.ocr import ocr_pdf, tesseract_available

        if not tesseract_available():
            typer.echo("warning: --ocr requested but Tesseract not found; skipping OCR.", err=True)
        else:
            ocr_fields = extract_offer(ocr_pdf(pdf))
            known = {c for c, _ in found}
            for code in ocr_fields.codes:
                if code not in known:
                    found.append((code, "ocr"))
                    known.add(code)
            if expiry_raw is None:
                expiry_raw, expiry_type = ocr_fields.expiry_raw, ocr_fields.expiry_type

    with _ledger(db_path) as conn:
        dbmod.init_db(conn)
        existing = dbmod.get_document_by_sha(conn, ex.sha256)
        if existing is not None:
            typer.echo(f"already ingested (doc {existing['id']}): {pdf.name}")
            return
        doc_id = dbmod.insert_document(

@db_app.command("rotate-key")
def db_rotate_key(
    db: Path | None = typer.Option(None, help="SQLite path (defaults to the local data dir)."),
) -> None:
    """Rotate the ledger's encryption key (SQLCipher rekey) and update the keychain."""
    from muneem import db as dbmod
    from muneem import db_key
    from muneem.errors import KeychainError

    db_path = db or load_config().db_path
    current = _require_encrypted_ledger(db_path)
    fresh = db_key.new_key()
    # Both keys keep a durable copy at every instant: stage the fresh key under the PENDING
    # keychain entry, rekey the file, and only then promote it to the primary entry. Any
    # interruption leaves a state that connect_encrypted can heal (file rekeyed -> pending key
    # promoted on next open; file untouched -> primary entry still matches it).
    db_key.set_pending_key(fresh)
    try:
        dbmod.rekey_database(db_path, current, fresh)
    except Exception as exc:
        # Don't trust the exception to tell us how far the rekey got — check the file.
        if dbmod.key_opens_database(db_path, fresh):
            pass  # the rekey actually completed; fall through to promotion below
        elif dbmod.key_opens_database(db_path, current):
            db_key.clear_pending_key()
            typer.echo(
                f"key rotation failed; the ledger is unchanged (still under the existing key): "
                f"{exc}",
                err=True,
            )
            raise typer.Exit(1) from exc
        else:
            typer.echo(
                "key rotation failed and the ledger opens with neither key. Both keys were kept "
                "in the keychain ('db-key' = old, 'db-key-pending' = new); restore the ledger "
                "from a backup. Do NOT delete either keychain entry until the ledger opens.",
                err=True,
            )
            raise typer.Exit(1) from exc
    try:
        db_key.set_key(fresh)  # promote: the file is verifiably under the fresh key now
    except KeychainError as exc:
        typer.echo(
            "the ledger was rekeyed but promoting the new key in the keychain failed; the new "
            "key is safe under 'db-key-pending' and the next ledger open will finish the "
            "promotion automatically.",
            err=True,
        )
        raise typer.Exit(1) from exc
    db_key.clear_pending_key()
    typer.echo(f"rotated the encryption key for {Path(db_path).name}.")
    typer.echo(
        "note: backups and any recovery file created BEFORE this rotation still use the old "
        "key. Re-run 'muneem db backup' and 'muneem db setup-recovery' now."
    )

## Review — feature is sound and safe; a few things worth tightening

The design is the right shape: the cross-check only *upgrades* a CC parse to `RECONCILED` when `Σ(Dr) − Σ(Cr)` matches the printed total-due within half a paisa; a mismatch or a missing total-due line stays `UNVERIFIED`, and unparseable amounts still quarantine the whole parse. Four things to consider — none are blockers, but the first two are worth doing:

### 1. The "Cr" (credit-balance) case is documented but not actually handled
The regex comment lists `Total Payment Due  12,345.67 Cr` as a supported form, but the trailing `Cr` is dropped — I verified `_TOTAL_DUE_RE` captures `12,345.67` (positive) and ignores the `Cr`. So when the customer has a **credit balance** (overpaid; true due is negative), `total_due` is stored as `+12345.67` while the actual net is ≈ `−12345.67` → the cross-check can *never* match → the statement silently stays `UNVERIFIED`. ### 2. `_TOTAL_DUE_RE` has no direct test

### 3. Worth a one-line note: the check assumes no carried-over balance
`Σ(Dr) − Σ(Cr)` equals "Total Amount Due" only when there's **no uncaptured previous/opening balance**.
