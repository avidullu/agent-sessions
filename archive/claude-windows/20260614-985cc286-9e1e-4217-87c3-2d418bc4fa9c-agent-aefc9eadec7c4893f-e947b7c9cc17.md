"""Per-user promotion gate — the safety rail for flag-gated personalization.

The owner adopted personalization as a **flag-gated** feature on a **generalization-first**
baseline: a cue/variant B may be flipped ON *for one user* only when that user's own
held-out evidence honestly clears the same small-N bar the global eval uses — otherwise we
fall back to the shipped baseline (variant A). This module is the decision logic for that
flip; it is pure (no I/O) and **reuses** ``backend.eval.eval_stats`` for every statistic
(``paired_delta_ci`` + ``paired_sign_test``) so the math is never reinvented.

The counter-case this guards against (and its mitigations, implemented here):

1. **Minimum-N floor.** Personalizing off 1-2 of a user's videos is the small-N-lies regime
   (see ``eval_stats`` C1). Refuse to promote below ``min_n`` videos — point estimates from
   too few held-out videos are noise, not signal.
2. **Honest lift, not a single mean.** Promote only when the paired ΔF1 95% bootstrap CI
   **excludes 0** *and* the paired sign test agrees (majority of videos win, p <= threshold).
   This is the global promotion bar applied per user.
3. **Structural guards are never disabled on "no lift".** A *structural* guard (e.g. Gate-A
   net-crossings — the held-shuttle false-positive killer) earns its keep through failure
   modes a given user's corpus may not contain yet. Measuring "no per-user lift" on a corpus
   that simply lacks those failure cases must NEVER turn the guard OFF. For ``structural=True``
   the decision is therefore **always "keep ON"**, independent of the measured delta.

Below the floor / insufficient evidence → fall back to the shipped baseline (the conservative
default), never an unsafe promotion. All additive, pure, deterministic (the bootstrap seed is
threaded through). Nothing here changes existing eval behaviour; callers opt in.
"""
from __future__ import annotations

@dataclass
class PromotionDecision:
    """Structured result of a per-user promotion decision.

    - ``promote`` — flip variant B ON for this user (only ever ``True`` for a non-structural
      cue that cleared the full bar).
    - ``keep_on`` — the cue/guard should be ON for this user. For a structural guard this is
      always ``True`` (it is never disabled on "no lift"); for a non-structural cue it equals
      ``promote``.
    - ``reason`` — short human-readable explanation of the decision.
    - the remaining fields surface the evidence (n, mean delta, CI bounds and the two tests)
      so a caller can log/audit exactly why a user was or wasn't personalized.
    """

    promote: bool
    keep_on: bool
    reason: str
    n: int
    mean_delta: float
    ci_low: float
    ci_high: float
    ci_excludes_zero: bool
    sign_test: Dict[str, float] = field(default_factory=dict)
    floor_met: bool = False
    structural_protected: bool = False

    def as_dict(self) -> Dict[str, Any]:
        """Plain-dict view (for JSON logging / telemetry)."""
        return {
            "promote": self.promote,
            "keep_on": self.keep_on,
            "reason": self.reason,
            "n": self.n,
            "mean_delta": self.mean_delta,
            "ci_low": self.ci_low,
            "ci_high": self.ci_high,
            "ci_excludes_zero": self.ci_excludes_zero,
            "sign_test": self.sign_test,
            "floor_met": self.floor_met,
            "structural_protected": self.structural_protected,
        }


def should_promote_for_user(
    a_scores: Sequence[float],
    b_scores: Sequence[float],
    *,
    structural: bool = False,
    min_n: int = 5,
    p_threshold: float = 0.05,
    confidence: float = 0.95,
    n_boot: int = 2000,
    seed: int = 0,
    eps: float = 0.0,
) -> PromotionDecision:
    """Decide whether to flip variant B ON for ONE user from that user's held-out scores.

    ``a_scores`` / ``b_scores`` are the user's **per-video** held-out F1 (or any score) for
    variant A (control / shipped baseline) and variant B (with the cue), aligned by video
    (same order). The decision rules (faithful to the module docstring):

    1. ``structural=True`` → the guard is **never disabled on "no lift"**: return
       ``keep_on=True, promote=False`` regardless of the measured per-user delta. Its value is
       in failure modes the user's corpus may not yet contain.
    2. Otherwise promote **iff ALL** hold — the minimum-N floor (``n >= min_n``), the paired
       ΔF1 bootstrap CI **excludes 0**, and the paired sign test agrees (majority of videos
       win, ``p_value <= p_threshold``). Any miss → fall back to the shipped baseline.

    Statistics come straight from ``eval_stats`` (``paired_delta_ci`` / ``paired_sign_test``);
    deterministic given ``seed``. Returns a :class:`PromotionDecision`.
    """
    n = min(len(a_scores), len(b_scores))
    floor_met = n >= min_n

    # --- Evidence (reuse eval_stats; never reinvent the statistics) ---------------------
    if n == 0:
        # No paired videos at all — nothing to test. Surface an empty-but-honest result.
        sign_test: Dict[str, float] = paired_sign_test([], eps)
        mean_delta, ci_low, ci_high, ci_excludes_zero = 0.0, 0.0, 0.0, False
    else:
        delta = paired_delta_ci(
            a_scores, b_scores,
            confidence=confidence, n_boot=n_boot, seed=seed, eps=eps,
        )
        mean_delta = float(delta["mean_delta"])
        ci_low = float(delta["ci_low"])
        ci_high = float(delta["ci_high"])
        ci_excludes_zero = bool(delta["ci_excludes_zero"])
        sign_test = delta["sign_test"]

    sign_p = float(sign_test.get("p_value", 1.0))
    sign_wins = float(sign_test.get("wins", 0.0))
    sign_losses = float(sign_test.get("losses", 0.0))
    # The sign test "agrees" with a promotion only if it is significant AND the majority
    # leans the *winning* way (more B-wins than B-losses) — a significant net *loss* must
    # never satisfy the promotion bar.
    sign_agrees = (sign_p <= p_threshold) and (sign_wins > sign_losses)

    # --- Rule 1: structural guards are never disabled on "no lift" -----------------------
    if structural:
        return PromotionDecision(
            promote=False,           # nothing to "promote" — it ships ON by design
            keep_on=True,            # ALWAYS kept ON, regardless of measured per-user lift
            reason=(
                "structural guard kept ON regardless of measured per-user lift "
                "(value is in failure modes this user's corpus may not yet contain)"
            ),
            n=n,
            mean_delta=mean_delta,
            ci_low=ci_low,
            ci_high=ci_high,
            ci_excludes_zero=ci_excludes_zero,
            sign_test=sign_test,
            floor_met=floor_met,
            structural_protected=True,
        )

    # --- Rule 2: minimum-N floor (refuse before even looking at lift) --------------------
    if not floor_met:
        return PromotionDecision(
            promote=False,
            keep_on=False,           # fall back to the shipped baseline (variant A)
            reason=(
                f"below minimum-N floor (n={n} < min_n={min_n}); "
                "falling back to shipped baseline"
            ),
            n=n,
            mean_delta=mean_delta,
            ci_low=ci_low,
            ci_high=ci_high,
            ci_excludes_zero=ci_excludes_zero,
            sign_test=sign_test,
            floor_met=False,
            structural_protected=False,
        )

    # --- Rule 2 (cont.): honest lift — CI excludes 0 AND sign test agrees ----------------
    if ci_excludes_zero and sign_agrees:
        promote = True
        reason = (
            f"promoted: n={n} >= {min_n}, ΔF1 95% CI "
            f"[{ci_low:.4f}, {ci_high:.4f}] excludes 0, "
            f"sign test {int(sign_wins)}-{int(sign_losses)} (p={sign_p:.4f} <= {p_threshold})"
        )
    else:
        promote = False
        misses = []
        if not ci_excludes_zero:
            misses.append(f"ΔF1 95% CI [{ci_low:.4f}, {ci_high:.4f}] spans 0")
        if not sign_agrees:
            misses.append(
                f"sign test does not agree ({int(sign_wins)}-{int(sign_losses)}, "
                f"p={sign_p:.4f})"
            )
        reason = (
            "insufficient evidence (" + "; ".join(misses) + "); "
            "falling back to shipped baseline"
        )

    return PromotionDecision(
        promote=promote,
        keep_on=promote,             # non-structural: keep_on tracks promote exactly
        reason=reason,
        n=n,
        mean_delta=mean_delta,
        ci_low=ci_low,
        ci_high=ci_high,
        ci_excludes_zero=ci_excludes_zero,
        sign_test=sign_test,
        floor_met=True,
        structural_protected=False,
    )
