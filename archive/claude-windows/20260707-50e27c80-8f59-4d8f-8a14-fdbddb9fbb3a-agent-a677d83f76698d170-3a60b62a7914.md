377	+class NvdecResidentFrontend:
378	+    """On-GPU frame source for the ``nvdec_resident`` detector pass (#506).
379	+
380	+    PyNvVideoCodec ``SimpleDecoder(output_color_type=RGB)`` decodes on the GPU's
381	+    NVDEC engine; frames arrive as CUDA tensors via DLPack and never touch the host.
382	+    ``read_chw`` then applies the cv2-faithful affine warp (``build_affine_grid``)
383	+    and ImageNet-normalizes on-device, yielding the same ``[3, inp_h, inp_w]``
384	+    float32 tensor per frame that the ImageDataset path assembles on the CPU.
385	+    """
386	+
387	+    def __init__(self, video_path: str, inp_wh, gpu_id: int = 0):
388	+        import PyNvVideoCodec as nvc
389	+        import torch
390	+
391	+        self._torch = torch
392	+        self._device = torch.device("cuda", gpu_id)
393	+        inp_w, inp_h = int(inp_wh[0]), int(inp_wh[1])
394	+        try:
395	+            self._dec = nvc.SimpleDecoder(
396	+                video_path, gpu_id=gpu_id, output_color_type=nvc.OutputColorType.RGB
397	+            )
398	+        except Exception as e:
399	+            raise RuntimeError(
400	+                f"PyNvVideoCodec RGB SimpleDecoder failed for {video_path}: {e!r} — "
401	+                f"decode_backend=nvdec_resident needs PyNvVideoCodec with RGB output "
402	+                f"and libnvcuvid/libnvidia-encode on the box (deploy/cloudrun/Dockerfile "
403	+                f"/ gpu_setup.sh stage them). Fall back to decode_backend=cpu."
404	+            ) from e
405	+        self.count = len(self._dec)
406	+        probe = self._decode_rgb_hwc(0)
407	+        self.orig_h, self.orig_w = int(probe.shape[0]), int(probe.shape[1])
408	+        trans_inp_inv = _wasb_inverse_affines(
409	+            self.orig_h, self.orig_w, (inp_w, inp_h), [0]
410	+        )[0]
411	+        self._grid = build_affine_grid(
412	+            trans_inp_inv, inp_w, inp_h, self.orig_w, self.orig_h
413	+        ).to(self._device)
414	+        self._mean = torch.tensor(IMAGENET_MEAN, device=self._device).view(3, 1, 1)
415	+        self._std = torch.tensor(IMAGENET_STD, device=self._device).view(3, 1, 1)
416	+
417	+    def _decode_rgb_hwc(self, index: int):
418	+        """Frame ``index`` as an on-GPU HWC RGB uint8 tensor (zero-copy via DLPack)."""
419	+        torch = self._torch
420	+        t = torch.from_dlpack(self._dec[index])
421	+        if t.dim() == 3 and t.shape[0] == 3 and t.shape[-1] != 3:
422	+            t = t.permute(1, 2, 0)  # CHW layout variant → HWC
423	+        if t.dim() != 3 or t.shape[-1] != 3:
424	+            raise RuntimeError(
425	+                f"unexpected NVDEC frame shape {tuple(t.shape)} for frame {index} — "
426	+                f"expected 3-channel RGB (is output_color_type=RGB supported here?)"
427	+            )
428	+        return t
429	+
430	+    def read_chw(self, index: int):
431	+        """Decoded frame ``index`` → the normalized ``[3, inp_h, inp_w]`` float32 CUDA
432	+        tensor the detector expects (warpAffine-faithful resize + ImageNet normalize)."""
433	+        import torch.nn.functional as F
434	+
435	+        hwc = self._decode_rgb_hwc(index)
436	+        chw = hwc.permute(2, 0, 1).unsqueeze(0).float() / 255.0
437	+        warped = F.grid_sample(
438	+            chw, self._grid, mode="bilinear", align_corners=False, padding_mode="border"
439	+        ).squeeze(0)
440	+        return (warped - self._mean) / self._std
441	+
442	+    def release(self):
443	+        self._dec = None

134	    def _nvdec_resident_requires_cuda(self) -> "WasbIndexConfig":
135	        # NVDEC decode is CUDA-only; reject the contradictory combo at config load instead of
136	        # failing deep inside a paid GPU run. "auto" is allowed here — the native runner's
137	        # healthcheck hard-fails if auto resolves to cpu with this backend on.
138	        if self.decode_backend == "nvdec_resident" and self.device not in ("cuda", "auto"):
139	            raise ValueError(
140	                f"indexing.wasb.decode_backend='nvdec_resident' requires device 'cuda' or "
141	                f"'auto' (got {self.device!r}) — NVDEC decode runs on the GPU."
142	            )
143	        return self

145	    @field_validator("batch_size")
146	    @classmethod
147	    def _batch_size_positive(cls, v: Optional[int]) -> Optional[int]:
148	        # A DataLoader batch must be >=1; reject at config time so a typo can't reach
149	        # `wasb_infer.py --batch-size` as an invalid value (the runner only passes the flag
150	        # when truthy, so 0 stays "off/parity-default").
151	        if v is not None and v < 1:
152	            raise ValueError(f"indexing.wasb.batch_size must be >= 1 when set (got {v})")
153	        return v

155	    @field_validator("prefetch_batches")
156	    @classmethod
157	    def _prefetch_batches_non_negative(cls, v: Optional[int]) -> Optional[int]:
158	        # 0 = off (the default). Reject negatives: the runner passes any truthy value as
159	        # `--prefetch-batches`, and `_PrefetchIter` clamps depth with max(1, int(depth)), so a
160	        # negative would SILENTLY enable prefetch at depth 1 instead of failing loudly.
161	        if v is not None and v < 0:
162	            raise ValueError(
163	                f"indexing.wasb.prefetch_batches must be >= 0 when set (got {v})"
164	            )
165	        return v

202	class IndexingConfig(BaseModel):
203	    default_segmenter: str = "yolo_hybrid"
204	    use_gpu: bool = True
205	    # Detector backend for the trajectory hybrids (wasb/tracknet). "auto" = the real
206	    # WSL runner (default, production); "stub" = a deterministic CPU mock (no GPU/WSL)
207	    # so the whole pipeline is regression-testable on a CPU box / CI; "native" = the
208	    # Linux-native WASB runner (no WSL/conda-activate; a GPU box — M3.5). The
209	    # RALLY_DETECTOR_IMPL env var overrides this. See pipeline/detectors/{stub,native_wasb}_runner.py.
210	    detector_impl: str = "auto"
211	    motion_threshold: float = 0.08
212	    min_segment_duration: float = 3.0
213	    dead_time_merge_gap: float = 2.0
214	    chunk_duration: float = 90.0
215	    chunk_overlap: float = 10.0
216	    detector_model: str = "fasterrcnn_resnet50_fpn"
217	    detector_score_threshold: float = 0.5
218	    yolo_velocity_threshold: float = 0.15
219	    yolo_frame_skip: int = 3
220	    yolo_tracker: str = "bytetrack.yaml"
221	    yolo_prefetch_workers: int = 2
222	    min_action_window_duration: float = 2.0
223	    # BOUNDED WINDOWING (over-merge fix W1, RALLY_QUALITY_RESEARCH.md §6). 0 = OFF; > 0 = a window
224	    # longer than this many seconds is recursively split at its largest internal inactivity gap (≥
225	    # ``window_min_split_gap`` s — the most likely rally boundary), so one window can't swallow
226	    # several rallies + the dead time between them. Never merges, only cuts (recall can't drop).
227	    # *** R1 MILESTONE DEFAULT-ON (cap=6): the smallest safe local-windowing flip. Measured n=13
228	    # golden, R2 tolF1: baseline→milestone (cap=6 + R4 below) Δ+0.222, sign 12/0, p=0.0005; cap=6
229	    # beats cap=12 Δ+0.110, sign 12/0; net over-seg guard PASS (mean merge_rate 0.651→0.161). ***
230	    max_window_duration: float = 6.0
231	    window_min_split_gap: float = (
232	        0.5  # min internal gap (s) that qualifies as a split point
233	    )
234	    # HARD-CAP fallback for W1: when True, an over-long window with NO internal inactivity gap
235	    # (a continuously-active trajectory — e.g. the real-footage mahadevpura-1 174s blob) is
236	    # chopped at its midpoint until ≤ max_window_duration, making the cap a true upper bound.
237	    # Only cuts (recall can't drop). OFF by default until measured/promoted.
238	    window_hard_cap: bool = False
239	    # R4 rally-END inactivity trim (s, 0 = OFF). After a rally the shuttle keeps moving slowly
240	    # (picked up / walked) above velocity_thresh, so the window over-extends its END (the measured
241	    # |Δend| gap vs golden). Trim the post-rally tail back to the last frame whose trailing
242	    # window stays dense with FAST motion (velocity > inactivity_rally_thresh). Only cuts.
243	    # *** R1 MILESTONE DEFAULT-ON (1.5/0.20/0.30): on top of cap=6, R4 adds a small but significant
244	    # end-precision gain — ΔtolF1 +0.040, sign 9/1/3, p=0.022, |Δend| 4.96→3.31s (n=13). The W1 cap
245	    # is the dominant lever; R4 is the marginal increment. See QUALITY_ITERATIONS.md "R1". ***
246	    window_inactivity_end: float = 1.5
247	    inactivity_rally_thresh: float = 0.20
248	    inactivity_min_density: float = 0.30
249	    # R4 density DENOMINATOR fix (code-review finding B). The R4 trim's fast-fraction denominator is
250	    # the COUNT of active samples in the trailing window, which sparse tracking inflates (one fast
251	    # sample after a gap → 100% dense → tail held open). When True, the denominator is the window's
252	    # TEMPORAL length instead. No-op under dense tracking (sample-count == temporal-span), so it only
253	    # corrects the sparse-track case. *** P2b PROMOTED DEFAULT-ON (2026-06-29): golden-set A/B at the
254	    # SERVED operating point (n=15, 595 rallies) — tolF1 0.353→0.364 (ΔtolF1 +0.010, sign 10W/4L, NO
255	    # regression), |Δend| 2.97→2.72s, seg_count_ratio −0.103 (p=0.006), split_rate −0.010 (p=0.016),
256	    # R1 over-seg guard PASS; served_regresses(eps=0.0)=False → promote_if_served_safe(structural)=True.
257	    # See docs/RALLY_DETECTION_FIXES_PLAN.md §7/§8. ***
258	    inactivity_temporal_density: bool = True
259	    # R5 blob-fallback (default OFF). LAST-RESORT splitter for the continuously-active blob: after
260	    # the W1 gap-split AND the R4 inactivity trim have each had their chance, a group STILL longer
261	    # than window_blob_factor × max_window_duration (no internal gap, no rally-end signal — e.g.
262	    # mahadevpura-1's ~65s residual ≈ 11× a 6s cap) is midpoint-chopped to ≤ the cap as a FORCED cut
263	    # (logged + flagged on the window) so the degenerate single-window outcome is avoided and the
264	    # clip is surfaced as needing rally-STATE cues, not a bigger cap. Differs from window_hard_cap
265	    # (which chops BEFORE R4 and over-segments normal clips): blob-fallback runs AFTER R4 and only
266	    # force-cuts the genuine blob (the factor gate). Only cuts (recall can't drop).
267	    window_blob_fallback: bool = False
268	    # Magnitude gate that keeps R5 surgical: only a group longer than this × max_window_duration is a
269	    # blob (a normal long rally ~1-2× the cap is left alone; the mahadevpura-1 residual is ~11×).
270	    # Default 4.0 is do-no-harm on the golden corpus (rescues only the continuously-active
271	    # clips, every other clip a no-op within noise) — see docs/QUALITY_ITERATIONS.md "R5".
272	    window_blob_factor: float = 4.0
273	    log_yolo_candidates: bool = True
274	    store_yolo_candidates: bool = True
275	    ai_handoff_padding: float = 2.0
276	    ai_max_workers: int = 5
277	    # Width (px) the gemini segmenter re-encodes chunks to before upload. Stream-copied
278	    # chunks of a high-bitrate (4K) source blow past the provider upload cap
279	    # (backend/providers/gemini.py::MAX_UPLOAD_MB) and are refused — observed live as
280	    # "0 segments / success". Re-encoding also makes chunk boundaries frame-accurate
281	    # (stream copy cuts at the nearest keyframe). 0 = legacy stream-copy at source res.
282	    # Matches gemini_refine's --clip-scale-width default.
283	    ai_chunk_scale_width: int = 1280
284	    # FULLY-LOCAL mode (default OFF): when true, the trajectory-hybrid segmenters
285	    # (wasb_hybrid / tracknet_hybrid / fusion_hybrid) skip the Phase-3 AI handoff entirely and
286	    # emit their local candidate windows AS the rallies — no provider, no API key, nothing
287	    # uploaded. Lets the local detector run standalone (e.g. to benchmark it against the Gemini
288	    # path, or to operate with zero cloud dependency). The pure `gemini` segmenter ignores this.
289	    skip_ai_handoff: bool = False
290	    # Optional debug knob: trim every video to the first N seconds before processing.
291	    debug_duration: Optional[float] = None

293	    tracknet: TrackNetConfig = Field(default_factory=TrackNetConfig)
294	    wasb: WasbIndexConfig = Field(default_factory=WasbIndexConfig)
295	    fusion: FusionConfig = Field(default_factory=FusionConfig)

925	def build_affine_grid(trans_inv, inp_w: int, inp_h: int, orig_w: int, orig_h: int):
926	    """``F.grid_sample`` sampling grid replicating ``cv2.warpAffine(img, trans,
927	    (inp_w, inp_h), INTER_LINEAR)`` — THE #506 lesson lives here.
928	
929	    The GPU resize MUST be WASB's affine warp (center/scale geometry + cv2's
930	    integer-coordinates-are-pixel-centers convention), NOT a plain resize:
931	    ``F.interpolate``'s half-pixel mapping shifts every sample by a constant
932	    sub-pixel offset, which tips the sigmoid past ``score_threshold`` on marginal
933	    frames and silently over-detects — F1 0.58 → 0.07 on the hard clip GX010128,
934	    while the easy clip masked it at 98% agreement. Locked by a cv2-parity unit test.
935	
936	    ``trans_inv`` is WASB's INVERSE affine (``get_affine_transform(..., inv=1)``),
937	    mapping output px → source px; each source position is normalized to [-1, 1]
938	    via ``gx = (2*x + 1)/W - 1`` (cv2's half-pixel convention). Pure torch+numpy —
939	    no WASB imports — so the parity test runs on any CPU box.
940	
941	    Returns a ``[1, inp_h, inp_w, 2]`` float32 grid for ``F.grid_sample(...,
942	    mode='bilinear', align_corners=False, padding_mode='border')``.
943	    """
944	    import numpy as np
945	    import torch
946	
947	    ys, xs = torch.meshgrid(
948	        torch.arange(inp_h, dtype=torch.float32),
949	        torch.arange(inp_w, dtype=torch.float32),

952	    out_coords = torch.stack([xs, ys, torch.ones_like(xs)], dim=-1)  # [H,W,3] output px
953	    mt = torch.from_numpy(np.ascontiguousarray(trans_inv, dtype=np.float32))  # [2,3]
954	    src = out_coords @ mt.transpose(0, 1)  # [H,W,2] source px
955	    gx = (2.0 * src[..., 0] + 1.0) / orig_w - 1.0
956	    gy = (2.0 * src[..., 1] + 1.0) / orig_h - 1.0

206	def manifest_compatible(
207	    manifest: dict,
208	    *,
209	    frames_dir: str,
210	    weights: str,
211	    sport: str,
212	    frames_in: int,
213	    frames_total: int,
214	    device: str = "cuda",
215	    decode_backend: str = "cpu",
216	) -> bool:
217	    """A cache is reusable only if the run that produced it matches this run's inputs.
218	
219	    ``device`` defaults to ``cuda`` so caches written before device-keying (no ``device``
220	    field) stay compatible with a CUDA run — but a cpu/mps run will NOT reuse a cuda cache
221	    (detections can differ numerically across devices), and vice-versa.
222	
223	    ``decode_backend`` follows the same pattern (#506): pre-existing caches (no field)
224	    are ``cpu``-decoded, so they stay compatible with cpu runs — but an ``nvdec_resident``
225	    run never reuses a cpu cache (NVDEC decode + the GPU affine warp are trajectory-level,
226	    not byte-level, equivalents), and vice-versa.
227	
228	    ``frames_dir`` is the **already-canonical cache key** the caller stored in the manifest
229	    (``run()`` passes its ``cache_key`` here, and writes the same value into the manifest):
230	    ``osp.abspath(frames_dir)`` for the frames-on-disk path, or the synthetic ``video:<abspath>``
231	    key for the streaming path. We therefore compare it VERBATIM — do NOT re-``abspath`` it here:
232	    ``osp.abspath("video:/x.mp4")`` is not recognised as absolute and gets the CWD prepended,
233	    so a streaming cache would never match itself and every re-run would recompute (issue #348).
234	    """
235	    if not manifest or manifest.get("version") != CACHE_VERSION:
236	        return False
237	    return (
238	        manifest.get("frames_dir") == frames_dir
239	        and manifest.get("weights") == weights
240	        and manifest.get("sport") == sport
241	        and manifest.get("frames_in") == frames_in
242	        and manifest.get("frames_total") == frames_total
243	        and manifest.get("device", "cuda") == device
244	        and manifest.get("decode_backend", "cpu") == decode_backend
245	    )

256	def load_and_clean_cache(cache_dir: str):
257	    """Read the longest *contiguous* (w == line index) valid prefix of det_raw.jsonl,
258	    rebuild the per-frame detection accumulation, and rewrite the file to exactly that
259	    prefix so a trailing half-written line from a crash can't corrupt future appends.
260	
261	    Returns (windows_done, det_by_fid) where det_by_fid maps frame_id -> list of
262	    [x, y, score, scale] candidates (accumulated across every window the frame is in,
263	    in window order — identical to a non-resumed run).
264	    """
265	    det_jsonl, _ = _cache_paths(cache_dir)
266	    det_by_fid = defaultdict(list)
267	    valid: list = []
268	    if osp.exists(det_jsonl):
269	        with open(det_jsonl, "r") as f:
270	            for line in f:
271	                s = line.strip()
272	                if not s:
273	                    continue
274	                try:
275	                    obj = json.loads(s)
276	                except json.JSONDecodeError:
277	                    break  # truncated trailing line (crash mid-flush)
278	                if obj.get("w") != len(valid):  # non-contiguous -> stop at the gap
279	                    break
280	                valid.append(s)
281	                for fid, plain in obj["f"]:
282	                    det_by_fid[int(fid)].extend([list(p) for p in plain])
283	        # Guarantee a clean append boundary by rewriting only the good prefix.
284	        _atomic_write_text(det_jsonl, ("\n".join(valid) + "\n") if valid else "")
285	    return len(valid), det_by_fid

337	class SequentialFrameStore:
338	    """In-memory sliding buffer over a forward-only frame reader, sized for the
339	    detector's sliding-window access pattern (peak O(window) frames, not O(video)).
340	
341	    The detector DataLoader runs with ``shuffle=False`` and (in streaming mode)
342	    ``num_workers=0``. ``ImageDataset.__getitem__(i)`` reads the frames of window ``i``
343	    in order — ``i, i+1, ..., i+window-1`` — and samples are requested ``i = start,
344	    start+1, ...``. So the global read sequence dips back by ``window-1`` at each window
345	    boundary (``0,1,2, 1,2,3, 2,3,4, ...`` for ``window=3``); the highest index seen so
346	    far never decreases. We keep only frames at or above ``max_seen - (window-1)`` (the
347	    earliest index any not-yet-finished window can still ask for) and decode each source
348	    frame exactly once via the forward-only ``reader(index)``.
349	    """
350	
351	    def __init__(self, reader, total: int, window: int = 1, start_index: int = 0):
352	        self._reader = reader
353	        self._total = int(total)
354	        self._window = max(1, int(window))
355	        self._buf: dict = {}
356	        # On a resumed run the detector's first needed frame is `start_index`; begin
357	        # pulling there so the reader can skip (seek past) the already-cached prefix
358	        # instead of decoding 0..start_index-1 just to throw them away.
359	        self._next = int(start_index)  # next index not yet pulled from the reader
360	        self._max_seen = int(start_index) - 1
361	        self._floor = int(start_index)  # lowest index we still keep (never evict below)
362	
363	    def get(self, index: int):
364	        index = int(index)
365	        if index < self._floor:
366	            raise KeyError(
367	                f"frame {index} already evicted (below floor {self._floor}); the access "
368	                f"pattern must stay within a window of the highest frame seen"
369	            )
370	        if index >= self._total:
371	            raise KeyError(f"frame {index} out of range (total {self._total})")
372	        # Pull forward from the reader until the requested index is buffered.
373	        while self._next <= index:
374	            self._buf[self._next] = self._reader(self._next)
375	            self._next += 1
376	        if index > self._max_seen:
377	            self._max_seen = index
378	        # Evict frames older than the earliest a still-running window could re-request:
379	        # once we've seen index `m`, no future window starts before `m - (window-1)`.
380	        new_floor = max(self._floor, self._max_seen - (self._window - 1))
381	        for k in range(self._floor, new_floor):
382	            self._buf.pop(k, None)
383	        self._floor = new_floor
384	        return self._buf[index]

396	def _bgr_to_pil_rgb(frame_bgr):
397	    """Convert an in-memory BGR uint8 frame (as ``cv2.VideoCapture`` yields) to the EXACT

401	    because PNG is lossless that round-trip equals ``cv2.cvtColor(frame_bgr, COLOR_BGR2RGB)``

403	    frame indistinguishable from the on-disk one to everything downstream.

406	    from PIL import Image
407	
408	    return Image.fromarray(cv2.cvtColor(frame_bgr, cv2.COLOR_BGR2RGB))

411	def _make_streamed_read_image(store, real_read_image):
412	    """Build the ``read_image`` replacement used during a streaming detector pass.
413	
414	    A synthetic frame path -> the in-memory decoded frame for its index (as a PIL RGB image,
415	    matching ``read_image``'s real return type); any path that doesn't parse to an index falls
416	    through to ``real_read_image``. Pure — imports no WASB module — so it is unit-testable
417	    without the WSL-only ``dataloaders`` package.
418	    """

420	    def _read_image(path, *args, **kwargs):
421	        idx = _index_from_synthetic_path(path)
422	        if idx < 0:
423	            return real_read_image(path, *args, **kwargs)
424	        return _bgr_to_pil_rgb(store.get(idx))

429	@contextmanager
430	def _streaming_read_image_patch(
431	    frame_loader, total: int, window: int = 1, start_index: int = 0
432	):
433	    """Scope a shim over WASB's ``read_image`` so ``ImageDataset`` is fed in-memory decoded
434	    frames during the detector pass, byte-for-byte as it would over real PNGs.
435	
436	    WASB's ``ImageDataset.__getitem__`` reads each frame via ``read_image(path)`` — NOT
437	    ``cv2.imread``. ``read_image`` (``utils.utils``) does ``Image.open(path).convert('RGB')``,
438	    returning a PIL **RGB** image, after an ``osp.exists(path)`` guard. We therefore feed it a
439	    PIL RGB image built from the streamed BGR frame (see ``_bgr_to_pil_rgb``), which is
440	    byte-identical to the disk round-trip, and the shim never touches the filesystem so the
441	    ``osp.exists`` guard is sidestepped for synthetic stream paths.
442	
443	    We patch ``dataloaders.dataset_loader.read_image`` — the binding the consumer actually
444	    calls (``dataset_loader`` does ``from utils import read_image``, so the name lives in that
445	    module's namespace; patching ``utils.read_image`` would NOT rebind the already-imported
446	    reference).
447	
448	    ``frame_loader(index) -> BGR uint8 ndarray`` is a forward-only sequential decoder wrapped
449	    in a ``SequentialFrameStore`` (sliding buffer sized to ``window`` = ``frames_in``); on a
450	    resume we start at ``start_index`` so the decoder seeks past already-cached windows. The
451	    original reader is restored on exit.
452	    """
453	    from dataloaders import dataset_loader as _dl
454	
455	    store = SequentialFrameStore(
456	        frame_loader, total, window=window, start_index=start_index
457	    )
458	    real_read_image = _dl.read_image
459	    # Rebind read_image in the consuming module for the duration of the detector pass;
460	    # restore on exit. (Plain assignment, not setattr-with-constant, to stay ruff B010-clean.)
461	    _dl.read_image = _make_streamed_read_image(store, real_read_image)  # type: ignore[assignment]
462	    try:
463	        yield store
464	    finally:
465	        _dl.read_image = real_read_image  # type: ignore[assignment]

471	class _PrefetchIter:
472	    """Bounded producer-thread wrapper over a batch iterator (the num_workers=0 DataLoader).
473	
474	    In streaming mode the DataLoader must stay single-process (the in-memory frame store +
475	    the cv2.imread shim live in this process and cannot be pickled to workers), so the
476	    CPU-side batch assembly (decode + PIL + ToTensor) SERIALIZES with GPU inference —
477	    measured on the first real L4 serving run (2026-07-03) this left the GPU <1% utilized.
478	    A single producer THREAD fixes the overlap without multiprocessing: cv2/PIL/numpy
479	    release the GIL for the heavy parts, the store's forward-only access pattern is
480	    preserved (one consumer thread iterating sequentially), and the bounded queue caps
481	    memory at ``depth`` batches. A producer exception is re-raised in the consumer."""
482	
483	    _END = object()
484	
485	    def __init__(self, src, depth: int = 2):
486	        import queue
487	        import threading
488	
489	        self._q: "queue.Queue" = queue.Queue(maxsize=max(1, int(depth)))
490	        self._exc: "BaseException | None" = None
491	
492	        def _fill() -> None:
493	            try:
494	                for item in src:
495	                    self._q.put(item)
496	            except BaseException as e:  # noqa: BLE001 - surfaced in __next__
497	                self._exc = e
498	            finally:
499	                self._q.put(self._END)
500	
501	        self._t = threading.Thread(target=_fill, name="wasb-prefetch", daemon=True)
502	        self._t.start()
503	
504	    def __iter__(self):
505	        return self
506	
507	    def __next__(self):
508	        item = self._q.get()
509	        if item is self._END:
510	            if self._exc is not None:
511	                raise self._exc
512	            raise StopIteration
513	        return item

827	def make_video_frame_loader(video_path: str):
828	    """Return ``(frame_loader, count_hint, release)`` for output-preserving streaming.
829	
830	    ``frame_loader(index) -> BGR uint8 ndarray`` decodes the video forward-only with one
831	    long-lived ``cv2.VideoCapture``. It MUST be called with non-decreasing indices (the
832	    detector's sliding-window order); it reads sequentially and only uses a frame-seek to
833	    skip forward when a *resume* starts past the current position. Each ``cap.read()``
834	    returns exactly the BGR uint8 array that ``cv2.imwrite``/``cv2.imread`` of a PNG would
835	    yield (PNG is lossless), so the detector sees identical pixels to the disk path.
836	
837	    Returns the count hint from the container so the caller can build the frame list;
838	    ``release`` closes the capture.
839	    """
840	    import cv2
841	
842	    cap = cv2.VideoCapture(video_path)
843	    if not cap.isOpened():
844	        raise SystemExit(f"cannot open video: {video_path}")
845	    count_hint = int(cap.get(cv2.CAP_PROP_FRAME_COUNT) or 0)
846	    state = {"pos": 0}  # next index cap.read() will return
847	
848	    def frame_loader(index: int):
849	        index = int(index)
850	        if index < state["pos"]:
851	            raise RuntimeError(
852	                f"streaming decode is forward-only; got index {index} after {state['pos']}"
853	            )
854	        if index > state["pos"]:
855	            # Forward skip only happens when resuming past cached windows. Seek once.
856	            cap.set(cv2.CAP_PROP_POS_FRAMES, index)
857	            state["pos"] = index
858	        ok, frame = cap.read()
859	        if not ok:
860	            raise RuntimeError(f"failed to decode frame {index} from {video_path}")
861	        state["pos"] += 1
862	        return frame
863	
864	    def release():
865	        cap.release()
866	
867	    return frame_loader, count_hint, release
