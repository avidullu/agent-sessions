"""Import/syntax smoke tests for the backend package.

Guards against a SyntaxError (or module-level import error) shipping while the
suite stays green: most of the suite imports individual backend modules, so a
broken file that nothing imports (e.g. the stitcher compiler, which only
backend.main pulls in) used to go unnoticed.

Both tests are offline and dependency-light: the syntax sweep never imports
anything, and the backend.main import runs in a subprocess with any missing
heavy deps (cv2/torch/numpy) stubbed, so they also pass in environments
without the GPU stack.
"""

import py_compile
import subprocess
import sys
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[1]


def test_backend_package_has_no_syntax_errors():
    """Every .py file under backend/ must byte-compile."""
    failures = []
    for path in sorted((REPO_ROOT / "backend").rglob("*.py")):
        try:
            py_compile.compile(str(path), doraise=True)
        except py_compile.PyCompileError as exc:
            failures.append(f"{path.relative_to(REPO_ROOT)}: {exc.msg}")
    assert not failures, "Syntax errors in backend/:\n" + "\n".join(failures)


def test_backend_main_imports():
    """backend.main (FastAPI app, /api/compile) must be importable.

    Runs in a subprocess so stubbing missing heavy deps cannot leak into this
    pytest process; cwd=repo root puts the package on sys.path.
    """
    prelude = (
        "import importlib.util, sys\n"
        "from unittest import mock\n"
        "for name in ('cv2', 'torch', 'numpy'):\n"
        "    if importlib.util.find_spec(name) is None:\n"
        "        sys.modules[name] = mock.MagicMock()\n"
        "import backend.main\n"
    )
    result = subprocess.run(
        [sys.executable, "-c", prelude],
        cwd=str(REPO_ROOT),
        capture_output=True,
        text=True,
    )
    assert result.returncode == 0, (
        "import backend.main failed:\n" + result.stderr.strip()

#### Subscripts and Superscripts

**IMPORTANT**: Never use Unicode subscript/superscript characters (₀₁₂₃₄₅₆₇₈₉, ⁰¹²³⁴⁵⁶⁷⁸⁹) in ReportLab PDFs. The built-in fonts do not include these glyphs, causing them to render as solid black boxes.

Instead, use ReportLab's XML markup tags in Paragraph objects:
```python
from reportlab.platypus import Paragraph
from reportlab.lib.styles import getSampleStyleSheet

styles = getSampleStyleSheet()

# Subscripts: use <sub> tag
chemical = Paragraph("H<sub>2</sub>O", styles['Normal'])

# Superscripts: use <super> tag
squared = Paragraph("x<super>2</super> + y<super>2</super>", styles['Normal'])
```

For canvas-drawn text (not Paragraph objects), manually adjust font the size and position rather than using Unicode subscripts/superscripts.
