115	def _finalize_reingest(video_id: str, segments, play_wm: int, cand_wm: int) -> None:
116	    """Commit a (re)segmentation with destroy-AFTER-confirm semantics.
117	
118	    On a non-empty run, drop the PRIOR rows (id <= the pre-run watermark) so only the
119	    fresh segments remain. On an empty run, drop the NEW partial rows (id > watermark) and
120	    keep the prior good results intact — a failed/empty re-run never destroys prior data.
121	    """
122	    if segments:
123	        db_instance.prune_segments(video_id, play_upto=play_wm, cand_upto=cand_wm)
124	    else:
125	        db_instance.prune_segments(video_id, play_after=play_wm, cand_after=cand_wm)

146	def _execute_processing(req: ProcessRequest, progress=None) -> Dict[str, Any]:
147	    """The full hash → validate → segment → report pipeline for one video.
148	
149	    This is the former synchronous /api/process body, unchanged in behavior, now run on
150	    the job worker thread (DEFERRED_HARDENING_PLAN #16). Returns the same response dict
151	    the sync endpoint used to return (UI compatibility); the per-video observability
152	    report is still emitted in `finally:` and grafted as response["reports"].
153	
154	    `progress` is an optional callable(stage: str) used to surface coarse job progress.
155	    Raises on unexpected internal errors — the caller records those as a job failure
156	    (after this function has already restored the pre-run segments, see below).
157	    """
158	    notify = progress or (lambda stage: None)
159	
160	    notify("hashing")
161	    # M2: resolve the input to a LOCAL path. A bare/local path is returned unchanged (identity —
162	    # byte-identical to today); a bucket/Drive URI is fetched to scratch first. The original
163	    # req.video_path (the source URI) is kept for the DB record + report; only the file-reading
164	    # consumers (hash / validators / segmenter) use the resolved local path.
165	    local_video = storage.acquire_local_input(req.video_path, getattr(global_config, "storage", None))
166	    video_id = compute_video_id(local_video)
167	    was_reingest = db_instance.get_video(video_id) is not None
168	
169	    # 1. Run pluggable validators (with-context variant surfaces ffprobe_meta for the report)
170	    notify("validating")
171	    logger.info(f"Running validations for {req.video_path}...")
172	    val_results, val_context = registry.run_all_with_context(local_video, global_config)
173	    failures = [r for r in val_results if not r.passed]
174	
175	    # typed AppConfig: attribute access for the default segmenter (with safe fallback)
176	    default_seg = getattr(getattr(global_config, "indexing", None), "default_segmenter", "motion")
177	    seg_name = req.segmenter or default_seg
178	    segmenter_actual = None
179	    segmentation_ran = False
180	    response: Optional[Dict[str, Any]] = None
181	    try:
182	        if failures:
183	            # add_video is an UPSERT (no cascade) — a failed re-validation only updates
184	            # status; it no longer destroys the video's prior good segments.
185	            db_instance.add_video(video_id, req.video_path, "failed", [r.model_dump() for r in val_results])
186	            response = {
187	                "video_id": video_id,
188	                "status": "failed",
189	                "message": "Video failed validation checks.",
190	                "results": [r.model_dump() for r in val_results],
191	            }
192	            return response
193	
194	        # 2. Run segmenter & indexer
195	        logger.info("[API] Video passed checks. Segmenting and indexing...")
196	        db_instance.add_video(video_id, req.video_path, "processed", [r.model_dump() for r in val_results])
197	
198	        segmenter_cls = segmenter_registry.get(seg_name)
199	        if not segmenter_cls:
200	            # Submit-time validation already 400s unknown names; this is the defensive
201	            # in-worker path (e.g. config default pointing at an unregistered segmenter).
202	            available = list(segmenter_registry.list_available().keys())
203	            response = {
204	                "video_id": video_id,
205	                "status": "failed",
206	                "message": f"Segmenter '{seg_name}' not found. Available segmenters: {available}",
207	                "results": [r.model_dump() for r in val_results],
208	                "play_segments": [],
209	            }
210	            return response
211	
212	        logger.info(f"[API] Using segmenter: {seg_name}")
213	        notify(f"segmenting ({seg_name})")
214	        segmenter_actual = seg_name
215	        # Destroy-AFTER-confirm: remember the pre-run rows, then keep new + drop old only
216	        # if the run yields segments; otherwise drop the new partial rows and keep the old.
217	        play_wm, cand_wm = db_instance.segment_watermarks(video_id)
218	        segmenter = segmenter_cls(db_instance, global_config)
219	        try:
220	            result = segmenter.process_video(video_id, local_video, req.player_pool, req.max_frames)
221	        except Exception:
222	            # A raising segmenter must not leave half-written rows: restore the pre-run
223	            # state (same destroy-after-confirm contract as the Failure branch), then let
224	            # the job worker record the failure.
225	            db_instance.prune_segments(video_id, play_after=play_wm, cand_after=cand_wm)
226	            raise
227	        segmentation_ran = True
228	
229	        if isinstance(result, Failure):
230	            logger.error(f"[API] Segmenter failed: {result.message}")
231	            db_instance.prune_segments(video_id, play_after=play_wm, cand_after=cand_wm)
232	            response = {
233	                "video_id": video_id,
234	                "status": "failed",
235	                "message": f"Segmenter '{seg_name}' failed: {result.message}",
236	                "results": [r.model_dump() for r in val_results],
237	                "play_segments": [],
238	            }
239	            return response
240	
241	        segments = result.value if hasattr(result, "value") else result
242	        notify("finalizing")
243	        _finalize_reingest(video_id, segments, play_wm, cand_wm)
244	
245	        response = {
246	            "video_id": video_id,
247	            "status": "success",
248	            "message": f"Successfully indexed {len(segments)} play segments using '{seg_name}' segmenter.",
249	            "results": [r.model_dump() for r in val_results],
250	            "play_segments": segments,
251	        }
252	        return response
253	    finally:
254	        # Always emit the per-video observability report (next to the video, or to
255	        # report.output_dir). Never raises. Surface the paths in the response.
256	        paths = emit_indexer_report(
257	            db=db_instance, video_id=video_id, video_path=req.video_path, config=global_config,
258	            val_results=val_results, val_context=val_context,
259	            segmenter_requested=req.segmenter, segmenter_actual=segmenter_actual,
260	            was_reingest=was_reingest, segmentation_ran=segmentation_ran,
261	        )
262	        if paths and isinstance(response, dict):
263	            response["reports"] = paths

266	@app.post("/api/process", status_code=202)
267	def process_video(req: ProcessRequest):
268	    """Submit a processing job (202 + job_id). Poll GET /api/jobs/{job_id} for state.
269	
270	    Fast-fail checks (bad path, missing file, unknown segmenter) happen here so the
271	    client gets an immediate 4xx; everything slow — including MD5 hashing of the file —
272	    runs on the worker (DEFERRED_HARDENING_PLAN #16). The row is created 'queued'; the
273	    worker drains it via `claim_due_job` (EXECUTION_POLICY.md P3 self-scheduling loop).
274	    """
275	    allowed_root = getattr(getattr(global_config, "security", None), "allowed_video_root", None)
276	    try:
277	        validate_input_path(req.video_path, allowed_root=allowed_root)
278	        # M2: resolve the input once (path or storage URI) — raises ValueError for an unsupported
279	        # scheme, which we map to a clean 400 (an unknown scheme must never 500).
280	        input_ref = storage.resolve_input_ref(req.video_path, getattr(global_config, "storage", None))
281	    except ValueError as e:
282	        raise HTTPException(status_code=400, detail=str(e)) from e
283	    # The local-existence fast-fail applies only to a LOCAL input, and stats the RESOLVED local path
284	    # (input_ref.key — e.g. local://… stripped to the bare path), not the raw URI string. A
285	    # bucket/Drive URI can't be cheaply os.path.exists()'d — its existence is verified when the worker
286	    # fetches it (a fetch miss fails the job loudly). validate_input_path above still runs for every
287	    # input: its null-byte guard always applies, and a configured allowed_video_root (hosted mode)
288	    # rejects a non-local URI as out-of-root.
289	    if input_ref.backend == "local" and not os.path.exists(input_ref.key):
290	        raise HTTPException(status_code=404, detail=f"Video file not found at '{req.video_path}'")
291	    if req.segmenter and not segmenter_registry.get(req.segmenter):
292	        available = list(segmenter_registry.list_available().keys())
293	        raise HTTPException(
294	            status_code=400,
295	            detail=f"Segmenter '{req.segmenter}' not found. Available segmenters: {available}"
296	        )
297	
298	    job_id = uuid.uuid4().hex
299	    if not db_instance.create_job(job_id, req.video_path, req.segmenter,
300	                                  req.player_pool, req.max_frames):
301	        raise HTTPException(status_code=500, detail="Failed to persist job record.")
302	    _get_executor().submit(_drain_due_jobs)
303	    return JSONResponse(
304	        status_code=202,
305	        content={"job_id": job_id, "status": "queued", "poll": f"/api/jobs/{job_id}"},
306	    )

377	@app.post("/api/compile")
378	def compile_highlights(req: CompileRequest):
379	    video = db_instance.get_video(req.video_id)
380	    if not video:
381	        raise HTTPException(status_code=404, detail="Indexed video record not found.")
382	
383	    # Retrieve matching play segments from db
384	    all_segments = db_instance.query_play_segments(req.video_id, {})
385	    matched_segments = [s for s in all_segments if s["id"] in req.segment_ids]
386	
387	    if not matched_segments:
388	        raise HTTPException(status_code=400, detail="No matching play segments found to stitch.")
389	
390	    # Reject path traversal / absolute output names (os.path.join would otherwise let an
391	    # absolute or '..'-laden output_filename write anywhere on disk).
392	    try:
393	        out_name = safe_output_filename(req.output_filename)
394	    except ValueError as e:
395	        raise HTTPException(status_code=400, detail=str(e)) from e
396	
397	    # The configured shot-count floor (stitching.min_shot_count) applies on the API
398	    # path too — parity with scripts/run_process_and_stitch.py. Segments WITHOUT shot_count
399	    # metadata are exempt: the floor filters known counts only, so explicitly
400	    # requested manual/legacy segments can't silently vanish under the default floor.
401	    stitching = getattr(global_config, "stitching", None) or StitchingConfig()
402	    kept = []
403	    for s in matched_segments:
404	        meta = s.get("metadata") or {}
405	        sc = meta.get("shot_count") if isinstance(meta, dict) else None
406	        try:
407	            if sc is not None and int(sc) < stitching.min_shot_count:
408	                continue
409	        except (TypeError, ValueError):
410	            pass  # unparseable count -> treat as unknown, keep
411	        kept.append(s)
412	    if len(kept) < len(matched_segments):
413	        logger.info(f"[compile] stitching.min_shot_count={stitching.min_shot_count}: dropped "
414	                    f"{len(matched_segments) - len(kept)}/{len(matched_segments)} segments below the floor.")
415	    if not kept:
416	        raise HTTPException(status_code=400,
417	                            detail=f"All {len(matched_segments)} matching segments fall below "
418	                                   f"stitching.min_shot_count={stitching.min_shot_count}.")
419	
420	    # Extract start/end time pairs
421	    time_pairs = [(s["start_time"], s["end_time"]) for s in kept]
422	
423	    # Run compiler with the configured stitching paddings + optional branding watermark
424	    output_dir = getattr(getattr(global_config, "output", None), "output_dir", "output") or "output"
425	    compiler = VideoCompiler(output_dir, begin_padding=stitching.begin_padding,
426	                             end_padding=stitching.end_padding, watermark=stitching.watermark)
427	    try:
428	        out_path = compiler.compile_highlights(video["filepath"], time_pairs, out_name)
429	        # `filepath` is server-local (not reachable by the browser); also return the ready-to-use
430	        # download URL so the SPA needn't hand-build it (P6 / PER_RALLY_SELECTION.md §4.3).
431	        download_url = f"/api/highlights/{quote(req.video_id)}/{quote(out_name)}"
432	        return {"status": "success", "filepath": out_path, "download_url": download_url}
433	    except Exception as e:
434	        raise HTTPException(status_code=500, detail=f"Highlight stitching failed: {str(e)}") from e

437	def run_cli_diagnose_clip(video_path: str, timestamp: float):
438	    """CLI debugging utility to examine segment properties around a timestamp."""
439	    print("=" * 60)
440	    print(f"DEBUGGING VIDEO CLIP: {video_path} at {timestamp}s")
441	    print("=" * 60)
442	
443	    cfg = load_config()  # legacy wrapper still works, returns dict for now
444	
445	    # 1. Validation check diagnostics
446	    print("[DIAGNOSTIC] Running validation framework...")
447	    results = registry.run_all(video_path, cfg)
448	    for r in results:
449	        status_symbol = "[PASS]" if r.passed else "[FAIL]"
450	        print(f"  {status_symbol} {r.validator_name}: {r.message}")
451	        if r.details:
452	            print(f"         Details: {r.details}")
453	
454	    # 2. Local segment analysis around the timestamp (sampling +/- 5 seconds)
455	    print("-" * 60)
456	    print("[DIAGNOSTIC] Analyzing local motion around the timestamp...")
457	    import cv2
458	    cap = cv2.VideoCapture(video_path)
459	    if not cap.isOpened():
460	        print("[FAIL] OpenCV could not open video.")
461	        return
462	
463	    fps = cap.get(cv2.CAP_PROP_FPS)
464	    total_frames = cap.get(cv2.CAP_PROP_FRAME_COUNT)
465	
466	    start_frame = max(0, int((timestamp - 3.0) * fps))
467	    end_frame = min(int(total_frames), int((timestamp + 3.0) * fps))
468	
469	    # Measure frame-by-frame changes in sample region
470	    cap.set(cv2.CAP_PROP_POS_FRAMES, start_frame)
471	    prev_gray = None
472	    motions = []
473	
474	    for _ in range(start_frame, end_frame):
475	        ret, frame = cap.read()
476	        if not ret:
477	            break
478	        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
479	        gray = cv2.GaussianBlur(gray, (21, 21), 0)
480	
481	        if prev_gray is not None:
482	            diff = cv2.absdiff(prev_gray, gray)
483	            _, thresh = cv2.threshold(diff, 25, 255, cv2.THRESH_BINARY)
484	            motion_ratio = cv2.countNonZero(thresh) / (thresh.shape[0] * thresh.shape[1])
485	            motions.append(motion_ratio)
486	        prev_gray = gray
487	
488	    cap.release()
489	
490	    if motions:
491	        avg_motion = sum(motions) / len(motions)
492	        # Support both legacy dict (from wrapper) and future direct model
493	        if hasattr(cfg, "indexing"):
494	            threshold = getattr(cfg.indexing, "motion_threshold", 0.08)
495	        else:
496	            threshold = cfg.get("indexing", {}).get("motion_threshold", 0.08)  # type: ignore[attr-defined]
497	        print(f"  Sample Frame Count: {len(motions)}")
498	        print(f"  Average motion index around timestamp: {round(avg_motion, 4)} (threshold: {threshold})")
499	        if avg_motion > threshold:
500	            print("  Result: [ACTIVE PLAY] High motion detected. Segment classifies as active play.")
501	        else:
502	            print("  Result: [DEAD TIME] Idle movement. Segment classified as dead time.")
503	    else:
504	        print("  [ERROR] No visual frames were extracted in the sample range.")
505	
506	    print("=" * 60)

203	def _probe_video_fps_width(path: str):
204	    """Robust ffprobe (fps, frame_width) for a real video. Returns None on ANY failure.
205	
206	    The detector trajectory is frame-indexed; the harness maps frame->time via fps, so a WRONG
207	    fps silently mis-aligns every rally label (e.g. a 60fps clip read as 30 doubles every time).
208	    cv2's ``_probe_fps_width`` silently defaults to (30, 1280) on a read miss — fine for the
209	    inference report (it flags the fallback) but corrupting for REAL training data. So here we
210	    probe with ffprobe (same tool as ``get_video_duration``) and return None so the caller SKIPS
211	    rather than guesses.
212	    """
213	    import json
214	    import subprocess
215	    try:
216	        out = subprocess.check_output(
217	            ["ffprobe", "-v", "error", "-select_streams", "v:0",
218	             "-show_entries", "stream=r_frame_rate,width", "-of", "json", path],
219	            stderr=subprocess.DEVNULL).decode()
220	        st = (json.loads(out).get("streams") or [{}])[0]
221	        num, _, den = str(st.get("r_frame_rate", "0/1")).partition("/")
222	        fps = float(num) / float(den or "1")
223	        width = float(st.get("width") or 0)
224	        if fps > 0 and width > 0:
225	            return fps, width
226	    except (subprocess.SubprocessError, ValueError, KeyError, OSError, json.JSONDecodeError):
227	        pass
228	    return None
