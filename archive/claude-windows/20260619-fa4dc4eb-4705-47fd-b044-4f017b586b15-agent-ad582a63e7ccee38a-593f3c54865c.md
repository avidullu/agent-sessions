22	
23	def load_split(manifest: Union[str, Iterable[Dict[str, Any]], os.PathLike[str]]) -> Tuple[Set[str], Set[str]]:
24	    """Load (train_ids, eval_ids) sets keyed by match id.
25	
26	    manifest: path to JSON (str/bytes/PathLike) or iterable of dicts. Each entry should have 'match_id'
27	    (or 'id' fallback; must be non-empty str for real match key; name/video_id deliberately not used as they are weak/MD5).
28	
29	    Raises ValueError on:
30	    - non-list manifest after load
31	    - no real match-level key (empty/falsy after explicit check)
32	    - split label not exactly "train" or "eval"
33	
34	    The collector 'split' field and cross-repo wiring is deferred; this supports
35	    synthetic or pre-split manifests for the pure-logic guard and tests.
36	    """
37	    if isinstance(manifest, (str, bytes, os.PathLike)):
38	        with open(manifest, "r", encoding="utf-8-sig") as f:
39	            raw = json.load(f)
40	            if not isinstance(raw, list):
41	                raise ValueError(f"manifest must be a list of entries, got {type(raw)}")
42	            data = cast(List[Dict[str, Any]], raw)
43	    else:
44	        if isinstance(manifest, dict):
45	            raise ValueError(f"manifest must be a list of entries, got {type(manifest)}")
46	        data = list(manifest)
47	        if not isinstance(data, list):
48	            raise ValueError(f"manifest must be a list of entries, got {type(data)}")
49	
50	    train_ids: Set[str] = set()
51	    eval_ids: Set[str] = set()
52	
53	    for entry in data:
54	        if not isinstance(entry, dict):
55	            continue
56	        # Explicit not-None + non-empty for real match key; no silent fallback to name/video_id (weak/MD5)
57	        mid = entry.get("match_id")
58	        if mid is None or str(mid).strip() == "":
59	            mid = entry.get("id")
60	        if mid is None or str(mid).strip() == "":
61	            raise ValueError(f"no real match-level key (match_id or id) in entry: {entry}")
62	        mid = str(mid).strip()
63	        sp = str(entry.get("split", "")).strip().lower()
64	        if sp not in ("train", "eval"):
65	            raise ValueError(f"split label must be exactly 'train' or 'eval', got {sp!r} in entry: {entry}")
66	        if sp == "eval":
67	            eval_ids.add(mid)
68	        else:
