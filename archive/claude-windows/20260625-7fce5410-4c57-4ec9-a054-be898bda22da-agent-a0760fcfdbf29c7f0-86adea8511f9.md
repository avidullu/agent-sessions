import subprocess
import sys

        return False

    # Default to quiet unless the caller chose a verbosity.
    if not any(a in ("-q", "-v", "-vv", "-vvv", "--quiet", "--verbose") for a in args):

    print(bar)
    print(" RESULT: " + ("PASS - all tests green" if rc == 0 else f"FAIL - pytest exit {rc}"))

    return rc
