def _rows_to_intervals(rows: List[dict]) -> List[Interval]:
    """Pull (start_time, end_time) out of DB segment rows, sorted by start."""
    ivs = [(float(r["start_time"]), float(r["end_time"])) for r in rows]
    ivs.sort(key=lambda iv: iv[0])
