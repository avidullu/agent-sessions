51	    A string with no ``scheme://`` prefix (including Windows ``C:\\...`` paths, which
52	    have no ``//``) is treated as a local filesystem path — preserving today's behaviour.

90	    #    os.path.isabs is platform- AND Python-version-dependent (on Windows + py3.13 a POSIX-style

92	    if s.startswith(("/", "\\")) or (len(s) >= 2 and s[1] == ":" and s[0].isalpha()):
93	        return s

95	        return input_root.rstrip("/") + "/" + s.lstrip("/")
96	    if input_backend and input_backend != "local":   # 4. prefix a bucket/key with the declared backend

268	def _probe_video_fps_width(path: str):
269	    """Robust ffprobe (fps, frame_width) for a real video. Returns None on ANY failure.
270	
271	    The detector trajectory is frame-indexed; the harness maps frame->time via fps, so a WRONG
272	    fps silently mis-aligns every rally label (e.g. a 60fps clip read as 30 doubles every time).
273	    cv2's ``_probe_fps_width`` silently defaults to (30, 1280) on a read miss — fine for the
274	    inference report (it flags the fallback) but corrupting for REAL training data. So here we
275	    probe with ffprobe (same tool as ``get_video_duration``) and return None so the caller SKIPS
276	    rather than guesses.
277	    """
278	    import json
279	    import subprocess
280	    try:
281	        out = subprocess.check_output(
282	            [ffprobe_bin(), "-v", "error", "-select_streams", "v:0",
283	             "-show_entries", "stream=r_frame_rate,width", "-of", "json", path],
284	            stderr=subprocess.DEVNULL).decode()
285	        st = (json.loads(out).get("streams") or [{}])[0]
286	        num, _, den = str(st.get("r_frame_rate", "0/1")).partition("/")
287	        fps = float(num) / float(den or "1")
288	        width = float(st.get("width") or 0)
289	        if fps > 0 and width > 0:
290	            return fps, width
291	    except (subprocess.SubprocessError, ValueError, KeyError, OSError, json.JSONDecodeError):
292	        pass
293	    return None

125	@dataclass(frozen=True)
126	class RallySpec:
127	    """One in-flight rally segment: the shuttle oscillates across the net midline.
128	
129	    ``start_frame`` — first frame index; ``n_frames`` — length; ``period`` — oscillation
130	    period in frames (governs net-crossing count); ``amplitude`` — px excursion from the
131	    net midline along the play axis. Deterministic sine motion → fixed velocity/crossings.
132	    """
133	
134	    start_frame: int
135	    n_frames: int
136	    period: float = 30.0
137	    amplitude: float = 400.0
138	
139	
140	@dataclass(frozen=True)
141	class GapSpec:
142	    """Dead-time between rallies — what separates one window from the next.
143	
144	    ``kind="still"`` — shuttle visible but resting (velocity ≈ 0 → no active frames);
145	    ``kind="occluded"`` — shuttle not visible (Visibility=0 → trajectory broken). Both
146	    end an active run so the next rally forms a distinct window.
147	    """
148	
149	    start_frame: int
150	    n_frames: int
151	    kind: str = "still"  # "still" | "occluded"
152	
153	
154	@dataclass(frozen=True)
155	class FixtureSpec:
156	    """A full synthetic scenario: a frame-indexed sequence of rallies and gaps + its GT.
157	
158	    ``gts`` are the hand-authored ground-truth rally intervals (seconds) — the synthetic
159	    "labels.csv". For a synthetic fixture they line up with the rallies by construction,
160	    which is exactly what makes the quality-floor a *plumbing-correctness* check, not a
161	    field-quality one (§6).
162	    """
163	
164	    slug: str
165	    description: str
166	    fps: float
167	    frame_width: float
168	    frame_height: float
169	    segments: Tuple[Any, ...]  # ordered (RallySpec | GapSpec)
170	    gts: Tuple[Interval, ...]
171	    net_midline: float = 640.0
172	
173	
174	def _rally_points(spec: RallySpec, net_midline: float, frame_height: float
175	                  ) -> List[Tuple[int, int, float, float]]:
176	    """Deterministic in-flight shuttle: sine oscillation across the net midline (play axis = x)."""
177	    rows: List[Tuple[int, int, float, float]] = []
178	    y_mid = frame_height / 2.0
179	    for i in range(spec.n_frames):
180	        x = net_midline + spec.amplitude * math.sin(2.0 * math.pi * i / spec.period)
181	        # A small orthogonal wobble keeps y-spread < x-spread so the play axis is unambiguously x.
182	        y = y_mid + 30.0 * math.sin(2.0 * math.pi * i / (spec.period / 2.0))
183	        rows.append((spec.start_frame + i, 1, x, y))
184	    return rows
185	
186	
187	def _gap_points(spec: GapSpec, net_midline: float, frame_height: float
188	                ) -> List[Tuple[int, int, float, float]]:
189	    """Dead-time rows: resting-visible (still) or absent (occluded)."""
190	    rows: List[Tuple[int, int, float, float]] = []
191	    y_mid = frame_height / 2.0
192	    for i in range(spec.n_frames):
193	        if spec.kind == "occluded":
194	            rows.append((spec.start_frame + i, 0, 0.0, 0.0))
195	        elif spec.kind == "still":
196	            rows.append((spec.start_frame + i, 1, net_midline, y_mid))
197	        else:  # pragma: no cover - guarded by builder
198	            raise ValueError(f"unknown gap kind {spec.kind!r} (use 'still'|'occluded')")
199	    return rows
200	
201	
202	def make_synthetic_trajectory(spec: FixtureSpec) -> List[Tuple[int, int, float, float]]:
203	    """Build the deterministic ``(Frame, Visibility, X, Y)`` rows for a synthetic scenario."""
204	    rows: List[Tuple[int, int, float, float]] = []
205	    for seg in spec.segments:
206	        if isinstance(seg, RallySpec):
207	            rows.extend(_rally_points(seg, spec.net_midline, spec.frame_height))
208	        elif isinstance(seg, GapSpec):
209	            rows.extend(_gap_points(seg, spec.net_midline, spec.frame_height))
210	        else:  # pragma: no cover - guarded by dataclass shape
211	            raise TypeError(f"unknown segment type {type(seg).__name__}")
212	    rows.sort(key=lambda r: r[0])
213	    return rows
214	
215	
216	def write_trajectory_csv(rows: Sequence[Tuple[int, int, float, float]], path: Path) -> None:
217	    """Write trajectory rows as the TrackNet CSV (``Frame,Visibility,X,Y``), LF + 6dp floats."""
218	    lines = ["Frame,Visibility,X,Y"]
219	    for frame, vis, x, y in rows:
220	        lines.append(f"{int(frame)},{int(vis)},{_fmt(x)},{_fmt(y)}")
221	    path.write_text("\n".join(lines) + "\n", encoding="utf-8", newline="\n")
222	
223	
224	def write_labels_csv(gts: Sequence[Interval], path: Path) -> None:
225	    """Write GT intervals as a ``start,end`` labels CSV (the strict gt_loader format), LF."""
226	    lines = ["start,end"]
227	    for s, e in gts:
228	        lines.append(f"{_fmt(s)},{_fmt(e)}")
229	    path.write_text("\n".join(lines) + "\n", encoding="utf-8", newline="\n")

241	def builtin_specs() -> List[FixtureSpec]:
242	    """The committed synthetic scenarios. Deterministic; chosen so each exercises a distinct
243	    windowing behaviour (one window / dead-time split / occlusion split)."""
244	    fps = 30.0
245	    fw, fh = 1280.0, 720.0
246	    net = 640.0
247	
248	    # 1) clean_single_rally: one ~4s in-flight rally → exactly 1 candidate window, crossings>0.
249	    single = FixtureSpec(
250	        slug="clean_single_rally",
251	        description="One continuous ~4s rally; shuttle oscillates across the net → 1 window.",
252	        fps=fps, frame_width=fw, frame_height=fh, net_midline=net,
253	        segments=(RallySpec(start_frame=0, n_frames=120),),
254	        # Window is [0.033, 3.967]; GT is the rally span the synthetic motion fills.
255	        gts=((0.0, 4.0),),
256	    )
257	
258	    # 2) multi_rally_with_deadtime: rally, 3s STILL dead-time, rally → 2 windows split by the
259	    #    low-velocity gap (> merge_gap=2.0s) — proves dead-time separates windows.
260	    r0 = RallySpec(start_frame=0, n_frames=120)
261	    gap_still = GapSpec(start_frame=120, n_frames=_fr(3.0, fps), kind="still")
262	    r1 = RallySpec(start_frame=120 + _fr(3.0, fps), n_frames=120)
263	    multi = FixtureSpec(
264	        slug="multi_rally_with_deadtime",
265	        description="Two ~4s rallies separated by 3s of still (resting-shuttle) dead-time → 2 windows.",
266	        fps=fps, frame_width=fw, frame_height=fh, net_midline=net,
267	        segments=(r0, gap_still, r1),
268	        gts=((0.0, 4.0), (7.0, 11.0)),
269	    )
270	
271	    # 3) occlusion_break: rally, 3s OCCLUDED (Visibility=0), rally → 2 windows split by the
272	    #    visibility break — proves occlusion (not just low velocity) separates windows.
273	    o0 = RallySpec(start_frame=0, n_frames=120)
274	    gap_occ = GapSpec(start_frame=120, n_frames=_fr(3.0, fps), kind="occluded")
275	    o1 = RallySpec(start_frame=120 + _fr(3.0, fps), n_frames=120)
276	    occ = FixtureSpec(
277	        slug="occlusion_break",
278	        description="Two ~4s rallies separated by 3s of shuttle occlusion (Visibility=0) → 2 windows.",
279	        fps=fps, frame_width=fw, frame_height=fh, net_midline=net,
280	        segments=(o0, gap_occ, o1),
281	        gts=((0.0, 4.0), (7.0, 11.0)),
282	    )
283	
284	    return [single, multi, occ]

296	def _fmt(x: float) -> str:
297	    """Render a float with up to 6dp, no scientific notation, ``-0.0`` normalised to ``0.0``."""
298	    r = round(float(x) + 0.0, _FLOAT_ROUND_DP)
299	    if r == 0.0:
300	        r = 0.0
301	    return f"{r:.6f}"
302	
303	
304	def _round(x: float) -> float:
305	    r = round(float(x) + 0.0, _FLOAT_ROUND_DP)
306	    return 0.0 if r == 0.0 else r

309	def _pinned_proposal(prov: Optional[Dict[str, Any]]) -> Tuple[float, float, float]:
310	    """The windowing proposal tuple ``(vt, mg, mwd)``, read from the fixture provenance when present
311	    (the pinned params travel with the fixture), else the module-level pinned defaults. NEVER
312	    ``windowing.SERVED``. NOTE: the provenance ``max_win`` (W1) is intentionally NOT read here —
313	    ``build_candidates`` takes only this 3-tuple, so the replay characterizes the W1-OFF path; the
314	    pinned ``max_win`` is a recorded annotation, not a knob the gate applies (see PINNED_MAX_WINDOW)."""
315	    if prov and "windowing" in prov:
316	        w = prov["windowing"]
317	        return (float(w["vt"]), float(w["mg"]), float(w["mwd"]))
318	    return (PINNED_VT, PINNED_MERGE_GAP, PINNED_MIN_WINDOW)

321	def replay_fixture(slug_or_dir: Any) -> Dict[str, Any]:
322	    """Replay one fixture through the shipped CONTROL path; return its recomputed snapshot.
323	
324	    Resolves ``slug_or_dir`` (a slug string or a fixture directory ``Path``) to its committed
325	    ``trajectory.csv`` + ``labels.csv`` + ``provenance.json``, then:
326	
327	        parse_trajectory_csv → build_candidates(proposal=pinned) → VideoCandidates(5 shuttle
328	        cols only) → predict(CONTROL, vc) → score(preds, gts, iou).
329	
330	    Returns ``{slug, schema_version, iou, candidates:[{start,end,shuttle:{5 names}}],
331	    control_segments:[[s,e]], score:{...}, gt_hash}``. CONTROL keeps every candidate then
332	    merges overlaps; NO numpy is touched.
333	    """
334	    fdir = _resolve_dir(slug_or_dir)
335	    slug = fdir.name
336	    traj_csv = fdir / "trajectory.csv"
337	    labels_csv = fdir / "labels.csv"
338	    prov = _read_json(fdir / "provenance.json") if (fdir / "provenance.json").is_file() else None
339	
340	    fps, fw = _fps_fw(prov, slug)
341	    iou = float(prov["iou"]) if (prov and "iou" in prov) else 0.5
342	    proposal = _pinned_proposal(prov)
343	
344	    intervals, feature_rows = distill_local.build_candidates(
345	        str(traj_csv), fps=fps, frame_width=fw, proposal=proposal)
346	
347	    # Build VideoCandidates DIRECTLY with ONLY the 5 shuttle feature columns. We never call
348	    # fusion_compare.load_videos / ablation.load_videos — those read c['person'] unconditionally
349	    # and would KeyError on these person-free fixtures (HARD CONSTRAINT 1).
350	    features: Dict[str, List[float]] = {
351	        name: [row[i] for row in feature_rows] for i, name in enumerate(SHUTTLE_FEATURES)
352	    }
353	    gts = load_gt_intervals(str(labels_csv), strict=True)
354	    vc = VideoCandidates(name=slug, intervals=intervals, features=features, gts=gts)
355	
356	    preds = predict(CONTROL, vc)  # CONTROL: keep all → merge_overlaps; pure-stdlib, no numpy
357	    sc = score(preds, gts, iou)
358	
359	    candidates: List[Dict[str, Any]] = []
360	    for j, (s, e) in enumerate(intervals):
361	        shuttle = {name: _round(feature_rows[j][i]) for i, name in enumerate(SHUTTLE_FEATURES)}
362	        # net_crossings is conceptually an integer count — keep it exact (int), per constraint 2.
363	        shuttle["net_crossings"] = int(round(feature_rows[j][SHUTTLE_FEATURES.index("net_crossings")]))
364	        candidates.append({"start": _round(s), "end": _round(e), "shuttle": shuttle})
365	
366	    return {
367	        "slug": slug,
368	        "schema_version": CURRENT_SCHEMA,
369	        "iou": _round(iou),
370	        "candidates": candidates,
371	        "control_segments": [[_round(s), _round(e)] for s, e in preds],
372	        "score": {
373	            "iou_threshold": _round(sc.iou_threshold),
374	            "num_pred": int(sc.num_pred),
375	            "num_gt": int(sc.num_gt),
376	            "true_positives": int(sc.true_positives),
377	            "false_positives": int(sc.false_positives),
378	            "false_negatives": int(sc.false_negatives),
379	            "precision": _round(sc.precision),
380	            "recall": _round(sc.recall),
381	            "f1": _round(sc.f1),
382	            "mean_iou": _round(sc.mean_iou),
383	            "mean_start_error": _round(sc.mean_start_error),
384	            "mean_end_error": _round(sc.mean_end_error),
385	        },
386	        "gt_hash": gt_hash(gts),
387	    }

429	def freeze_fixture(slug: str, fixtures_dir: Path = FIXTURES_DIR,
430	                   iou: float = 0.5) -> Dict[str, Any]:
431	    """Compute ``expected.json`` + ``provenance.json`` for one fixture from its COMMITTED
432	    ``trajectory.csv`` + ``labels.csv`` via the replay (so expected is correct-by-construction).
433	
434	    Idempotent: rewrites the same bytes given the same committed inputs. Requires the trajectory
435	    + labels CSVs to already exist (use ``--freeze-synthetic`` to generate them first).
436	    """
437	    fdir = fixtures_dir / slug
438	    traj_csv, labels_csv = fdir / "trajectory.csv", fdir / "labels.csv"
439	    if not traj_csv.is_file() or not labels_csv.is_file():
440	        raise FileNotFoundError(
441	            f"{slug}: trajectory.csv/labels.csv missing — run --freeze-synthetic to generate them")
442	
443	    # Provenance first (it pins the windowing the replay reads), then replay → expected.
444	    spec = _spec_by_slug().get(slug)
445	    fps, fw = (spec.fps, spec.frame_width) if spec is not None else _fps_fw_from_prov(fdir, slug)
446	    prov = _provenance(spec if spec is not None else fdir, slug, fps, fw, iou)
447	    _write_json(fdir / "provenance.json", prov)
448	
449	    expected = replay_fixture(fdir)
450	    _write_json(fdir / "expected.json", expected)
451	    return expected

454	def freeze_synthetic(fixtures_dir: Path = FIXTURES_DIR) -> List[str]:
455	    """Generate trajectory.csv + labels.csv for every built-in scenario, freeze expected.json +
456	    provenance.json, and (re)write the manifest. Returns the slugs frozen."""
457	    specs = builtin_specs()
458	    fixtures_dir.mkdir(parents=True, exist_ok=True)
459	    slugs: List[str] = []
460	    for spec in specs:
461	        fdir = fixtures_dir / spec.slug
462	        fdir.mkdir(parents=True, exist_ok=True)
463	        rows = make_synthetic_trajectory(spec)
464	        write_trajectory_csv(rows, fdir / "trajectory.csv")
465	        write_labels_csv(spec.gts, fdir / "labels.csv")
466	        freeze_fixture(spec.slug, fixtures_dir=fixtures_dir)
467	        slugs.append(spec.slug)
468	    _write_manifest(specs, fixtures_dir)
469	    return slugs

472	def refresh_snapshot(slug: Optional[str] = None, all_slugs: bool = False,
473	                     fixtures_dir: Path = FIXTURES_DIR) -> List[str]:
474	    """Re-freeze expected.json (+ provenance.json) from the COMMITTED trajectory.csv + labels.csv —
475	    NO regeneration of the trajectory/labels. ⚠ §6: this can rubber-stamp a real regression; the
476	    mitigation is reviewing the visible diff + recording a reason (the CLI requires one)."""
477	    if all_slugs:
478	        targets = [d.name for d in sorted(fixtures_dir.iterdir())
479	                   if d.is_dir() and (d / "trajectory.csv").is_file()]
480	    elif slug:
481	        targets = [slug]
482	    else:
483	        raise ValueError("refresh_snapshot needs --slug S or --all")
484	    for s in targets:
485	        freeze_fixture(s, fixtures_dir=fixtures_dir)
486	    return targets

import argparse
import json
import os

def remap(src: List[Dict[str, Any]], corpus_prefix: str
          ) -> Tuple[List[Dict[str, Any]], List[Tuple[str, str]]]:
    """Return (kept, dropped). ``kept`` entries have VM-relative trajectory/labels paths;
    ``dropped`` is ``[(name, reason)]`` for entries missing a local trajectory or label."""
    kept: List[Dict[str, Any]] = []
    dropped: List[Tuple[str, str]] = []
    for v in src:
        traj_ok = os.path.exists(v["trajectory"])
        lab_ok = os.path.exists(v.get("labels", ""))
