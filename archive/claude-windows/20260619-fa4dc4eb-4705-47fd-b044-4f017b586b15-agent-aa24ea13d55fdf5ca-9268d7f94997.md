def gate_intervals_to_court(intervals: List[Tuple[float, float]], points: List[Any],
                            polygon: Optional[Polygon], fps: float,
                            frame_width: float, frame_height: float,
                            min_in_court_frac: float = 0.5, height_pad: float = 0.0
                            ) -> List[Tuple[float, float]]:
    """Drop candidate rallies whose shuttle is MAJORITY out-of-court — the window-level gate.

    Point-level gating (:func:`gate_points_to_court`) is leaky: a background-court rally whose shuttle
    occasionally drifts into the prominent court keeps a few in-court points, enough to hold a (sparse)
    window (measured: GX010142 rally #1 is 90% adjacent-court yet a 10% central tail survives a hard
    point gate). This drops an interval whose in-court point fraction is below ``min_in_court_frac``,
    so a rally that is *mostly* in the adjacent court is removed wholesale. No-op (returns all
    intervals) when ``polygon`` is None/degenerate. Subtractive-only."""
    if not polygon or len(polygon) < 3 or frame_width <= 0 or frame_height <= 0 or fps <= 0:
        return intervals
    return [iv for iv in intervals
            if in_court_fraction(iv, points, polygon, fps, frame_width, frame_height, height_pad)
            >= min_in_court_frac]

def in_court_fraction(interval: Tuple[float, float], points: List[Any], polygon: Polygon,
                      fps: float, frame_width: float, frame_height: float,
                      height_pad: float = 0.0) -> float:
    """Fraction of a candidate rally's VISIBLE shuttle points that lie inside the prominent court.

    ``interval`` is ``(start_s, end_s)``; points carry ``.frame``. Returns 1.0 when there are no
    visible points in the interval (a window with no shuttle is not an out-of-court false positive —
    leave that judgement to the windower)."""
    if not polygon or len(polygon) < 3 or frame_width <= 0 or frame_height <= 0 or fps <= 0:
        return 1.0
    fs, fe = interval[0] * fps, interval[1] * fps
    n = inside = 0
    for p in points:
        if getattr(p, "visible", False) and fs <= p.frame <= fe:
            n += 1
            if admitted(p.x / frame_width, p.y / frame_height, polygon, height_pad):
                inside += 1
    return 1.0 if n == 0 else inside / n

def point_in_polygon(point: Tuple[float, float], polygon: Sequence[Tuple[float, float]]) -> bool:
    """Ray-casting point-in-polygon. `point` and `polygon` share a coordinate space
    (we use normalised 0-1). Empty/degenerate polygon (<3 vertices) ⇒ False."""
    if polygon is None or len(polygon) < 3:
        return False
    x, y = point
    inside = False
    n = len(polygon)
    j = n - 1
    for i in range(n):
        xi, yi = polygon[i]
        xj, yj = polygon[j]
        # Does the horizontal ray at y cross edge (i, j)?
        if (yi > y) != (yj > y):
            x_cross = (xj - xi) * (y - yi) / (yj - yi) + xi if yj != yi else xi
            if x < x_cross:
                inside = not inside
        j = i
    return inside

def admitted(nx: float, ny: float, polygon: Polygon, height_pad: float = 0.0) -> bool:
    """Is the normalised point inside the court floor polygon, OR within ``height_pad`` above it?

    The pseudo-3D air column: shifting the point DOWN by ``height_pad`` (toward the floor) and
    finding it inside means the original sits within ``height_pad`` above the floor → a high shuttle
    over the prominent court, admit it."""
    if point_in_polygon((nx, ny), polygon):
        return True
    return height_pad > 0.0 and point_in_polygon((nx, ny + height_pad), polygon)

    """

    points = TrackNetRunner.parse_trajectory_csv(video["trajectory"])
