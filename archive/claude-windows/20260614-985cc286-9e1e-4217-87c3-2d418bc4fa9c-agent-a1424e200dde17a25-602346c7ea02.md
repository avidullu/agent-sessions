32	def bootstrap_ci(values: Sequence[float], confidence: float = 0.95,

34	    """Percentile bootstrap CI for the mean of ``values``. Deterministic given ``seed``.

69	def paired_sign_test(deltas: Sequence[float], eps: float = 0.0) -> Dict[str, float]:
70	    """Two-sided exact sign test over per-fold deltas (B − A).
71	
72	    ``wins`` = folds where B beat A by more than ``eps``; ``losses`` the reverse; ties
73	    excluded. ``p_value`` is the exact two-sided binomial tail under p=0.5 — the
74	    distribution-free "did B win on enough *videos*?" check.
75	    """
76	    wins = sum(1 for d in deltas if d > eps)
77	    losses = sum(1 for d in deltas if d < -eps)
78	    ties = len(deltas) - wins - losses
79	    n_eff = wins + losses
80	    if n_eff == 0:
81	        p = 1.0
82	    else:
83	        k = min(wins, losses)
84	        tail = sum(math.comb(n_eff, j) for j in range(k + 1)) * (0.5 ** n_eff)
85	        p = min(1.0, 2.0 * tail)
86	    return {"wins": float(wins), "losses": float(losses), "ties": float(ties),
87	            "n_effective": float(n_eff), "p_value": p}

at the SERVED windowing.

**Why this exists.** ``per_user_gate.should_promote_for_user`` already grades a B−A lift
honestly (min-N floor + ΔF1 CI excludes 0 + paired sign test). But it grades whatever scores

def served_regresses(a_scores: Sequence[float], b_scores: Sequence[float], *,
                     eps: float = 0.0, n_boot: int = 2000, seed: int = 0) -> Dict[str, Any]:
    """Does variant B REGRESS variant A at the served windowing? (the eval≠serve guard core).

    ``a_scores``/``b_scores`` are per-video held-out scores **measured at the SERVED windowing**
    (aligned by video). Returns the paired B−A summary (mean ΔF1 + bootstrap CI + sign test) plus
    a single ``regresses`` boolean:

      - ``regresses=True``  when the served mean ΔF1 is below ``-eps`` (a real served drop), OR
        the served ΔF1 CI lies **entirely** below ``-eps`` (a confidently-negative band).
      - ``regresses=False`` when the cue holds the line at the served operating point (mean ≥
        ``-eps`` and the CI is not a confident net-negative).

    ``eps`` is the regression tolerance: 0.0 = "must not drop at all on the served mean". A tiny
    positive ``eps`` lets noise-level wobble through (e.g. 0.005 ≈ half an F1 point at n≈6).
    """
    delta = paired_delta_ci(a_scores, b_scores, n_boot=n_boot, seed=seed)
    mean_delta = float(delta["mean_delta"])
    ci_low = float(delta["ci_low"])
    ci_high = float(delta["ci_high"])
    # Regression if the served mean dropped beyond tolerance, OR the whole CI is net-negative.
    mean_regresses = mean_delta < -eps
    ci_confidently_negative = ci_high < -eps
    regresses = bool(mean_regresses or ci_confidently_negative)
    return {
        "regresses": regresses,
        "mean_delta": mean_delta,
        "ci_low": ci_low,
        "ci_high": ci_high,
        "ci_excludes_zero": bool(delta["ci_excludes_zero"]),
        "sign_test": delta["sign_test"],
        "n": int(delta["n"]),
        "eps": float(eps),
    }


@dataclass
class ServedGateResult:
    """A promotion decision that BOTH measures the cue on the served windowing AND reports the
    proposal-windowing view, so the eval≠serve contrast is never hidden.

    - ``promote`` — final verdict. ``True`` only when the underlying honest gate would promote
      *and* the served check did not veto (the cue does not regress at the served windowing).
    - ``served_safe`` — the eval≠serve guard's verdict in isolation (cue does not regress served).
    - ``vetoed_by_served`` — ``True`` when the proposal/base gate WOULD have promoted but the
      served regression check overruled it (the exact Gate-A failure mode this module prevents).
    - ``base_decision`` — the ``PromotionDecision`` from ``should_promote_for_user`` on the
      *promotion-grade* scores (served by default — see :func:`promote_if_served_safe`).
    - ``served`` / ``proposal`` — the paired B−A summary at each windowing (means + CIs + sign
      tests), so a reviewer sees "+0.074 proposal / −0.008 served" side by side.
    - ``reason`` — short human-readable explanation of the final verdict.

    promote: bool
    served_safe: bool
    vetoed_by_served: bool
    reason: str
    base_decision: PromotionDecision
    served: Dict[str, Any]
    proposal: Optional[Dict[str, Any]] = field(default=None)

def promote_if_served_safe(
    served_a: Sequence[float],
    served_b: Sequence[float],
    *,
    proposal_a: Optional[Sequence[float]] = None,
    proposal_b: Optional[Sequence[float]] = None,
    structural: bool = False,
    min_n: int = 5,
    p_threshold: float = 0.05,
    served_regress_eps: float = 0.0,
    n_boot: int = 2000,
    seed: int = 0,
) -> ServedGateResult:
    """Promote a cue ONLY if it does not regress at the SERVED windowing (the foundational rule).

    The honest bar (``should_promote_for_user``: min-N floor + ΔF1 CI excludes 0 + sign test) is
    applied to the **served** scores — the operating point production runs — so a proposal-only
    win can never carry a promotion. The optional ``proposal_*`` scores are scored too and
    reported alongside, purely to surface the eval≠serve contrast (they NEVER promote on their
    own).

    Args:
      served_a/served_b: per-video held-out scores for variant A (control) and B (the cue),
        measured at the **SERVED** windowing. These drive the decision.
      proposal_a/proposal_b: the same scores at the PROPOSAL windowing (optional). Reported for
        contrast; the served regression guard still has the final say.
      structural: forwarded to ``should_promote_for_user`` — a structural guard (e.g. Gate-A vs
        held-shuttle FPs) is never *disabled* on "no lift", but it is ALSO not auto-*promoted* to
        the served default if it regresses served. So for ``structural=True`` we report
        ``keep_on`` honestly but ``promote`` reflects the served-safety check.
      served_regress_eps: served regression tolerance (0.0 = "no served-mean drop allowed").

    Returns a :class:`ServedGateResult` with the final verdict + both windowings' evidence.
    """
    # 1) The honest gate, applied to the SERVED scores (never the proposal ones).
    base = should_promote_for_user(
        served_a, served_b,
        structural=structural, min_n=min_n, p_threshold=p_threshold,
        n_boot=n_boot, seed=seed,
    )

    # 2) The eval≠serve guard: does B regress A at the served windowing?
    served = served_regresses(served_a, served_b, eps=served_regress_eps,
                              n_boot=n_boot, seed=seed)
    served_safe = not served["regresses"]

    # 3) Optional proposal-windowing view (reported, never promoting on its own).
    proposal: Optional[Dict[str, Any]] = None
    if proposal_a is not None and proposal_b is not None:
        proposal = served_regresses(proposal_a, proposal_b, eps=served_regress_eps,
                                    n_boot=n_boot, seed=seed)

    # 4) Compose the verdict. The served guard can only ever REMOVE a promotion, never add one.
    if structural:
        # A structural guard ships ON regardless of measured lift (base.keep_on stays True), but
        # it is only auto-promoted to the SERVED default when it is served-safe. If it regresses
        # served, promote=False (don't bake an eval-only win into production) while keep_on holds.
        promote = served_safe
        vetoed = False
        if served_safe:
            reason = ("structural guard kept ON; served-safe (does not regress at the served "
                      f"windowing, ΔF1 {served['mean_delta']:+.4f}) → eligible for served default")
        else:
            reason = ("structural guard kept ON, but NOT promoted to the served default: it "
                      f"REGRESSES at the served windowing (ΔF1 {served['mean_delta']:+.4f}); "
                      "an eval-only win must not be served")
    else:
        would_promote = base.promote
        if would_promote and not served_safe:
            promote = False
            vetoed = True
            reason = ("VETOED by the served-windowing guard: the cue clears the honest bar but "
                      f"REGRESSES at the served windowing (served ΔF1 {served['mean_delta']:+.4f}"
                      f", CI [{served['ci_low']:+.4f}, {served['ci_high']:+.4f}]). A win on a "
                      "windowing production doesn't serve cannot promote.")
        elif would_promote:
            promote = True
            vetoed = False
            reason = ("promoted: clears the honest bar AND is served-safe (served ΔF1 "
                      f"{served['mean_delta']:+.4f}). " + base.reason)
        else:
            promote = False
            vetoed = False
            reason = base.reason

    return ServedGateResult(
        promote=promote,
        served_safe=served_safe,
        vetoed_by_served=vetoed,
        reason=reason,
        base_decision=base,
        served=served,
        proposal=proposal,
    )

def gate_a_scores_at(points: List[Any], gts: List[Interval], fps: float, frame_width: float,
                     preset: W.WindowingPreset, iou: float = IOU,
                     min_crossings: int = MIN_CROSSINGS, merge_gap: float = 1.0
                     ) -> Tuple[float, float]:
    """One video's (CONTROL F1, Gate-A F1) at ``preset``'s windowing.

    Builds candidates under ``preset``, then scores two static variants vs ``gts``:
      - CONTROL  — every candidate window, overlap-merged (the un-gated baseline).
      - Gate-A   — keep only windows whose shuttle net-crossing count >= ``min_crossings``.

    ``merge_gap`` is the post-keep overlap-merge gap (the harness's ``merge_overlaps`` default),
    independent of the windowing preset's own coalescing gap. Returns the two F1 scores."""
