420	def _parse_secs(value: Optional[str]) -> Optional[float]:
421	    """Decimal seconds or mm:ss / hh:mm:ss -> float; None if blank/unparseable."""
422	    s = (value or "").strip()
423	    if not s:
424	        return None
425	    if ":" in s:
426	        try:
427	            total = 0.0
428	            for part in s.split(":"):
429	                total = total * 60 + float(part)
430	            return total
431	        except ValueError:
432	            return None
433	    try:
434	        return float(s)
435	    except ValueError:
