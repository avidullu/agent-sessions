257	def _probe_video_fps_width(path: str):
258	    """Robust ffprobe (fps, frame_width) for a real video. Returns None on ANY failure.
259	
260	    The detector trajectory is frame-indexed; the harness maps frame->time via fps, so a WRONG
261	    fps silently mis-aligns every rally label (e.g. a 60fps clip read as 30 doubles every time).
262	    cv2's ``_probe_fps_width`` silently defaults to (30, 1280) on a read miss — fine for the
263	    inference report (it flags the fallback) but corrupting for REAL training data. So here we
264	    probe with ffprobe (same tool as ``get_video_duration``) and return None so the caller SKIPS
265	    rather than guesses.
266	    """
267	    import json
268	    import subprocess
269	    try:
270	        out = subprocess.check_output(
271	            ["ffprobe", "-v", "error", "-select_streams", "v:0",
272	             "-show_entries", "stream=r_frame_rate,width", "-of", "json", path],
273	            stderr=subprocess.DEVNULL).decode()
274	        st = (json.loads(out).get("streams") or [{}])[0]
275	        num, _, den = str(st.get("r_frame_rate", "0/1")).partition("/")
276	        fps = float(num) / float(den or "1")
277	        width = float(st.get("width") or 0)
278	        if fps > 0 and width > 0:
279	            return fps, width
280	    except (subprocess.SubprocessError, ValueError, KeyError, OSError, json.JSONDecodeError):
281	        pass
282	    return None
