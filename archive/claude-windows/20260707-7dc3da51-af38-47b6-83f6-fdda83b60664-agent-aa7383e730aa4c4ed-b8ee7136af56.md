+## 5. Source Scanning
+
+Live scanning is opt-in:
+
+1. A first live message from an unknown Telegram channel/group creates a

55	class PolicyBudget:
56	    """Bounded execution budget for opt-in deep inspection."""
57	
58	    max_bytes_per_item: int
59	    timeout_s: float
60	    max_concurrent: int
61	    _semaphore: asyncio.Semaphore | None = field(
62	        default=None,
63	        init=False,
64	        repr=False,
65	        compare=False,
66	    )
67	
68	    def __post_init__(self) -> None:
69	        if self.max_concurrent > 0:
70	            object.__setattr__(self, "_semaphore", asyncio.Semaphore(self.max_concurrent))
71	
72	    async def inspect(self, fetcher: ContentFetcher, content_item: ContentItem) -> StageResult:
73	        """Run ``fetcher`` under the timeout and concurrency budget."""
74	        if self._semaphore is None:
75	            raise ValueError("max_concurrent must be positive")
76	
77	        loop = asyncio.get_running_loop()
78	        deadline = loop.time() + self.timeout_s
79	        await self._acquire_before(deadline)
80	        try:
81	            remaining = deadline - loop.time()
82	            if remaining <= 0:
83	                raise PolicyBudgetExceeded("timeout_s", timeout_s=self.timeout_s)
84	            return await _await_with_budget_timeout(
85	                fetcher.inspect(content_item, self),
86	                timeout_s=remaining,
87	                configured_timeout_s=self.timeout_s,
88	            )
89	        finally:
90	            self._semaphore.release()
91	
92	    async def _acquire_before(self, deadline: float) -> None:
93	        assert self._semaphore is not None
94	        remaining = deadline - asyncio.get_running_loop().time()
95	        if remaining <= 0:
96	            raise PolicyBudgetExceeded("timeout_s", timeout_s=self.timeout_s)
97	        try:
98	            await asyncio.wait_for(self._semaphore.acquire(), timeout=remaining)
99	        except TimeoutError as exc:
100	            raise PolicyBudgetExceeded("timeout_s", timeout_s=self.timeout_s) from exc

103	async def _await_with_budget_timeout(
104	    awaitable: Awaitable[StageResult],
105	    *,
106	    timeout_s: float,
107	    configured_timeout_s: float,
108	) -> StageResult:
109	    task = asyncio.ensure_future(awaitable)
110	    done, _pending = await asyncio.wait({task}, timeout=timeout_s)
111	    if task not in done:
112	        task.cancel()
113	        with contextlib.suppress(asyncio.CancelledError):
114	            await task
115	        raise PolicyBudgetExceeded("timeout_s", timeout_s=configured_timeout_s)
116	    return await task

+## 10. Deduplication
 
-The userbot will not forward the same video twice thanks to a **three-tier

+Dedup is automatic and metadata-first:
 
 | Tier | Key | Catches |
 |---|---|---|
-| **0** | `(user_id, channel_id, message_id)` | The same message instance (re-edits, restarts). |
-| **1** | `(user_id, document_id)` | The same Telegram file **forwarded** across channels — Telegram reuses `document.id`. DB-enforced via a partial unique index. |
-| **2** | `(size_bytes, duration_s, height, width)` | A re-upload of the identical file (new `document.id`, same bytes). Heuristic lookup, no DB constraint. |
-
-This is fully automatic and adds no bandwidth cost — it relies only on metadata
-already present in the message. A re-encode (different resolution/codec/size)
-is **intentionally not deduped** — it is a different file and is evaluated
-independently by the filter engine.
+| 0 | source/message | Same Telegram message instance. |
+| 1 | Telegram `document.id` | Same file forwarded across channels. |
+| 2 | size/duration/height/width | Re-upload of identical media metadata. |

-`document_id` column is added with `ALTER TABLE` (nullable; existing rows get
