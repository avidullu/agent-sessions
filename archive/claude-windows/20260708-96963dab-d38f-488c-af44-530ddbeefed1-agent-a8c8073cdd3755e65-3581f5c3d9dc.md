104	class JobWaiting(Exception):
105	    """Raised inside the pipeline to SELF-SCHEDULE a job (EXECUTION_POLICY.md P3,
106	    WAIT_AND_RESUME). It means "I can't finish now — park me and resume in `delay_sec`",
107	    NOT a failure. The worker persists `next_poll_at` + `progress` (via `set_job_waiting`),
108	    releases the thread, and re-claims the job when due (`claim_due_job`). No work to date
109	    is the segmenter's `_execute_processing`, run UNDER the destroy-after-confirm contract.
110	
111	    The wiring is additive: the production segmenter path never raises this today (zero
112	    behaviour change), so a job runs exactly as before. The hook is what a later slice
113	    (resumable AI handoff) raises once a long op exhausts its in-process wait budget.
114	    """
115	
116	    def __init__(self, delay_sec: float, progress: Optional[str] = None) -> None:
117	        super().__init__(f"job parked for {delay_sec:.0f}s")
118	        self.delay_sec = max(0.0, float(delay_sec))
119	        self.progress = progress

145	def _run_claimed_job(job: Dict[str, Any], config: Any, db: Any) -> None:
146	    """Run ONE job already claimed (`status='running'`) by `claim_due_job`, recording its
147	    terminal state — OR parking it for resume on `JobWaiting` (WAIT_AND_RESUME).
148	
149	    Job `status` is the EXECUTION state (queued|waiting|running|done|failed): 'done' means
150	    the worker finished — the pipeline's own verdict (validation failure etc.) lives in
151	    result["status"]; 'failed' means the worker itself crashed/raised; 'waiting' means the
152	    job self-scheduled and will be re-claimed when `next_poll_at` is due.
153	    """
154	    import backend.main as _main
155	
156	    job_id = job["id"]
157	    # Dispatch by kind: 'compile' (#329, the async reel stitch) vs 'process' (NULL/default — the
158	    # original /api/process pipeline). Both record their terminal state the same way; only process
159	    # jobs feed the M8 cost ledger + M11 flywheel (a compile has no detection usage/segments).
160	    is_compile = job.get("kind") == "compile"
161	    try:
162	        if is_compile:
163	            result = _main._execute_compile(
164	                config,
165	                db,
166	                job,
167	                progress=lambda stage: db.set_job_progress(job_id, stage),
168	            )
169	        else:
170	            req = _main._job_to_request(job)
171	            result = _main._execute_processing(
172	                config,
173	                db,
174	                req,
175	                progress=lambda stage: db.set_job_progress(job_id, stage),
176	                # #524 win #2: hand the drain the submit-time md5 already on the job row
177	                # (``submit_time_video_id``, #484 — server-derived, byte-identical to
178	                # compute_video_id) so it can skip re-hashing the fetched clip. "" ⇒ unknown
179	                # (local upload / non-gcs URI) → the drain hashes as before.
180	                trusted_video_id=job.get("video_id") or None,
181	            )
182	        if result.get("video_id"):
183	            db.set_job_video_id(job_id, result["video_id"])
184	        db.mark_job_done(job_id, result)
185	        if not is_compile:
186	            _maybe_record_job_cost(
187	                job, result, config, db
188	            )  # M8 cost ledger (default-OFF; process jobs only)
189	            _maybe_run_flywheel(
190	                job, result, config
191	            )  # M11 data-flywheel (process jobs only)
192	    except _main.JobWaiting as w:
193	        # Cooperative re-enqueue: persist progress + next_poll_at, release the worker.
194	        logger.info(f"Job {job_id} parked for {w.delay_sec:.0f}s (resume when due).")
195	        db.set_job_waiting(job_id, w.delay_sec, progress=w.progress)
196	    except Exception as e:
197	        logger.error(f"Job {job_id} crashed: {e}\n{traceback.format_exc()}")
198	        db.mark_job_failed(job_id, f"{type(e).__name__}: {e}")

266	def _drain_due_jobs(config: Any, db: Any) -> int:
267	    """One worker tick: claim and run every currently-runnable job (queued, or waiting whose
268	    `next_poll_at` is due), earliest-due first, until the table has nothing runnable.
269	
270	    Each parked job (`JobWaiting` → status 'waiting') drops out of this drain and is picked
271	    up by a FUTURE drain once due; we schedule that follow-up drain via `_schedule_drain` so
272	    a single-worker box keeps making progress with no central timer (the DB is the clock).
273	    Returns the number of jobs that reached a terminal/parked state this tick.
274	    """
275	    import backend.main as _main
276	
277	    ran = 0
278	    while True:
279	        job = db.claim_due_job()
280	        if job is None:
281	            break
282	        ran += 1
283	        _main._run_claimed_job(job, config, db)
284	    # If anything is parked-but-not-yet-due, arrange a wake-up so it resumes without a new
285	    # submit. On the single-box executor this is a delayed re-drain; multi-worker just relies
286	    # on the next submit/drain. Best-effort — never raises into the worker.
287	    _main._schedule_next_drain_if_waiting(config, db)
288	    return ran

291	def _schedule_next_drain_if_waiting(config: Any, db: Any) -> None:
292	    """If any job is parked in 'waiting', arm a re-drain for when the soonest one is due, so a
293	    self-scheduled job resumes with no further client submit. The drain itself re-checks
294	    due-ness, so an early wake is harmless and a clamped wake for a far-future job is fine."""
295	    import backend.main as _main
296	
297	    try:
298	        delay = db.seconds_until_next_due()
299	    except Exception as e:  # pragma: no cover - defensive; never wedge the worker
300	        logger.error(f"Could not compute next-due delay: {e}")
301	        return
302	    if delay is None:
303	        return  # nothing waiting
304	    # Clamp so a far-future job doesn't pin a thread forever; the drain re-checks due-ness.
305	    _main._arm_wake_timer(max(0.0, min(delay, _main._MAX_DRAIN_WAKE_SEC)), config, db)

308	def _arm_wake_timer(delay_sec: float, config: Any, db: Any) -> None:
309	    """Arm a single daemon timer that re-submits a drain after ``delay_sec`` (no central
310	    scheduler — the DB is the clock). Isolated as the sole timer-creating seam so tests can
311	    monkeypatch it to a no-op and assert the *scheduling decision* without real waiting."""
312	    import backend.main as _main
313	
314	    timer = threading.Timer(
315	        delay_sec,
316	        lambda: _main._get_executor().submit(_main._drain_due_jobs, config, db),
317	    )
318	    timer.daemon = True
319	    timer.start()
320

322	#: A parked job further out than this is re-checked by a capped wake rather than one long
323	#: sleeping timer — keeps the wake cadence bounded without a central scheduler.
324	_MAX_DRAIN_WAKE_SEC = 60.0

339	def resume_interrupted_remote_jobs(config: Any, db: Any) -> int:
340	    """Re-arm the durable done-signal poll for md5-keyed RUNNING process jobs a restart
341	    orphaned (the rows ``recover_stale_jobs`` deliberately left alone). Call once at startup,
342	    after ``recover_stale_jobs`` and before any drain. Each job resumes on its own daemon
343	    thread (bounded by the handful of rows a single-instance CP can have in flight); rows are
344	    NEVER re-dispatched from here. Returns the number of resumes started.
345	
346	    On a box that cannot dispatch to Cloud Run at all (the appliance), no remote worker can
347	    exist — the rows get today's honest terminal failure instead."""
348	    rows = db.interrupted_remote_process_jobs()
349	    if not rows:
350	        return 0
351	    if not _cloud_dispatch_possible(config):
352	        for job in rows:
353	            db.mark_job_failed(str(job["id"]), "interrupted by server restart")
354	        logger.warning(
355	            "[JOBS] %d interrupted md5-keyed process job(s) failed: this server cannot "
356	            "dispatch to Cloud Run, so no remote worker can be in flight.",
357	            len(rows),
358	        )
359	        return 0
360	    # Prove each row was actually CLOUD-dispatched (#494 review round 2): an explicit
361	    # compute='cloud' always was; NULL means "server default", which is remote ONLY when the
362	    # default target itself is cloud_run (on a default-local server with cloud-override
363	    # capability, a NULL row ran in-process — no remote worker exists to resume).
364	    default_is_cloud = (
365	        getattr(getattr(config, "deployment", None), "compute_target", "") == "cloud_run"
366	    )
367	    started = 0
368	    for job in rows:
369	        # Same normalization the execution path applies (_maybe_dispatch_remote lowercases
370	        # req.compute) — "LOCAL"/" cloud " must mean at recovery exactly what they meant at run.
371	        requested = (str(job.get("compute") or "")).strip().lower()
372	        if requested != "cloud" and not default_is_cloud:
373	            db.mark_job_failed(str(job["id"]), "interrupted by server restart")
374	            continue
375	        t = threading.Thread(
376	            target=_resume_one_remote_job,
377	            args=(config, db, dict(job)),
378	            daemon=True,
379	            name=f"resume-{str(job['id'])[:8]}",
380	        )
381	        t.start()
382	        started += 1
383	    if started:
384	        logger.warning(
385	            "[JOBS] re-armed the durable done-signal poll for %d interrupted remote job(s).",
386	            started,
387	        )
388	    return started

391	def _resume_one_remote_job(config: Any, db: Any, job: Dict[str, Any]) -> None:
392	    """One orphaned job's resume: poll ``outputs/<md5>/report.json`` (bounded by the same
393	    ``deployment.remote_poll_max_wait_sec`` budget a live dispatch gets), then complete or
394	    fail the job HONESTLY from what the bucket says. Never dispatches."""
395	    from backend import orchestration, storage
396	    from backend.infrastructure.execution_policy import (
397	        ExecutionPolicy,
398	        PolicyTimeout,
399	        poll_until,
400	    )
401	
402	    job_id = str(job["id"])
403	    video_id = str(job["video_id"])
404	    try:
405	        raw_params = job.get("params")
406	        params = (
407	            raw_params
408	            if isinstance(raw_params, dict)
409	            else json.loads(raw_params or "{}")
410	        )
411	        if params.get("force"):
412	            # Defense-in-depth (#494 review): the SQL carve-out already excludes forced rows,
413	            # but a forced job must NEVER complete from the bucket cache — force deliberately
414	            # bypasses it, and nothing proves the report belongs to the forced run rather
415	            # than an older completed one.
416	            db.mark_job_failed(
417	                job_id,
418	                "control plane restarted during a forced reprocess; the cached result "
419	                "cannot be trusted for force=true — re-submit with force",
420	            )
421	            return
422	        if (str(job.get("compute") or "")).strip().lower() == "local":
423	            # Defense-in-depth (#494 review round 2): an explicit local run has no remote
424	            # worker; a same-md5 bucket report would be an OLDER run's output, not this job's.
425	            db.mark_job_failed(job_id, "interrupted by server restart")
426	            return
427	        sc = orchestration._storage_settings(config)
428	        output_root = ((sc or {}).get("output_root") or "").rstrip("/")
429	        if not sc or not output_root:
430	            db.mark_job_failed(
431	                job_id,
432	                "interrupted by server restart (no output bucket configured to recover from)",
433	            )
434	            return
435	        report_uri = f"{output_root}/outputs/{video_id}/report.json"
436	        max_wait = float(
437	            getattr(getattr(config, "deployment", None), "remote_poll_max_wait_sec", 3900.0)
438	            or 3900.0
439	        )
440	        policy = ExecutionPolicy(
441	            poll_every_n_sec=_RESUME_POLL_EVERY_SEC,
442	            op_timeout_sec=120.0,
443	            max_wait_sec=max_wait,
444	        )
445	        db.set_job_progress(job_id, "recovering: waiting for the in-flight cloud run")
446	        try:
447	            poll_until(
448	                lambda: True if storage.exists(report_uri, config=sc) else None,
449	                policy,
450	                label=f"resume-{job_id[:8]}",
451	                logger=logger,
452	                # The #493 live heartbeat: job.progress moves once per still-waiting tick.
453	                on_tick=lambda elapsed: db.set_job_progress(
454	                    job_id,
455	                    "recovering: waiting for the in-flight cloud run — "
456	                    f"{int(elapsed) // 60} min",
457	                ),
458	            )
459	        except PolicyTimeout:
460	            db.mark_job_failed(
461	                job_id,
462	                "control plane restarted mid-run and the cloud worker's result never "
463	                "appeared within the poll budget; re-submit the video (a finished result "
464	                "re-hydrates instantly)",
465	            )
466	            return
467	        payload = orchestration.cached_process_result(
468	            db, config, video_id, str(job.get("video_path") or "")
469	        )
470	        if payload is not None:
471	            db.mark_job_done(job_id, payload)
472	            logger.warning(
473	                "[JOBS] resumed job %s: ingested the in-flight worker's result for %s.",
474	                job_id,
475	                video_id,
476	            )
477	            return
478	        # The report exists but did not realize into a usable result — surface its verdict.
479	        status = "unreadable"
480	        try:
481	            report = orchestration._read_bucket_json(report_uri, sc) or {}
482	            status = str(report.get("status") or "unreadable")
483	        except Exception:  # noqa: BLE001 — the failure message is best-effort
484	            pass
485	        db.mark_job_failed(
486	            job_id,
487	            f"cloud worker finished without a usable result after a control-plane restart "
488	            f"(report status: {status}); re-submit the video",
489	        )
490	    except Exception as e:  # noqa: BLE001 — a resume crash must land as a job failure, not vanish
491	        logger.error("[JOBS] resume of job %s crashed: %s", job_id, e, exc_info=True)
492	        try:
493	            db.mark_job_failed(job_id, f"restart recovery failed: {e}")
494	        except Exception:  # noqa: BLE001
495	            pass
