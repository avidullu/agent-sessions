32	    (we use normalised 0-1). Empty/degenerate polygon (<3 vertices) ⇒ False."""
33	    if polygon is None or len(polygon) < 3:
34	        return False

36	    inside = False

38	    j = n - 1
39	    for i in range(n):
40	        xi, yi = polygon[i]
41	        xj, yj = polygon[j]
42	        # Does the horizontal ray at y cross edge (i, j)?
43	        if (yi > y) != (yj > y):
