+"""Guard the git-versioned Roora golden label CSVs (data/golden_labels/).
+
+These files are the human-verified record of the paid Roora annotation pilot
+(the only other copy is the GCS archive at
+gs://khelsutra-rally-corpus/archive/roora-pilot-review-20260606/). The tests
+pin (a) that the multi-sport goldens stay loadable through the canonical CSV
+loader with zero skipped rows, and (b) that the badminton review record keeps
+its key human corrections (they exist nowhere upstream — the as-delivered
+vendor parse in data/roora_badminton_canonical.csv does NOT contain them).
+"""
+import csv
+import os
+import warnings
+
+from src.parsers.cvat.rally_cvat import load_canonical_csv
+
+GOLDEN_DIR = os.path.join(os.path.dirname(__file__), "..", "data", "golden_labels")
+
+
+def _load_clean(name):
+    """load_canonical_csv, asserting no malformed-row warning was raised."""
+    with warnings.catch_warnings(record=True) as caught:
+        warnings.simplefilter("always")
+        records = load_canonical_csv(os.path.join(GOLDEN_DIR, name))
+    assert not caught, f"{name}: loader warned: {[str(w.message) for w in caught]}"
+    return records
+
+
+def test_pickleball_golden_loads_cleanly():
+    records = _load_clean("pickleball_rallies.csv")
+    assert len(records) == 6
+    assert all(r.end_time > r.start_time for r in records)
+    assert [r.rally_number for r in records] == list(range(1, 7))
+
+
+def test_tabletennis_golden_loads_cleanly():
+    records = _load_clean("tabletennis_rallies.csv")
+    assert len(records) == 6
+    assert all(r.end_time > r.start_time for r in records)
+    assert [r.rally_number for r in records] == list(range(1, 7))
+
+
+def test_badminton_golden_keeps_human_corrections():
+    path = os.path.join(GOLDEN_DIR, "golden_labels_badminton.csv")
+    with open(path, newline="", encoding="utf-8") as f:
+        rows = {row["rally_number"]: row for row in csv.DictReader(f)}
+
+    assert len(rows) == 26
+
+    # Rally 2 start was human-corrected from the vendor's 0:02.503 to 0:29.
+    assert rows["2"]["start_time"] == "29.000"
+    assert rows["2"]["status"].startswith("CORRECTED")
+    # Rally 7 end was corrected from 1:51 to 1:38.
+    assert rows["7"]["end_time"] == "98.000"
+    # Rallies 7.5 and 16.5 were MISSED by the vendor and added by human review.
+    assert rows["7.5"]["status"] == "MISSED-added"
+    assert rows["16.5"]["status"] == "MISSED-added"
