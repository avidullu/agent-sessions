145	def _rally_points(spec: RallySpec, net_midline: float, frame_height: float
146	                  ) -> List[Tuple[int, int, float, float]]:
147	    """Deterministic in-flight shuttle: sine oscillation across the net midline (play axis = x)."""
148	    rows: List[Tuple[int, int, float, float]] = []
149	    y_mid = frame_height / 2.0
150	    for i in range(spec.n_frames):
151	        x = net_midline + spec.amplitude * math.sin(2.0 * math.pi * i / spec.period)
152	        # A small orthogonal wobble keeps y-spread < x-spread so the play axis is unambiguously x.
153	        y = y_mid + 30.0 * math.sin(2.0 * math.pi * i / (spec.period / 2.0))
154	        rows.append((spec.start_frame + i, 1, x, y))
155	    return rows
156	
157	
158	def _gap_points(spec: GapSpec, net_midline: float, frame_height: float
159	                ) -> List[Tuple[int, int, float, float]]:
160	    """Dead-time rows: resting-visible (still) or absent (occluded)."""
161	    rows: List[Tuple[int, int, float, float]] = []
162	    y_mid = frame_height / 2.0
163	    for i in range(spec.n_frames):
164	        if spec.kind == "occluded":
165	            rows.append((spec.start_frame + i, 0, 0.0, 0.0))
166	        elif spec.kind == "still":
167	            rows.append((spec.start_frame + i, 1, net_midline, y_mid))
168	        else:  # pragma: no cover - guarded by builder
169	            raise ValueError(f"unknown gap kind {spec.kind!r} (use 'still'|'occluded')")
170	    return rows

# --------------------------------------------------------------------------------------
# Helpers — manufacture a SYNTHETIC "real" source (trajectory + labels CSV) in tmp_path.
# --------------------------------------------------------------------------------------


def _write_source(tmp_path, *, frame_offset: int = 0):
    """Write a synthetic source trajectory+labels CSV pair into ``tmp_path``.

    Reuses M1's first built-in scenario (a clean single rally), optionally shifted by
    ``frame_offset`` frames (and the matching label seconds) to model an arbitrary absolute
    capture timeline. Returns ``(traj_path, labels_path, fps, frame_width, gts)``.
    """
    spec = gf.builtin_specs()[0]  # clean_single_rally
    fps, fw = spec.fps, spec.frame_width
    rows = gf.make_synthetic_trajectory(spec)
    if frame_offset:
        rows = [(f + frame_offset, v, x, y) for (f, v, x, y) in rows]
    gts = list(spec.gts)
    if frame_offset:
        off_sec = frame_offset / fps
        gts = [(s + off_sec, e + off_sec) for (s, e) in gts]

    traj = tmp_path / "src_trajectory.csv"
    labels = tmp_path / "src_labels.csv"
    gf.write_trajectory_csv(rows, traj)
    gf.write_labels_csv(gts, labels)
    return traj, labels, fps, fw, gts

@pytest.mark.parametrize(

        ({"minors_free": True, "owned": True, "license": "   "}, "license=blank"),

def test_real_fixture_is_deattributed(tmp_path):
    """A successful safe freeze is de-attributed: kind=cleared_real, fx_-prefixed surrogate slug,

        fps=fps, frame_width=fw, fixtures_dir=fixtures)

    slug = expected["slug"]
    assert slug.startswith("fx_"), f"surrogate slug must be opaque fx_<hex>, got {slug!r}"

    manifest = gf.load_manifest(fixtures)
    rows = [e for e in manifest if e["slug"] == slug]

def test_time_origin_reset_is_metric_invariant(tmp_path):
    """A large-offset source frozen WITH reset has IDENTICAL score to the SAME source frozen
    WITHOUT reset — but its absolute control_segments shift toward 0 (severed timeline)."""
    fps = gf.builtin_specs()[0].fps
    offset_frames = 9000  # +5 minutes at 30fps — a large absolute capture-timeline offset
    traj, labels, fw = None, None, None

    # Two independent sources from the SAME synthetic scenario, both offset identically.
    traj, labels, fps, fw, _ = _write_source(tmp_path, frame_offset=offset_frames)

    with_reset = gf.freeze_real_fixture(
        "scenario", traj, labels, slug="fx_reset_on",
        license="owned", minors_free=True, owned=True,
        fps=fps, frame_width=fw, fixtures_dir=tmp_path / "with",
        time_origin_reset=True)
    without_reset = gf.freeze_real_fixture(
        "scenario", traj, labels, slug="fx_reset_off",
        license="owned", minors_free=True, owned=True,
        fps=fps, frame_width=fw, fixtures_dir=tmp_path / "without",
        time_origin_reset=False)

    # (1) Score is bit-identical — metrics.score is built from pure differences.
    assert with_reset["score"] == without_reset["score"], (
        "time-origin reset must be metric-invariant (score built from pure differences)")
    assert with_reset["gt_hash"] != without_reset["gt_hash"] or True  # hash MAY differ (shifted gts)

    # (2) Absolute timeline differs — the reset pulls control_segments toward 0.
    seg_on = with_reset["control_segments"]
    seg_off = without_reset["control_segments"]
    assert len(seg_on) == len(seg_off) and len(seg_on) >= 1
    assert seg_on[0][0] < seg_off[0][0], "reset must shift the first segment start toward 0"
    # The shift equals the offset in seconds (within rounding).
    shift = seg_off[0][0] - seg_on[0][0]
    assert abs(shift - offset_frames / fps) < 1e-3, (
        f"segment shift {shift} should equal the frame offset in seconds {offset_frames / fps}")
    # Reset trajectory starts at frame 0.
    first_frame = int((tmp_path / "with" / "fx_reset_on" / "trajectory.csv")
                      .read_text(encoding="utf-8").splitlines()[1].split(",")[0])
    assert first_frame == 0, "with reset, the trajectory must start at Frame 0"
