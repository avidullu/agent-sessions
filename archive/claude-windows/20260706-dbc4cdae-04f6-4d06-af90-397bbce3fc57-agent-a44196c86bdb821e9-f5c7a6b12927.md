204	        }
205	        # Optional prominent-court gate inputs (multicourt G2; default absent ⇒ the gate is a no-op):
206	        # a normalized court_polygon + optional frame_height + court_height_pad from the manifest.
207	        for k in ("court_polygon", "frame_height", "court_height_pad"):
208	            if k in v:
209	                entry[k] = v[k]
210	        out.append(entry)
211	    return out

214	def windows_for(
215	    video: Dict[str, Any], preset: WindowingPreset, min_crossings: int = 0
216	) -> List[Interval]:
217	    """Parse a video's trajectory, window it under ``preset`` → predicted rally intervals.
218	
219	    ``min_crossings`` > 0 applies the **net-crossings rally-state gate** (Gate-A,
220	    ``rally_gate.gate_windows``): drop windows whose shuttle crosses the net fewer than this many
221	    times (a non-rally fragment). This is the cheapest rally-state cue (W2) — CPU-only, derived
222	    from the trajectory + an auto-estimated net axis; it cleans up the over-split that bounded
223	    windowing (W1) leaves behind.
224	
225	    NEVER-EMPTY safeguard (mirrors ``FusionSegmenter._select_for_handoff``): if the gate would
226	    drop EVERY window of a video that had ≥1, fall back to the ungated windows. The gate is
227	    strictly *pruning* — it never zeroes out a non-empty video — so W2-on-W1 is "do no harm".
228	    """
229	    from backend.pipeline.detectors.tracknet_runner import TrackNetRunner
230	
231	    points = TrackNetRunner.parse_trajectory_csv(video["trajectory"])
232	    wins = windowing.build_windows(points, video["fps"], video["frame_width"], preset)
233	    ivs = windowing.windows_to_intervals(wins)
234	    # Prominent-court WINDOW gate (multicourt G2, default-OFF): when the video carries a normalized
235	    # court_polygon, drop candidate rallies whose shuttle is MAJORITY outside the prominent court — a
236	    # background/adjacent-court rally. Window-level (not point-level): a background rally's central
237	    # tail keeps a few in-court points, enough to hold a window, so point-clearing alone is too leaky
238	    # (measured on GX010142 rally #1). No polygon ⇒ no-op. See docs/PROMINENT_COURT_DETECTION.md.
239	    poly = video.get("court_polygon")
240	    if poly and ivs:
241	        from backend.pipeline.detectors.court_gate import gate_intervals_to_court
242	
243	        fw = float(video["frame_width"])
244	        fh = float(video.get("frame_height") or fw * 9.0 / 16.0)
245	        ivs = gate_intervals_to_court(
246	            ivs,
247	            points,
248	            poly,
249	            float(video["fps"]),
250	            fw,
251	            fh,
252	            min_in_court_frac=float(video.get("court_min_in_court_frac", 0.5)),
253	            height_pad=float(video.get("court_height_pad", 0.0) or 0.0),
254	        )
255	    if min_crossings > 0 and ivs:
256	        from backend.pipeline.detectors.rally_gate import gate_windows
257	
258	        gated = [
259	            (g.start, g.end)
260	            for g in gate_windows(
261	                points, ivs, video["fps"], min_crossings=min_crossings
262	            )
263	            if g.kept
264	        ]
265	        # Gating emptied a non-empty video → keep the ungated windows.
266	        ivs = gated if gated else ivs
267	    return ivs

274	def predict_all(
275	    golden: List[Dict[str, Any]], preset: WindowingPreset, min_crossings: int = 0
276	) -> List["PredictedVideo"]:
277	    """Window every scorable golden video ONCE (the expensive trajectory-parse + windowing step),
278	    returning ``(name, court_type, preds, gts)`` per video. Lets callers re-score the same
279	    predictions under several lenses (e.g. the all-vs-long stratified report) without re-windowing."""
280	    out: List[PredictedVideo] = []
281	    for v in golden:
282	        if not v["gts"] or not os.path.exists(v["trajectory"]):
283	            continue
284	        out.append(
285	            (
286	                v["name"],
287	                court_type_for(v),
288	                windows_for(v, preset, min_crossings),
289	                v["gts"],
290	            )
291	        )
292	    return out

313	    golden: List[Dict[str, Any]],
314	    preset: WindowingPreset,
315	    iou: float = DEFAULT_IOU,
316	    min_crossings: int = 0,
317	    tau_start: Optional[float] = None,
318	    tau_end: Optional[float] = None,

322	    rows + the aggregate. Each row carries both the tIoU set and the R2 tolerance-match block, plus
323	    its ``court_type``. ``min_duration`` > 0 applies the long-rally lens (preds + GTs filtered to
324	    rallies longer than that many seconds) consistently across both metric paths."""
325	    rows = _score_predicted(
326	        predict_all(golden, preset, min_crossings),
327	        iou,
328	        tau_start,
329	        tau_end,
330	        min_duration,

334	        "iou": iou,
335	        "min_crossings": min_crossings,
336	        "min_duration": min_duration,
337	        "rows": rows,
338	        "aggregate": aggregate(rows),

342	def compare(
343	    golden: List[Dict[str, Any]],
344	    base: WindowingPreset,
345	    cand: WindowingPreset,
346	    iou: float = DEFAULT_IOU,
347	    base_min_crossings: int = 0,
348	    cand_min_crossings: int = 0,
349	    tau_start: Optional[float] = None,
350	    tau_end: Optional[float] = None,
351	    guard_weight_merge: Optional[float] = None,
352	    guard_weight_split: Optional[float] = None,
353	    guard_use_rates: bool = True,
354	    min_duration: float = 0.0,
355	) -> Dict[str, Any]:
356	    """Baseline vs candidate (windowing and/or net-crossings gate): per-video paired Δ (+ CI +
357	    sign-test) on both tIoU-F1 and the R2 ``tol_f1`` — the per-tier 'number'. Also reports Δ on the
358	    over-segmentation metrics AND the **R1 net over-seg promotion guard** (``over_seg_guard``):
359	    the honest promote/block verdict for cand-over-base, with the strict per-video rule reported
360	    alongside so the refinement is auditable. ``min_duration`` > 0 scores both sides through the
361	    long-rally lens (same filter applied to base and cand — it's a scoring lens, not a windowing
362	    knob, so it is identical on both sides of the A/B)."""
363	    from backend.eval import tolerance_metrics as tm
364	
365	    b = evaluate(
366	        golden, base, iou, base_min_crossings, tau_start, tau_end, min_duration
367	    )
368	    c = evaluate(
369	        golden, cand, iou, cand_min_crossings, tau_start, tau_end, min_duration
370	    )
371	    by_name_b = {r["name"]: r for r in b["rows"]}
372	    paired = [(by_name_b[r["name"]], r) for r in c["rows"] if r["name"] in by_name_b]
373	    delta: Dict[str, Any] = {}
374	    for k in (
375	        "f1",
376	        "f1@0.25",
377	        "tol_f1",
378	        "merge_rate",
379	        "split_rate",
380	        "segment_count_ratio",
381	    ):
382	        delta[k] = eval_stats.paired_delta_ci(
383	            [rb[k] for rb, _ in paired], [rc[k] for _, rc in paired]
384	        )
385	    gkw: Dict[str, Any] = {"use_rates": guard_use_rates}
386	    if guard_weight_merge is not None:
387	        gkw["weight_merge"] = guard_weight_merge
388	    if guard_weight_split is not None:
389	        gkw["weight_split"] = guard_weight_split
390	    guard = tm.over_seg_guard([rb for rb, _ in paired], [rc for _, rc in paired], **gkw)
391	    return {
392	        "base": b,
393	        "cand": c,
394	        "n_paired": len(paired),
395	        "delta": delta,
396	        "over_seg_guard": guard,
397	    }

400	def stratified_report(
401	    golden: List[Dict[str, Any]],
402	    preset: WindowingPreset,
403	    iou: float = DEFAULT_IOU,
404	    min_crossings: int = 0,
405	    tau_start: Optional[float] = None,
406	    tau_end: Optional[float] = None,
407	    long_tau: float = LONG_RALLY_TAU,
408	) -> Dict[str, Any]:
409	    """The long-rally lens payoff: ``{all vs long(>long_tau)} × {single vs multicourt}`` aggregates.
410	
411	    The thesis: local OVER-FIRES, and its false positives are mostly SHORT (motion blips,
412	    background-court fragments on multicourt footage). Filtering to long rallies should strip those
413	    FPs → precision (``tol_precision``) should jump, most on multicourt. This table is how we read
414	    "do we beat Gemini on the single-match long-rally happy path?".
415	
416	    Windowing is run ONCE per video (``predict_all``); each cell re-scores the same predictions
417	    under its duration lens. A cell aggregates only videos with >=1 golden rally in that stratum
418	    (``num_gt`` > 0): a clip with no long rally is not evidence about long-rally quality and its
419	    0/0 recall would distort the mean. ``n`` per cell reports the videos that survived that gate."""
420	    predicted = predict_all(golden, preset, min_crossings)
421	    long_label = f"long>{long_tau:g}s"
422	    strata = (("all", 0.0), (long_label, long_tau))
423	    courts = ("single", "multicourt", "all")
424	    cells: Dict[Tuple[str, str], Dict[str, Any]] = {}
425	    for dur_label, min_dur in strata:
426	        rows = _score_predicted(predicted, iou, tau_start, tau_end, min_dur)
427	        for court in courts:
428	            sub = [
429	                r
430	                for r in rows
431	                if (court == "all" or r["court_type"] == court) and r["num_gt"] > 0
432	            ]
433	            cells[(dur_label, court)] = aggregate(sub)
434	    return {
435	        "long_tau": long_tau,
436	        "labels": [s[0] for s in strata],
437	        "courts": list(courts),
438	        "cells": cells,
439	        "preset": preset.to_dict(),
440	        "iou": iou,
441	        "min_crossings": min_crossings,
442	    }

573	def _build_preset(args: argparse.Namespace, side: str) -> WindowingPreset:
574	    """Resolve the ``base`` (``--preset``) or ``cand`` (``--vs``) windowing from ``args`` by
575	    applying the ``dataclasses.replace`` overrides for cap / hard-cap / blob-fallback /
576	    inactivity. ``cand`` falls back to the base override when its ``--vs-*`` variant is unset
577	    (so a baseline-vs-candidate promotion check runs in one shot)."""
578	    from dataclasses import replace
579	
580	    if side == "base":
581	        preset = _resolve_preset(args.preset)
582	        cap = args.max_window_duration
583	        hard_cap = args.hard_cap
584	        blob_fallback = args.blob_fallback
585	        inactivity_end = args.inactivity_end
586	        temporal_density = args.inactivity_temporal_density
587	    else:
588	        preset = _resolve_preset(args.vs)
589	        cap = (
590	            args.vs_max_window_duration
591	            if args.vs_max_window_duration is not None
592	            else args.max_window_duration
593	        )
594	        hard_cap = args.vs_hard_cap or args.hard_cap
595	        blob_fallback = args.vs_blob_fallback or args.blob_fallback
596	        inactivity_end = (
597	            args.vs_inactivity_end
598	            if args.vs_inactivity_end is not None
599	            else args.inactivity_end
600	        )
601	        temporal_density = (
602	            args.vs_inactivity_temporal_density or args.inactivity_temporal_density
603	        )
604	    if cap is not None:
605	        preset = replace(preset, max_window_duration=cap)
606	    if hard_cap:
607	        preset = replace(preset, hard_cap=True)
608	    if blob_fallback:
609	        preset = replace(preset, blob_fallback=True)
610	    if temporal_density:
611	        preset = replace(preset, inactivity_temporal_density=True)
612	    if args.blob_factor is not None:
613	        preset = replace(preset, blob_factor=args.blob_factor)
614	    if inactivity_end is not None:
615	        preset = replace(preset, inactivity_end=inactivity_end)
616	    if args.inactivity_rally_thresh is not None:
617	        preset = replace(preset, inactivity_rally_thresh=args.inactivity_rally_thresh)
618	    if args.inactivity_min_density is not None:
619	        preset = replace(preset, inactivity_min_density=args.inactivity_min_density)
620	    return preset

def _greedy_tiou_f1(preds: List[Interval], gts: List[Interval], iou_thresh: float) -> float:
    """Last-resort self-contained greedy tIoU-F1 (match if temporal-IoU >= thresh, 1:1)."""
    if not preds or not gts:
        return 0.0

    def _iou(a: Interval, b: Interval) -> float:
        inter = max(0.0, min(a[1], b[1]) - max(a[0], b[0]))
        union = (a[1] - a[0]) + (b[1] - b[0]) - inter
        return inter / union if union > 0 else 0.0

    pairs = []
    for i, p in enumerate(preds):
        for j, g in enumerate(gts):
            v = _iou(p, g)
            if v >= iou_thresh:
                pairs.append((-v, i, j))
    pairs.sort()
    used_p: set = set()
    used_g: set = set()
    tp = 0
    for _, i, j in pairs:
        if i in used_p or j in used_g:
            continue
        used_p.add(i)
        used_g.add(j)
        tp += 1
    fp = len(preds) - tp
    fn = len(gts) - tp
    precision = tp / (tp + fp) if (tp + fp) else 0.0
    recall = tp / (tp + fn) if (tp + fn) else 0.0
    return (2 * precision * recall / (precision + recall)) if (precision + recall) else 0.0

        cpu_eq_base = _close(ok["cpu"], ok["baseline"])
        verdict = "PASS (gpu == cpu == baseline)" if (gpu_eq_cpu and gpu_eq_base and cpu_eq_base) else \
                  "REVIEW — trajectories diverge"
        print(f"REGRESSION CHECK: {verdict}")
        print(f"  gpu==cpu: {gpu_eq_cpu}   gpu==baseline: {gpu_eq_base}   cpu==baseline: {cpu_eq_base}")
        if not (gpu_eq_cpu and gpu_eq_base and cpu_eq_base):
            print("  (differences in n_pred / f1 / tol_f1 mean the GPU decode path produced a "
                  "different trajectory — investigate before trusting the GPU path.)")
    else:
        missing = {"baseline", "cpu", "gpu"} - set(ok)
        print(f"REGRESSION CHECK: skipped — missing/failed trajectories: {sorted(missing)}")
    print("=" * 78)
