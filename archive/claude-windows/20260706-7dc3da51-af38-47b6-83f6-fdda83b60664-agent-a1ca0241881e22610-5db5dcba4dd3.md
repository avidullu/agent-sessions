def parse_size_mb(value_str: str, unit: str) -> int:
    """Parse a size value + unit string into mebibytes.

    >>> parse_size_mb("10", "MB")
    10
    >>> parse_size_mb("1", "GB")
    1024
    >>> parse_size_mb("500", "KB")
    1

    Note: returns the raw arithmetic result; callers must enforce
    ``FilterConfig`` domain constraints (e.g. ``max_size_mb >= 1``).
    """
    value = int(value_str)
    unit = unit.lower().rstrip("b")  # "MB" → "m", "KB" → "k", "GB" → "g"
    multipliers: dict[str, int] = {"": 1, "m": 1, "k": 1, "g": 1024, "t": 1024 * 1024}
    if unit in multipliers:
        raw = value * multipliers[unit]
        if unit == "k":
            raw = max(1, raw // 1024)
        return raw
    raise ValueError(f"Unknown size unit: {unit}")


def parse_duration_s(value_str: str, unit: str) -> int:
    """Parse a duration value + unit string into seconds.

    >>> parse_duration_s("15", "m")
    900
    >>> parse_duration_s("2", "h")
    7200
    >>> parse_duration_s("90", "s")
    90

    Note: returns the raw arithmetic result; callers must enforce
    ``FilterConfig`` domain constraints (e.g. max >= min).
    """
    value = int(value_str)
    unit = unit.lower()
    multipliers: dict[str, int] = {"": 1, "s": 1, "m": 60, "h": 3600}
    if unit in multipliers:
        return value * multipliers[unit]
    raise ValueError(f"Unknown time unit: {unit}")

def _format_duration(seconds: int | None) -> str:
    """Format seconds as a human-readable duration string."""
    if seconds is None:
        return "unlimited"
    if seconds == 0:
        return "0s"
    if seconds < 60:
        return f"{seconds}s"
    if seconds < 3600:
        m, s = divmod(seconds, 60)
        return f"{m}m{s}s" if s else f"{m}m"
    h, rem = divmod(seconds, 3600)
    m, s = divmod(rem, 60)
    parts = f"{h}h"
    if m:
        parts += f"{m}m"
    if s:
        parts += f"{s}s"
    return parts


def _format_size(mb: int | None) -> str:
    """Format mebibytes as a human-readable size string."""
    if mb is None:
        return "unlimited"
    if mb == 0:
        return "0MB"
    if mb >= 1024:
        gb = mb / 1024
        if gb == int(gb):
            return f"{int(gb)}GB"
        return f"{gb:.1f}GB"
    return f"{mb}MB"

def make_config_command_handler(
    user: User,
    conn: aiosqlite.Connection,
    engine_holder: dict[str, FilterEngine],
) -> Callable:
    """Return an async handler for /set and /show config commands.

    The handler only responds to messages the user sends to themselves
    (Saved Messages). It persists config changes via ``db.save_filter_config``
    and hot-reloads the filter engine through ``engine_holder``.

    ``engine_holder`` is a mutable dict that both the live-message handler and
    the backfill handler read from, so a config change is picked up by all
    components immediately.
    """

    async def on_command(event: object) -> None:
        message = getattr(event, "message", None)
        if message is None:
            return

        # Only respond to messages the user sent to themselves.
        if not getattr(message, "out", False):
            return
        peer = getattr(message, "peer_id", None)
        peer_user_id = getattr(peer, "user_id", None) if peer else None
        if peer_user_id != user.id:
            return

        text: str = getattr(message, "message", "") or ""
        # Strip trailing whitespace so mobile keyboards don't silently
        # reject commands (e.g. "/show config " → "/show config").
        text = text.rstrip()
        client: TelegramClient = getattr(event, "client", None)  # type: ignore[assignment]

        # --- /set size <value><unit> ----------------------------------------
        m = _SIZE_RE.match(text)
        if m:
            value_str, unit = m.group(1), m.group(2) or ""
            try:
                new_mb = parse_size_mb(value_str, unit)
            except ValueError:
                await _reply(client, message, "❌ Invalid size format. Use e.g. `/set size 10MB`.")
                return
            if new_mb < 1:
                await _reply(
                    client, message, "❌ Size must be at least 1 MB (`FilterConfig` constraint)."
                )
                return
            cfg = await _update_config(conn, user.id, max_size_mb=new_mb)
            engine_holder[ENGINE_KEY] = FilterEngine(cfg)
            await _reply(
                client,
                message,
                f"✅ Max file size set to **{_format_size(new_mb)}**." f"\n\n{format_config(cfg)}",
            )
            return

        # --- /set length <value><unit> --------------------------------------
        m = _LENGTH_RE.match(text)
        if m:
            value_str, unit = m.group(1), m.group(2) or ""
            try:
                new_secs = parse_duration_s(value_str, unit)
            except ValueError:
                await _reply(
                    client, message, "❌ Invalid duration format. Use e.g. `/set length 15m`."
                )
                return
            # Enforce max_duration_s >= min_duration_s so every video isn't
            # silently rejected (and so the FilterConfig validator passes on
            # next startup).
            current_cfg = await db.load_filter_config(conn, user.id)
            min_secs = current_cfg.min_duration_s or 0
            if new_secs < min_secs:
                await _reply(
                    client,
                    message,
                    f"❌ Max duration ({_format_duration(new_secs)}) must be "
                    f">= min duration ({_format_duration(min_secs)}).",
                )
                return
            cfg = await _update_config(conn, user.id, max_duration_s=new_secs)
            engine_holder[ENGINE_KEY] = FilterEngine(cfg)
            await _reply(
                client,
                message,
                f"✅ Max duration set to **{_format_duration(new_secs)}**."
                f"\n\n{format_config(cfg)}",
            )
            return

        # --- /set min length <value><unit> ----------------------------------
        m = _MIN_LENGTH_RE.match(text)
        if m:
            value_str, unit = m.group(1), m.group(2) or ""
            try:
                new_min = parse_duration_s(value_str, unit)
            except ValueError:
                await _reply(
                    client,
                    message,
                    "❌ Invalid duration format. Use e.g. `/set min length 5s`.",
                )
                return
            # Enforce min_duration_s <= max_duration_s.
            current_cfg = await db.load_filter_config(conn, user.id)
            max_secs = current_cfg.max_duration_s or 0
            if max_secs > 0 and new_min > max_secs:
                await _reply(
                    client,
                    message,
                    f"❌ Min duration ({_format_duration(new_min)}) must be "
                    f"<= max duration ({_format_duration(max_secs)}).",
                )
                return
            cfg = await _update_config(conn, user.id, min_duration_s=new_min)
            engine_holder[ENGINE_KEY] = FilterEngine(cfg)
            await _reply(
                client,
                message,
                f"✅ Min duration set to **{_format_duration(new_min)}**."
                f"\n\n{format_config(cfg)}",
            )
            return

        # --- /set min resolution <N>p --------------------------------------
        m = _MIN_RES_RE.match(text)
        if m:
            new_height = int(m.group(1))
            # Pre-validate for a friendly message (FilterConfig also enforces ge=1).
            if new_height < 1:
                await _reply(client, message, "❌ Resolution must be at least 1p.")
                return
            cfg = await _update_config(conn, user.id, target_height=new_height)
            engine_holder[ENGINE_KEY] = FilterEngine(cfg)
            await _reply(
                client,
                message,
                f"✅ Min resolution set to **{new_height}p**." f"\n\n{format_config(cfg)}",

            state = m.group(2).lower()

            field_map = {
                "gif": "exclude_gif",
                "round": "exclude_round",
                "nosound": "exclude_nosound",
            }
            cfg = await _update_config(conn, user.id, **{field_map[what]: enabled})
            engine_holder[ENGINE_KEY] = FilterEngine(cfg)
            await _reply(
                client,
                message,
                f"✅ {labels[what]} exclusion: **{'ON' if enabled else 'OFF'}**."
                f"\n\n{format_config(cfg)}",
            )
            return

        # Friendly error for invalid /set exclude (e.g. "gif true", missing on/off).
        if _EXCLUDE_HELP_RE.match(text):
            await _reply(
                client,
                message,
                "❌ Usage: `/set exclude <gif|round|nosound> <on|off>`\n"
                "Example: `/set exclude gif on`",
            )
            return

        # --- /show config ---------------------------------------------------
        if _SHOW_CONFIG_RE.match(text):
            cfg = await db.load_filter_config(conn, user.id)
            await _reply(client, message, format_config(cfg))
            return

        # --- /show commands -------------------------------------------------
        if _SHOW_COMMANDS_RE.match(text):
            await _reply(client, message, format_commands())
            return

    return on_command
