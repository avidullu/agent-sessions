124	def roc_auc(y: np.ndarray, p: np.ndarray) -> float:
125	    """Rank-based AUC (Mann-Whitney) with MIDRANK tie handling.
126	
127	    Uses scipy.stats.rankdata (average method) so tied probabilities get averaged ranks —
128	    without this, an all-tied classifier scored a spurious order-dependent 0.25..0.75
129	    instead of 0.5, and dead-time frames produce identical all-zero feature rows (guaranteed
130	    ties), so the headline AUC carried an unquantified bias."""
131	    pos, neg = p[y > 0.5], p[y <= 0.5]
132	    if pos.size == 0 or neg.size == 0:
133	        return float("nan")
134	    ranks = rankdata(p)  # midranks (ties averaged)
135	    return float((ranks[y > 0.5].sum() - pos.size * (pos.size + 1) / 2) / (pos.size * neg.size))

138	def probs_to_segments(probs: np.ndarray, times: np.ndarray, threshold: float,
139	                      merge_gap: float = 1.0, min_dur: float = 1.0, smooth: int = 7) -> List[Interval]:
140	    p = uniform_filter1d(probs, max(1, smooth), mode="nearest")
141	    on = p >= threshold
142	    runs: List[Interval] = []
143	    i, n = 0, len(on)
144	    while i < n:
145	        if on[i]:

147	            while j + 1 < n and on[j + 1]:

156	            merged[-1] = (merged[-1][0], e)

25	    """Parse a timestamp string to float seconds.
26	
27	    Accepts decimal seconds or colon-separated MM:SS / HH:MM:SS.
28	    Raises ValueError on anything unparseable so bad annotations fail loudly.

35	        if len(parts) > 3:
36	            raise ValueError(f"too many ':' segments in timestamp '{value}'")

38	        for part in parts:                 # most-significant first (H, M, S)
