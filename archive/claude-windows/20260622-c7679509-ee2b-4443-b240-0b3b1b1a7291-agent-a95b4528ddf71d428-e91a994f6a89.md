144	    def _backup_before_migrate(self, conn: sqlite3.Connection, from_version: int) -> None:
145	        """Snapshot the DB to ``<db>.bak-v<from_version>`` before an upgrade migration (P2d).
146

148	        failure WARNS but does NOT block the upgrade (the ladder is itself crash-safe; refusing to
149	        start because a backup couldn't be written would be worse than proceeding without one)."""
150	        backup = f"{self.db_path}.bak-v{from_version}"

153	            shutil.copy2(self.db_path, backup)

155	        except (OSError, sqlite3.Error) as exc:

344	                ON user_models(user_id) WHERE is_active = 1

348	    def _migrate_to_v6(self, cursor):
349	        # v6: per-job COST LEDGER (COMPUTE_DECOUPLED_SERVING M8, D8). NULLABLE cost columns on
350	        # `jobs` (NULL = no cost recorded = byte-identical to today; nothing records until
351	        # billing.enabled + a real pricebook) + a VERSIONED `pricebook` table. Cost is stored in
352	        # USD (matches GCP billing = one source of truth); INR is a display conversion at a
353	        # configurable FX rate. ALTER (not recreate) so v1-v5 job rows survive; ADD COLUMNs are
354	        # idempotent so an interrupted upgrade can re-run them. Additive ONLY — no served change.
355	        for col, typ in (
356	            ("owner_id", "TEXT"),               # who pays (the non-owner user / tenant)
357	            ("compute_backend", "TEXT"),        # local_gpu | cloud_run | ...
358	            ("gpu_type", "TEXT"),               # L4 | T4 | ... (selects the per-second GPU rate)
359	            ("gpu_active_seconds", "REAL"),
360	            ("wall_seconds", "REAL"),
361	            ("bytes_out", "INTEGER"),           # egress
362	            ("gemini_cost_usd", "REAL"),        # provider-reported, already USD
363	            ("pricebook_version", "TEXT"),
364	            ("cost_usd", "REAL"),               # computed total (USD = source of truth)
365	            ("cost_breakdown_json", "TEXT"),    # {gpu_seconds: .., egress_gb: .., gemini_usd: ..}
366	        ):
367	            self._add_column_idempotent(cursor, f"ALTER TABLE jobs ADD COLUMN {col} {typ}")
368	        # Partial index keeps the NULL=no-cost fast path churn-free while making per-owner billing cheap.
369	        cursor.execute("CREATE INDEX IF NOT EXISTS idx_jobs_owner ON jobs(owner_id) "
370	                       "WHERE owner_id IS NOT NULL")
371	        # Versioned pricebook: one row per (version, resource, gpu_type). Rates change + differ by
372	        # gpu_type → re-version (never mutate a shipped version, so a recorded cost stays auditable).
373	        cursor.execute("""
374	            CREATE TABLE IF NOT EXISTS pricebook (
375	                version TEXT NOT NULL,
376	                resource TEXT NOT NULL,
377	                gpu_type TEXT NOT NULL DEFAULT '',
378	                unit_price_usd REAL NOT NULL DEFAULT 0.0,
379	                currency TEXT NOT NULL DEFAULT 'USD',
380	                effective_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
381	            )
382	        """)
383	        cursor.execute("CREATE UNIQUE INDEX IF NOT EXISTS idx_pricebook_key "
384	                       "ON pricebook(version, resource, gpu_type)")
385	        # Seed a PLACEHOLDER version with every rate 0.0 → cost_usd computes to 0 (nothing
386	        # user-facing changes). The owner INSERTs a real, calibrated version (real GCP $/GPU-s,
387	        # egress $/GB) and flips billing.pricebook_version to it — never overwrite this seed.
388	        for resource, gpu_type in (("gpu_seconds", "L4"), ("gpu_seconds", "T4"),
389	                                   ("wall_seconds", ""), ("egress_gb", "")):
390	            cursor.execute(
391	                "INSERT OR IGNORE INTO pricebook (version, resource, gpu_type, unit_price_usd, "
392	                "currency) VALUES ('placeholder-v0', ?, ?, 0.0, 'USD')", (resource, gpu_type))
393	        cursor.execute("PRAGMA user_version = 6")

701	    def get_pricebook_rates(self, version: str, gpu_type: Optional[str] = None) -> Dict[str, float]:
702	        """Resolve a flat ``{resource: unit_price_usd}`` for a pricebook ``version``. The per-second
703	        GPU rate is selected by ``gpu_type``; non-GPU resources (wall/egress) carry ``gpu_type=''``.
704	        An unknown version → ``{}`` (so every cost computes to 0.0)."""
705	        gt = gpu_type or ""
706	        rates: Dict[str, float] = {}
707	        with self._connect() as conn:
708	            rows = conn.cursor().execute(
709	                "SELECT resource, gpu_type, unit_price_usd FROM pricebook WHERE version = ?",
710	                (version,)).fetchall()
711	        for r in rows:
712	            res, row_gt, price = r["resource"], (r["gpu_type"] or ""), float(r["unit_price_usd"])
713	            if res == billing.GPU_SECONDS:
714	                if row_gt == gt:        # the GPU rate for THIS gpu_type
715	                    rates[res] = price
716	            else:
717	                rates[res] = price      # wall/egress: gpu_type-independent
718	        return rates

720	    def record_job_cost(self, job_id: str, *, usage: Dict[str, float], pricebook_version: str,
721	                        gpu_type: Optional[str] = None, compute_backend: Optional[str] = None,
722	                        owner_id: Optional[str] = None, margin: float = 0.0) -> Optional[Dict[str, Any]]:
723	        """Compute the job's cost from ``usage`` × the pricebook + persist every cost column.
724	        Returns the stored cost dict, or ``None`` on write failure. The caller gates this on
725	        ``billing.enabled`` (default-OFF) — nothing records until billing is turned on."""
726	        rates = self.get_pricebook_rates(pricebook_version, gpu_type)
727	        # SILENT-UNDER-BILL GUARD: a GPU-seconds AMOUNT with no RATE for this gpu_type would bill at
728	        # $0 indistinguishably from a free run (a new SKU / typo / gpu_type=None). Surface it loudly +
729	        # in the breakdown so the gap is auditable, not invisible. (A missing amount = $0 IS safe.)
730	        gpu_amt = float(usage.get(billing.GPU_SECONDS, 0.0) or 0.0)
731	        unpriced_gpu = gpu_amt > 0 and billing.GPU_SECONDS not in rates
732	        if unpriced_gpu:
733	            logger.warning(
734	                "[BILLING] no pricebook rate for gpu_type=%r in version=%r — %.1f GPU-seconds billed "
735	                "at $0 (UNPRICED; recorded in cost_breakdown_json). Add a rate or fix gpu_type.",
736	                gpu_type, pricebook_version, gpu_amt)
737	        cost_usd, breakdown = billing.compute_cost_usd(usage, rates)
738	        if unpriced_gpu:
739	            breakdown["unpriced_gpu_seconds"] = gpu_amt
740	        if margin:
741	            marked = billing.apply_margin(cost_usd, margin)
742	            breakdown["margin"] = round(marked - cost_usd, 6)   # keep Σ(breakdown) == cost_usd
743	            cost_usd = marked
744	        bytes_out = int(float(usage.get(billing.EGRESS_GB, 0.0) or 0.0) * 1e9)
745	        ok = self._execute_write(
746	            "UPDATE jobs SET owner_id=?, compute_backend=?, gpu_type=?, gpu_active_seconds=?, "
747	            "wall_seconds=?, bytes_out=?, gemini_cost_usd=?, pricebook_version=?, cost_usd=?, "
748	            "cost_breakdown_json=? WHERE id = ?",
749	            (owner_id, compute_backend, gpu_type, usage.get(billing.GPU_SECONDS),
750	             usage.get(billing.WALL_SECONDS), bytes_out, usage.get(billing.GEMINI_USD),
751	             pricebook_version, cost_usd, json.dumps(breakdown), job_id),
752	            f"Failed to record cost for job {job_id}")
753	        if not ok:
754	            return None
755	        return {"cost_usd": cost_usd, "cost_breakdown": breakdown,
756	                "pricebook_version": pricebook_version}
