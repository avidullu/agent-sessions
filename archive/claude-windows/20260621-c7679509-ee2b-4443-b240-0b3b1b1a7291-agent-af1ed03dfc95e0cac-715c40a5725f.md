267	def _probe_video_fps_width(path: str):
268	    """Robust ffprobe (fps, frame_width) for a real video. Returns None on ANY failure.
269	
270	    The detector trajectory is frame-indexed; the harness maps frame->time via fps, so a WRONG
271	    fps silently mis-aligns every rally label (e.g. a 60fps clip read as 30 doubles every time).
272	    cv2's ``_probe_fps_width`` silently defaults to (30, 1280) on a read miss — fine for the
273	    inference report (it flags the fallback) but corrupting for REAL training data. So here we
274	    probe with ffprobe (same tool as ``get_video_duration``) and return None so the caller SKIPS
275	    rather than guesses.
276	    """
277	    import json
278	    import subprocess
279	    try:
280	        out = subprocess.check_output(
281	            ["ffprobe", "-v", "error", "-select_streams", "v:0",
282	             "-show_entries", "stream=r_frame_rate,width", "-of", "json", path],
283	            stderr=subprocess.DEVNULL).decode()
284	        st = (json.loads(out).get("streams") or [{}])[0]
285	        num, _, den = str(st.get("r_frame_rate", "0/1")).partition("/")
286	        fps = float(num) / float(den or "1")
287	        width = float(st.get("width") or 0)
288	        if fps > 0 and width > 0:
289	            return fps, width
290	    except (subprocess.SubprocessError, ValueError, KeyError, OSError, json.JSONDecodeError):
291	        pass
292	    return None

493	def _setup_logging(verbose: bool) -> None:
494	    """Configure logging for the worker CLI so the pipeline's existing INFO/ERROR diagnostics are
495	    VISIBLE — the worker previously ran with no logging config, so every segmenter/detector log
496	    (detector choice, 'Running native WASB inference', availability failures) was silently dropped.
497	    That is exactly why a 0-rally cold-start gave nothing to analyse. Configured here (the CLI
498	    entry), not at import, so importing the module never reconfigures a caller's logging."""
499	    # force=True: basicConfig is a no-op if the root already has handlers (e.g. a library
500	    # configured logging at import), which would leave the level at WARNING and re-suppress the
501	    # diagnostics. Forcing reconfiguration means the worker reliably controls its own verbosity.
502	    logging.basicConfig(
503	        level=logging.DEBUG if verbose else logging.INFO,
504	        format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
505	        datefmt="%H:%M:%S",
506	        force=True,
507	    )
