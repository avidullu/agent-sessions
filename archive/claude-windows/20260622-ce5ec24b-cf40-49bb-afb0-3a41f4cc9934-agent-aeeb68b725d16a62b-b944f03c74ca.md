52	async function request<T>(path: string, opts: RequestOpts = {}): Promise<T> {
53	  const cid = opts.cid ?? newCorrelationId();
54	  const url = `${BASE}${path}`;
55	  const method = opts.method ?? "GET";
56	  log("info", "api.request", { cid, method, url });
57	
58	  let res: Response;
59	  try {
60	    res = await fetch(url, {
61	      method,
62	      body: opts.body,
63	      headers: { "content-type": "application/json", "x-correlation-id": cid, ...(opts.headers ?? {}) },
64	    });
65	  } catch (err) {
66	    // Loud failure, never silent (LOCAL_APPLIANCE_ARCHITECTURE.md §5).
67	    log("error", "api.network_error", { cid, method, url, error: String(err) });
68	    throw new ApiError(`Network error calling ${method} ${url}`, { cid, cause: err });
69	  }
70	
71	  if (!res.ok) {
72	    const body = await res.text().catch(() => "");
73	    log("error", "api.http_error", { cid, method, url, status: res.status, body: body.slice(0, 500) });
74	    throw new ApiError(`${method} ${url} failed: ${res.status}`, { cid, status: res.status, body });
75	  }
76	
77	  const data = (await res.json()) as T;
78	  log("info", "api.response", { cid, method, url, status: res.status });
79	  return data;
