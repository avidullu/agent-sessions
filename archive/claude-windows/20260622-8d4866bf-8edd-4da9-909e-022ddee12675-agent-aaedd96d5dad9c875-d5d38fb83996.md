def compile_highlights(req: CompileRequest):
    video = db_instance.get_video(req.video_id)
    if not video:
        raise HTTPException(status_code=404, detail="Indexed video record not found.")

    # Retrieve matching play segments from db
    all_segments = db_instance.query_play_segments(req.video_id, {})
    matched_segments = [s for s in all_segments if s["id"] in req.segment_ids]

    if not matched_segments:
        raise HTTPException(status_code=400, detail="No matching play segments found to stitch.")

    # Reject path traversal / absolute output names (os.path.join would otherwise let an
    # absolute or '..'-laden output_filename write anywhere on disk).
    try:
        out_name = safe_output_filename(req.output_filename)
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e)) from e

    # The configured shot-count floor (stitching.min_shot_count) applies on the API
    # path too — parity with scripts/run_process_and_stitch.py. Segments WITHOUT shot_count
    # metadata are exempt: the floor filters known counts only, so explicitly
    # requested manual/legacy segments can't silently vanish under the default floor.
    stitching = getattr(global_config, "stitching", None) or StitchingConfig()
    kept = []
    for s in matched_segments:
        meta = s.get("metadata") or {}
        sc = meta.get("shot_count") if isinstance(meta, dict) else None
        try:
            if sc is not None and int(sc) < stitching.min_shot_count:
                continue
        except (TypeError, ValueError):
            pass  # unparseable count -> treat as unknown, keep
        kept.append(s)
    if len(kept) < len(matched_segments):
        logger.info(f"[compile] stitching.min_shot_count={stitching.min_shot_count}: dropped "
                    f"{len(matched_segments) - len(kept)}/{len(matched_segments)} segments below the floor.")
    if not kept:
        raise HTTPException(status_code=400,
                            detail=f"All {len(matched_segments)} matching segments fall below "
                                   f"stitching.min_shot_count={stitching.min_shot_count}.")

    # Extract start/end time pairs
    time_pairs = [(s["start_time"], s["end_time"]) for s in kept]

    # the UI locale the SPA rendered in (req.locale): the reel's watermark reads in the user's language
    # with a script-appropriate font. Unknown/absent locale keeps the configured default unchanged.
    output_dir = getattr(getattr(global_config, "output", None), "output_dir", "output") or "output"
