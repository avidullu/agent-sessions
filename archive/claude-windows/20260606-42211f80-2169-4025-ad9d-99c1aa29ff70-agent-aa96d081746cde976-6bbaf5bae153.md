local function append_row(n, start_s, end_s, why)
  local exists = file_exists(out_path)
  local f, err = io.open(out_path, "a")

-- Remove the last appended data line by rewriting the file without its tail.
-- Lua has no ftruncate, so we read all, drop the tail, and rewrite.

  local f = io.open(out_path, "rb")
  if not f then return false, "CSV not found" end
  local data = f:read("*a") or ""
  f:close()

## 2. Rate workflow
1. Open the badminton video in VLC and start playback. Pause/scrub freely.
2. When a rally begins, click **Mark START** (snapshots the current time).
3. Pick the point-stop reason in the **Ending reason** dropdown
   (winner / forced_error / unforced_error / service_fault / let / other).
4. When the rally ends, click **Mark END** -- one CSV row is appended
   immediately. The dropdown value at that moment is what gets written.
5. Made a mistake? **Undo last** disarms an uncommitted START, or removes the
   last written row (one level of undo).
6. The status label shows: last action, current time, whether a START is armed,
   rallies written, and the **CSV output path**.

**Output path:** written next to the video as `<video-stem>.rallies.csv`
when VLC can resolve the file path; otherwise it falls back to your home dir
(`%USERPROFILE%\rally_labels.csv`). The exact path is always shown in the dialog.

**CSV format:** header `rally_number,start_time,end_time,ending_reason`,
times in **decimal seconds**. Re-enabling the extension keeps appending to the
same file (it does not truncate).

## 3. Ingest with the collector
Once you have the CSV, import it directly (do NOT trim/re-encode the video --
the pipeline keys on the file checksum):

```
python -m src.cli.curate import-local --sport badminton \

  --annotations-path "C:\path\to\match.rallies.csv"
```

(Add `--fps <F>` / `--video-id <ID>` if your collector workflow uses them.)
The importer lowercases the header and reads `start_time`/`end_time`
(required, decimal seconds), `rally_number` (optional, auto-numbered if blank),
and `ending_reason` (carried verbatim); any other columns are ignored.

decimal seconds via `csv.DictReader`.

## 4. Notes & gotchas
- VLC's `time` is in MICROSECONDS; the extension divides by 1e6 -> seconds.
- It does not auto-react to pause on purpose (that callback is broken in VLC 4.0
  and flaky on macOS). You explicitly click START/END, which is more precise.
- Keep `ending_reason` to the 6 dropdown values -- downstream QA enforces the
  closed vocabulary; the importer itself does not validate it.
- A rally must be > 0s: if END is not after START it is rejected (with a hint),
  which keeps the file free of degenerate rows the loaders would drop anyway.

APPROACH: Button-dialog, not a pause-hook. The button-click model is event-driven (VLC invokes each button's callback on its main loop), so it never blocks VLC the way a polling loop in activate() would (keep_alive() only resets the watchdog, it cannot make a loop responsive). It is portable: the {"playing-listener"}/status_changed pause callback is unreliable on macOS (VLC #22778) and the input/meta listeners are broken in VLC 4.0 (#27558), so the extension declares NO listener capabilities. It is also more precise and ergonomic: the rater can pause and scrub to the exact frame, THEN press Mark START / Mark END, instead of being forced to commit a time at whatever instant a pause fires. Each callback reads vlc.var.get(input,"time") (microseconds in 3.x) and divides by 1e6 for the decimal seconds the collector requires.

    f:write(HEADER)
  end
  local line = string.format("%d,%.3f,%.3f,%s\n", n, start_s, end_s, why)
  f:write(line)
  f:close()
  last_line_len = #line   -- track byte length so Undo can truncate it
  return true
end

-- Remove the last appended data line by rewriting the file without its tail.
-- Lua has no ftruncate, so we read all, drop the tail, and rewrite.
local function remove_last_row()
  if not last_line_len then return false, "nothing to undo" end
  local f = io.open(out_path, "rb")
  if not f then return false, "CSV not found" end
  local data = f:read("*a") or ""
  f:close()
  if #data < last_line_len then return false, "CSV shorter than last row" end
  local kept = data:sub(1, #data - last_line_len)
  -- If only the header remains, blank the file entirely so a fresh header
  -- is rewritten next time.
  if kept == HEADER then
    kept = ""
  end
  local wf = io.open(out_path, "wb")
  if not wf then return false, "cannot rewrite CSV" end
  wf:write(kept)
  wf:close()
  last_line_len = nil   -- only one level of undo
  return true
end

function mark_end()

  if not t then set_status("No media playing -- cannot mark END."); return end
  if not pending_start then
    set_status("No START armed -- press \"Mark START\" first.")
    return
  end
  local s, e = pending_start, t
  if e <= s then
    -- Tolerate the rater pressing END before START position: swap if it yields

        fmt_clock(e), fmt_clock(s)))
      return
    end
    s, e = e, s
  end
  local _, reason = w_reason:get_value()   -- (id, text); we want the text
  if not reason or reason == "" then reason = "other" end
  local n = rally_count + 1
  local ok, err = append_row(n, s, e, reason)

end
