55	def _finalize_reingest(db, video_id: str, segments, play_wm: int, cand_wm: int) -> None:
56	    """Commit a (re)segmentation with destroy-AFTER-confirm semantics.
57	
58	    On a non-empty run, drop the PRIOR rows (id <= the pre-run watermark) so only the
59	    fresh segments remain. On an empty run, drop the NEW partial rows (id > watermark) and
60	    keep the prior good results intact — a failed/empty re-run never destroys prior data.
61	    """
62	    if segments:
63	        db.prune_segments(video_id, play_upto=play_wm, cand_upto=cand_wm)
64	    else:
65	        db.prune_segments(video_id, play_after=play_wm, cand_after=cand_wm)

107	def _ingest_remote_segments(
108	    db, video_id: str, outputs_uri: str, storage_config: dict, *, max_rows: int = 20000
109	) -> int:
110	    """Read a cloud job's ``intervals.csv`` (start,end,shot_count,ending_reason) from the bucket and
111	    persist each rally via the SAME ``db.add_play_segment`` the in-process segmenters call — so
112	    ``/api/query`` + ``/api/compile`` read the rows with zero knowledge of where they were produced.
113	    Returns the count ingested. Mirrors ``cloud_run._read_json``'s fetch-to-a-real-dst pattern (cloud
114	    backends REQUIRE a dst). Hardened against a hostile/corrupt worker output: a malformed row is
115	    skipped, non-finite / overflowing values are rejected, the loop is bounded, and the temp dir is
116	    cleaned. May raise (missing/unreadable CSV, fetch error) — the caller treats that as a job failure
117	    and prunes any partial rows (destroy-AFTER-confirm)."""
118	    uri = f"{outputs_uri.rstrip('/')}/intervals.csv"
119	    ref = storage.parse_uri(uri)
120	    n = 0
121	    with tempfile.TemporaryDirectory(prefix="cr-ingest-") as td:
122	        local = storage.fetch(
123	            uri, os.path.join(td, ref.name or "intervals.csv"), config=storage_config
124	        )
125	        with open(local, newline="", encoding="utf-8") as f:
126	            for row in csv.DictReader(f):
127	                if n >= max_rows:
128	                    logger.warning(
129	                        "[API] cloud ingest hit the %d-row cap (%s) — ignoring the rest.",
130	                        max_rows,
131	                        uri,
132	                    )
133	                    break
134	                try:
135	                    start, end = float(row["start"]), float(row["end"])
136	                except (KeyError, TypeError, ValueError):
137	                    continue
138	                if not (math.isfinite(start) and math.isfinite(end)):
139	                    continue  # reject inf/nan times rather than persist a junk rally
140	                meta: Dict[str, Any] = {
141	                    "source": "cloud_run",
142	                    "ending_reason": (row.get("ending_reason") or "").strip(),
143	                }
144	                sc = (row.get("shot_count") or "").strip()
145	                if sc:
146	                    try:
147	                        meta["shot_count"] = int(
148	                            float(sc)
149	                        )  # OverflowError on 'inf'/'1e400'
150	                    except (ValueError, OverflowError):
151	                        meta["shot_count"] = sc
152	                db.add_play_segment(video_id, start, end, metadata=meta)
153	                n += 1
154	    return n

157	def _cloud_available(cfg) -> bool:
158	    """Whether THIS server can satisfy a per-request ``compute="cloud"`` dispatch (item 3).
159	
160	    True when the control plane is wired for Cloud Run: either the configured default target is
161	    already ``cloud_run`` (the operator chose cloud serving), or the Cloud Run env coordinates
162	    (``CLOUD_RUN_JOB`` + ``GOOGLE_CLOUD_PROJECT``) and an output bucket (``storage.output_root``)
163	    are present so a forced per-request override can actually dispatch + collect a result.
164	
165	    Surfaced in ``/api/capabilities`` so the SPA only offers the cloud toggle when it would work."""
166	    deployment = getattr(cfg, "deployment", None)
167	    if (getattr(deployment, "compute_target", "") or "") == "cloud_run":
168	        return True
169	    has_job = bool(
170	        os.environ.get("CLOUD_RUN_JOB") and os.environ.get("GOOGLE_CLOUD_PROJECT")
171	    )
172	    out_root = (getattr(getattr(cfg, "storage", None), "output_root", "") or "").strip()
173	    return bool(has_job and out_root)

176	def _maybe_dispatch_remote(
177	    cfg,
178	    db,
179	    req: "ProcessRequest",
180	    video_id: str,
181	    seg_name: str,
182	    val_results,
183	    play_wm: int,
184	    cand_wm: int,
185	    notify,
186	) -> Optional[Tuple[Dict[str, Any], bool]]:
187	    """Cloud-serving (COMPUTE_DECOUPLED_SERVING M6 → the API path): when ``deployment.compute_target
188	    == "cloud_run"``, run the heavy DETECT on a Cloud Run GPU Job and INGEST its rally intervals into
189	    THIS DB, so the rest of the flow (query → compile/local-stitch) is unchanged.
190	
191	    Returns ``(response_dict, ran)`` when the cloud path is taken — ``ran`` is whether the cloud job
192	    actually EXECUTED (``False`` for a pre-dispatch config rejection, so the observability report
193	    doesn't claim a fake segmentation run) — or ``None`` to fall through to the in-process segmenter
194	    (**DEFAULT-OFF**: ``get_compute_backend`` returns ``None`` for every non-``cloud_run`` target, so
195	    the in-process path stays byte-identical)."""
196	    from backend.compute import ComputeJobSpec, get_compute_backend
197	
198	    def _failed(msg: str, ran: bool) -> Tuple[Dict[str, Any], bool]:
199	        # Shape-match the in-process segmenter-fail branch + drop any partial NEW rows (id > play_wm);
200	        # prior good rows (id <= play_wm) are preserved (destroy-AFTER-confirm).
201	        db.prune_segments(video_id, play_after=play_wm, cand_after=cand_wm)
202	        # Provenance even on failure, and HONEST about what ran: `path` is "cloud_run" only when the
203	        # job actually dispatched + executed remotely (ran=True, e.g. a non-zero worker rc / ingest
204	        # error); a pre-dispatch reject (ran=False) ran NOWHERE → "not_run". `target` records the
205	        # intended backend, `requested` the picker choice, `dispatched` whether the remote job ran.
206	        return (
207	            {
208	                "video_id": video_id,
209	                "status": "failed",
210	                "message": msg,
211	                "results": [r.model_dump() for r in val_results],
212	                "play_segments": [],
213	                "compute": {
214	                    "path": "cloud_run" if ran else "not_run",
215	                    "requested": req.compute,
216	                    "target": "cloud_run",
217	                    "dispatched": ran,
218	                },
219	            },
220	            ran,
221	        )
222	
223	    # SPA compute-picker (item 3): a per-request choice overrides the server's configured default.
224	    # "local" forces the in-process segmenter; "cloud" forces a Cloud Run dispatch (guarded by
225	    # _cloud_available so we fail clearly rather than silently running local); None / unknown → default.
226	    choice = (req.compute or "").strip().lower()
227	    if choice and choice not in ("local", "cloud"):
228	        logger.warning(
229	            "[API] ignoring unknown compute=%r (expected 'local'|'cloud'); using server default.",
230	            req.compute,
231	        )
232	        choice = ""
233	    if choice == "local":
234	        return None  # explicit "run on my machine" — in-process regardless of the server's target
235	    if choice == "cloud":
236	        if not _cloud_available(cfg):
237	            return _failed(
238	                "cloud-serving was requested but this server isn't configured for it "
239	                "(no Cloud Run target). Pick 'run on my machine', or configure cloud serving.",
240	                False,
241	            )
242	        compute = get_compute_backend(cfg, target_override="cloud_run")
243	    else:
244	        compute = get_compute_backend(
245	            cfg
246	        )  # server default (default-OFF unless cloud_run)
247	    if compute is None:
248	        return None  # default-OFF → run the in-process segmenter below (unchanged)
249	
250	    storage_cfg = getattr(cfg, "storage", None)
251	
252	    # The Cloud Run job fetches the input FROM THE BUCKET — it can't read a path local to this box.
253	    try:
254	        input_ref = storage.resolve_input_ref(req.video_path, storage_cfg)
255	    except ValueError as e:
256	        return _failed(
257	            f"cloud-serving: unresolvable input '{req.video_path}': {e}", False
258	        )
259	    if input_ref.backend == "local":
260	        # A browser upload lands on THIS box's local disk; the Cloud Run job (a different machine)
261	        # can't read it. Stage it UP to the input bucket first, then dispatch from there (#461). Only
262	        # possible when a CLOUD input_root is configured; without one we genuinely can't stage → fail.
263	        # Classify the ROOT ITSELF (parse_uri), not storage.input_backend: the staged URI below is
264	        # built by joining under input_root, and storage.put routes by parse_uri(staged_uri).backend —
265	        # so an explicit cloud root (gs://…, s3://…) stages fine with input_backend left at its
266	        # default, while a bare/local-looking root can't stage no matter what input_backend says
267	        # (resolve_input_uri only consults input_backend when input_root is EMPTY, and an empty root
268	        # gives us nothing to build the staged URI under).
269	        input_root = (getattr(storage_cfg, "input_root", "") or "").strip()
270	        try:
271	            root_backend = (
272	                storage.parse_uri(input_root).backend if input_root else "local"
273	            )
274	        except ValueError:  # unsupported scheme in input_root — nowhere real to stage to
275	            root_backend = "local"
276	        if root_backend == "local":
277	            return _failed(
278	                "cloud-serving needs a cloud input (e.g. gs://…), or a cloud storage.input_root to "
279	                "stage local uploads into; a path local to this machine can't be read by the Cloud "
280	                "Run job.",
281	                False,
282	            )
283	        local_src = input_ref.key  # resolved local path of the upload
284	        staged_uri = f"{input_root.rstrip('/')}/{video_id}/{os.path.basename(local_src)}"
285	        try:
286	            notify("staging upload to bucket")
287	            # Thread the active storage settings so backend-specific config survives (S3 endpoint_url/
288	            # region/addressing for R2·DO·MinIO; GCS project/client) — not a fresh default client.
289	            storage.put(
290	                local_src,
291	                staged_uri,
292	                config=(
293	                    storage_cfg.model_dump() if storage_cfg is not None else None
294	                ),
295	            )
296	        except Exception as e:  # noqa: BLE001 — a staging failure is a clean job failure, not a 500
297	            logger.error(
298	                "[API] cloud-serving: failed to stage %s -> %s: %s",
299	                local_src,
300	                staged_uri,
301	                e,
302	                exc_info=True,
303	            )
304	            return _failed(
305	                f"cloud-serving: failed to stage the uploaded file to {input_root}: {e}",
306	                False,
307	            )
308	        # Repoint BOTH the dispatch and the DB video record at the staged bucket object: the L4 job
309	        # DETECTs from it, and a later /api/compile re-fetches the same source (the local upload is
310	        # gone once this instance recycles). add_video is an UPSERT — it only updates filepath.
311	        req.video_path = staged_uri
312	        input_ref = storage.resolve_input_ref(staged_uri, storage_cfg)
313	        db.add_video(
314	            video_id, staged_uri, "processed", [r.model_dump() for r in val_results]
315	        )
316	    out_root = (getattr(storage_cfg, "output_root", "") or "").strip()
317	    if not out_root:
318	        return _failed(
319	            "cloud-serving needs storage.output_root set to a bucket (e.g. gs://your-bucket) "
320	            "— that's where the Cloud Run job writes the result.",
321	            False,
322	        )
323	
324	    notify("dispatching (cloud_run)")
325	    logger.info(
326	        "[API] cloud-serving: dispatching %s -> Cloud Run Job (out=%s)",
327	        req.video_path,
328	        out_root,
329	    )
330	    # Forward the engine's AI-handoff stance so the cloud job runs the SAME detection: with
331	    # skip_ai_handoff=True the WASB windows become rallies directly (no Gemini key needed); without
332	    # forwarding the cloud job would default to the handoff and yield 0 rallies when it has no key.
333	    # (player_pool / max_frames are still dropped on the cloud path — a separate documented follow-up.)
334	    overrides = []
335	    sk = getattr(getattr(cfg, "indexing", None), "skip_ai_handoff", None)
336	    if sk is not None:
337	        overrides.append(f"indexing.skip_ai_handoff={'true' if sk else 'false'}")
338	    spec = ComputeJobSpec(
339	        video_uri=req.video_path,
340	        out_uri=out_root,
341	        segmenter=seg_name,
342	        config_overrides=tuple(overrides),
343	    )
344	    try:
345	        result = compute.run_infer(
346	            spec
347	        )  # dispatch the Cloud Run Job + bucket-poll to completion
348	    except Exception as e:  # noqa: BLE001 — any dispatch/poll error is a clean job failure, not a 500
349	        logger.error("[API] cloud dispatch error: %s", e, exc_info=True)
350	        return _failed(f"cloud dispatch error: {e}", True)
351	    if result.returncode != 0:
352	        return _failed(f"cloud job failed (returncode {result.returncode}).", True)
353	    if result.video_id and result.video_id != video_id:
354	        # The worker re-hashes the bucket bytes; a divergence means the input changed between this
355	        # box's hash and the worker's — ingest under the LOCAL id but flag it so it's diagnosable.
356	        logger.warning(
357	            "[API] cloud video_id %s != local %s — input bytes differ between the API hash "
358	            "and the worker hash; ingesting under the local id.",
359	            result.video_id,
360	            video_id,
361	        )
362	
363	    notify("ingesting (cloud_run)")
364	    storage_dict = (
365	        storage_cfg.model_dump()  # type: ignore[union-attr]  # hasattr-guarded; mypy can't narrow
366	        if hasattr(storage_cfg, "model_dump")
367	        else {}
368	    )
369	    try:
370	        _ingest_remote_segments(db, video_id, result.outputs_uri, storage_dict)
371	    except Exception as e:  # noqa: BLE001 — a missing/partial/corrupt result must not 500 or leak rows
372	        logger.error("[API] cloud ingest failed: %s", e, exc_info=True)
373	        return _failed(
374	            f"cloud-serving: could not read the result from {result.outputs_uri}: {e}",
375	            True,
376	        )
377	
378	    # Only the NEW cloud rows (id > the pre-run watermark); _finalize_reingest then drops the old ones.
379	    segments = [
380	        s for s in db.query_play_segments(video_id, {}) if int(s.get("id", 0)) > play_wm
381	    ]
382	    _finalize_reingest(db, video_id, segments, play_wm, cand_wm)
383	    # Compute-path PROVENANCE: the GCP-verifiable proof the cloud path ran. `execution` is the real
384	    # Cloud Run execution name (cross-check: gcloud run jobs executions describe <execution>).
385	    provenance = {
386	        "path": "cloud_run",
387	        "requested": req.compute,
388	        "job": os.environ.get("CLOUD_RUN_JOB", "rally-worker"),
389	        "region": os.environ.get("CLOUD_RUN_REGION", "asia-south1"),
390	        "execution": result.execution,
391	        "outputs_uri": result.outputs_uri,
392	    }
393	    logger.info(
394	        "[COMPUTE] cloud_run dispatched: execution=%s job=%s region=%s out=%s video_id=%s",
395	        result.execution,
396	        provenance["job"],
397	        provenance["region"],
398	        result.outputs_uri,
399	        video_id,
400	    )
401	    return (
402	        {
403	            "video_id": video_id,
404	            "status": "success",
405	            "message": f"Successfully indexed {len(segments)} play segments (cloud_run / '{seg_name}').",
406	            "results": [r.model_dump() for r in val_results],
407	            "play_segments": segments,
408	            "compute": provenance,
409	        },
410	        True,
411	    )

414	def _execute_processing(
415	    config=None, db=None, req=None, progress=None
416	) -> Dict[str, Any]:
417	    """The full hash → validate → segment → report pipeline for one video.
418	
419	    Accepts explicit ``config``/``db`` (the new app-factory pattern) OR reads the
420	    module-level ``global_config``/``db_instance`` (backward-compatible with existing
421	    tests that pass only ``req``). Route handlers always pass explicit params; older
422	    test code that calls ``_execute_processing(req)`` still works.
423	
424	    ``progress`` is an optional callable(stage: str) used to surface coarse job progress.
425	    Raises on unexpected internal errors — the caller records those as a job failure
426	    """
427	    import backend.main as _main  # call-time seam: main's globals/model stay patchable
428	
429	    # Backward-compatible shim: if called as _execute_processing(req) (old pattern),
430	    # shift positional args — req arrives in position 0 (config), progress in position 1 (db).
431	    if config is not None and not isinstance(config, AppConfig):
432	        # Old signature: _execute_processing(req, progress=None)
433	        if req is None and db is not None and isinstance(db, _main.ProcessRequest):
434	            req = db
435	            db = None
436	        elif req is None and isinstance(config, _main.ProcessRequest):
437	            req = config
438	            config = None
439	    # Resolve config/db from module globals when not passed explicitly
440	    if config is None:
441	        config = _main.global_config
442	    if db is None:
443	        db = _main.db_instance
444	
445	    notify = progress or (lambda stage: None)
446	
447	    notify("hashing")
448	    # M2: resolve the input to a LOCAL path. A bare/local path is returned unchanged (identity —
449	    # byte-identical to today); a bucket/Drive URI is fetched to scratch first. The original
450	    # req.video_path (the source URI) is kept for the DB record + report; only the file-reading
451	    # consumers (hash / validators / segmenter) use the resolved local path.
452	    local_video = ""
453	    try:
454	        local_video = storage.acquire_local_input(
455	            req.video_path, getattr(config, "storage", None)
456	        )
457	        video_id = _main.compute_video_id(local_video)
458	    except Exception:
459	        storage.cleanup_acquired_local_input(
460	            req.video_path, local_video, getattr(config, "storage", None)
461	        )
462	        raise
463	    try:
464	        existing_video = db.get_video(video_id)
465	    except Exception:
466	        storage.cleanup_acquired_local_input(
467	            req.video_path, local_video, getattr(config, "storage", None)
468	        )
469	        raise
470	    was_reingest = existing_video is not None
471	
472	    # typed AppConfig: attribute access for the default segmenter (with safe fallback)
473	    default_seg = getattr(
474	        getattr(config, "indexing", None), "default_segmenter", "motion"
475	    )
476	    seg_name = req.segmenter or default_seg
477	    segmenter_actual = None
478	    segmentation_ran = False
479	    val_results: List[Any] = []
480	    val_context: Dict[str, Any] = {}
481	    # Compute-path PROVENANCE (the validation method): which backend actually ran the DETECT —
482	    # "in_process" once the local segmenter is reached, set on the response as "cloud_run" by
483	    # _maybe_dispatch_remote when it dispatches. None ⇒ nothing ran (validation/config fail).
484	    compute_actual: Optional[str] = None
485	    response: Optional[Dict[str, Any]] = None
486	    try:
487	        if (
488	            existing_video
489	            and existing_video.get("status") == "processed"
490	            and not req.force
491	        ):
492	            cached_segments = db.query_play_segments(video_id, {})
493	            if cached_segments:
494	                logger.info(
495	                    "[API] Reusing cached segmentation for video_id=%s; force=true bypasses cache.",
496	                    video_id,
497	                )
498	                response = {
499	                    "video_id": video_id,
500	                    "status": "success",
501	                    "message": (
502	                        f"Reused {len(cached_segments)} cached play segments. "
503	                        "Pass force=true to reprocess this video."
504	                    ),
505	                    "results": existing_video.get("validation_results", []),
506	                    "play_segments": cached_segments,
507	                    "cached": True,
508	                    "compute": {
509	                        "path": "cached",
510	                        "requested": getattr(req, "compute", None),
511	                    },
512	                }
513	                return response
514	
515	        # 1. Run pluggable validators (with-context variant surfaces ffprobe_meta for the report)
516	        notify("validating")
517	        logger.info(f"Running validations for {req.video_path}...")
518	        val_results, val_context = registry.run_all_with_context(local_video, config)
519	        failures = [r for r in val_results if not r.passed]
520	
521	        if failures:
522	            # add_video is an UPSERT (no cascade) — a failed re-validation only updates
523	            # status; it no longer destroys the video's prior good segments.
524	            db.add_video(
525	                video_id,
526	                req.video_path,
527	                "failed",
528	                [r.model_dump() for r in val_results],
529	            )
530	            response = {
531	                "video_id": video_id,
532	                "status": "failed",
533	                "message": _validation_failure_message(failures),
534	                "results": [r.model_dump() for r in val_results],
535	            }
536	            return response
537	
538	        # 2. Run segmenter & indexer
539	        logger.info("[API] Video passed checks. Segmenting and indexing...")
540	        db.add_video(
541	            video_id, req.video_path, "processed", [r.model_dump() for r in val_results]
542	        )
543	
544	        segmenter_cls = segmenter_registry.get(seg_name)
545	        if not segmenter_cls:
546	            # Submit-time validation already 400s unknown names; this is the defensive
547	            # in-worker path (e.g. config default pointing at an unregistered segmenter).
548	            available = list(segmenter_registry.list_available().keys())
549	            response = {
550	                "video_id": video_id,
551	                "status": "failed",
552	                "message": f"Segmenter '{seg_name}' not found. Available segmenters: {available}",
553	                "results": [r.model_dump() for r in val_results],
554	                "play_segments": [],
555	            }

699	def _resolve_compile(
700	    db, cfg, video_id: str, segment_ids: List[int], output_filename: str
701	):
702	    """Shared compile resolution — run at SUBMIT (fast-fail 4xx) AND in the WORKER (re-resolve, since a
703	    row could change between submit and run). Verifies the video exists, the requested segments match
704	    and survive the ``stitching.min_shot_count`` floor (segments WITHOUT a known shot_count are exempt
705	    so manual/legacy picks can't silently vanish), and the output name is traversal-safe. Raises
706	    ``_CompileError(404/400)``. Returns ``(video_row, kept_segments, safe_out_name)``."""
707	    video = db.get_video(video_id)
708	    if not video:
709	        raise _CompileError(404, "Indexed video record not found.")
710	    # Reject path traversal / absolute names (os.path.join would otherwise write anywhere on disk).
711	    try:
712	        out_name = safe_output_filename(output_filename)
713	    except ValueError as e:
714	        raise _CompileError(400, str(e)) from e
715	
716	    all_segments = db.query_play_segments(video_id, {})
717	    matched = [s for s in all_segments if s["id"] in segment_ids]
718	    if not matched:
719	        raise _CompileError(400, "No matching play segments found to stitch.")
720	
721	    stitching = getattr(cfg, "stitching", None) or StitchingConfig()
722	    kept = []
723	    for s in matched:
724	        meta = s.get("metadata") or {}
725	        sc = meta.get("shot_count") if isinstance(meta, dict) else None
726	        try:
727	            if sc is not None and int(sc) < stitching.min_shot_count:
728	                continue
729	        except (TypeError, ValueError):
730	            pass  # unparseable count -> treat as unknown, keep
731	        kept.append(s)
732	    if len(kept) < len(matched):
733	        logger.info(
734	            "[compile] stitching.min_shot_count=%s: dropped %d/%d segments below the floor.",
735	            stitching.min_shot_count,
736	            len(matched) - len(kept),
737	            len(matched),
738	        )
739	    if not kept:
740	        raise _CompileError(
741	            400,
742	            f"All {len(matched)} matching segments fall below "
743	            f"stitching.min_shot_count={stitching.min_shot_count}.",
744	        )
745	    return video, kept, out_name

419	    ``resolution_fps_check`` validator already probed ffprobe and stored ``details={width,height,fps}``

557	def _cached_remote_source(video_id: str, input_ref, cfg) -> str:
558	    """Fetch a remote source once into an output-scoped source/proxy cache."""
559	    cache_path = _source_cache_path(video_id, input_ref, cfg)
560	    if os.path.isfile(cache_path):
561	        return cache_path
562	    lock = _source_cache_lock(video_id)
563	    with lock:
564	        if os.path.isfile(cache_path):
565	            return cache_path
566	        cache_dir = os.path.dirname(cache_path)
567	        os.makedirs(cache_dir, exist_ok=True)
568	        need = _remote_input_size_bytes(input_ref, getattr(cfg, "storage", None))
569	        require_free_bytes(cache_dir, need, stage="source-cache")
570	        tmp = os.path.join(
571	            cache_dir, f".{os.path.basename(cache_path)}.{uuid.uuid4().hex}.tmp"
572	        )
573	        try:
574	            storage.fetch(input_ref.uri, tmp, config=_storage_settings(cfg))
575	            os.replace(tmp, cache_path)
576	        finally:
577	            try:
578	                if os.path.exists(tmp):
579	                    os.remove(tmp)
580	            except OSError:
581	                pass
582	        return cache_path

585	def _warm_annotation_proxy(video_id: str, src_local: str, cfg) -> None:
586	    """Background, best-effort: generate the annotation proxy ONCE so subsequent /api/source loads
587	    serve the lighter, seekable copy. Deduped; never blocks the request; the source keeps serving if
588	    it fails."""
589	    with _proxy_lock:
590	        if video_id in _proxy_inflight:
591	            return
592	        _proxy_inflight.add(video_id)
593	    try:
594	        from backend.utils.annotation_proxy import make_annotation_proxy, needs_proxy
595	
596	        if not needs_proxy(src_local):
597	            return  # already light enough — no proxy worth making
598	        make_annotation_proxy(
599	            src_local, os.path.join(_proxies_dir(cfg), f"{video_id}.mp4")
600	        )
601	    except Exception as e:  # noqa: BLE001 - warming is best-effort
602	        logger.warning("proxy: warm failed for %s: %s", video_id, e)
603	    finally:
604	        with _proxy_lock:
605	            _proxy_inflight.discard(video_id)

608	@app.get("/api/source/{video_id}")
609	def get_source(video_id: str, request: Request):
610	    """Stream a video for in-UI annotation (P8), Range-capable so the browser can scrub. Serves a
611	    frame-preserving PROXY (STUDIO_MEDIA_LIFECYCLE.md D4 — downscaled + seekable, labels still keyed
612	    to the master's MD5) once one exists; otherwise serves the source and WARMS a proxy in the
613	    background for next time — non-blocking, so the first load is never delayed. Remote sources are
614	    fetched once into an output-scoped source cache, then served and proxied from that stable copy."""
615	    rec = request.app.state.db.get_video(video_id)
616	    if not rec or not rec.get("filepath"):
617	        raise HTTPException(status_code=404, detail="Video not found.")
618	    cfg = request.app.state.config
619	
620	    # 1) A ready proxy wins. Bare-name + containment guarded like /api/highlights.
621	    try:
622	        proxy_name = safe_output_filename(f"{video_id}.mp4")
623	        proxy_path = resolve_under_root(
624	            os.path.join(_proxies_dir(cfg), proxy_name), _proxies_dir(cfg)
625	        )
626	    except ValueError:
627	        proxy_path = ""
628	    if proxy_path and os.path.isfile(proxy_path):
629	        return FileResponse(proxy_path, media_type="video/mp4")
630	
631	    # 2) Otherwise resolve + serve the source.
632	    try:
633	        input_ref = storage.resolve_input_ref(
634	            rec["filepath"], getattr(cfg, "storage", None)
635	        )
636	    except (
637	        ValueError
638	    ) as e:  # unsupported scheme etc. — a clean client error, never a 500
639	        raise HTTPException(status_code=400, detail=str(e)) from e
640	    except Exception as e:  # noqa: BLE001 - a fetch miss is a bad-gateway, surfaced (not swallowed)
641	        raise HTTPException(
642	            status_code=502, detail=f"Could not resolve the source video: {e}"
643	        ) from e
644	
645	    if input_ref.backend == "local":
646	        local = input_ref.key
647	    else:
648	        try:
649	            local = _cached_remote_source(video_id, input_ref, cfg)
650	        except HTTPException:
651	            raise
652	        except Exception as e:  # noqa: BLE001 - a fetch miss is surfaced (not swallowed)
653	            raise HTTPException(
654	                status_code=502, detail=f"Could not resolve the source video: {e}"
655	            ) from e
656	    if not os.path.isfile(local):
657	        raise HTTPException(status_code=404, detail="Source video file not found.")
658	
659	    # 3) Warm a proxy in the background for next time.
660	    threading.Thread(
661	        target=_warm_annotation_proxy, args=(video_id, local, cfg), daemon=True
662	    ).start()
663	
664	    media = "video/quicktime" if local.lower().endswith(".mov") else "video/mp4"
665	    return FileResponse(local, media_type=media)

1147	def run_cli_diagnose_clip(video_path: str, timestamp: float):
1148	    """CLI debugging utility to examine segment properties around a timestamp."""
1149	    print("=" * 60)
1150	    print(f"DEBUGGING VIDEO CLIP: {video_path} at {timestamp}s")
1151	    print("=" * 60)
1152	
1153	    cfg = load_config()  # legacy wrapper still works, returns dict for now
1154	
1155	    # 1. Validation check diagnostics
1156	    print("[DIAGNOSTIC] Running validation framework...")
1157	    results = registry.run_all(video_path, cfg)
1158	    for r in results:
1159	        status_symbol = "[PASS]" if r.passed else "[FAIL]"
1160	        print(f"  {status_symbol} {r.validator_name}: {r.message}")
1161	        if r.details:
1162	            print(f"         Details: {r.details}")
1163	
1164	    # 2. Local segment analysis around the timestamp (sampling +/- 5 seconds)
1165	    print("-" * 60)
1166	    print("[DIAGNOSTIC] Analyzing local motion around the timestamp...")
1167	    # Lazy import: cv2 is NOT installed in the LIGHT tier (torch-free default)
1168	    # and this --debug-clip path is CLI-only, not on the served API path.
1169	    import cv2
1170	
1171	    cap = cv2.VideoCapture(video_path)
1172	    if not cap.isOpened():
1173	        print("[FAIL] OpenCV could not open video.")
1174	        return
1175	
1176	    fps = cap.get(cv2.CAP_PROP_FPS)
1177	    total_frames = cap.get(cv2.CAP_PROP_FRAME_COUNT)
1178	
1179	    start_frame = max(0, int((timestamp - 3.0) * fps))
1180	    end_frame = min(int(total_frames), int((timestamp + 3.0) * fps))
1181	
1182	    # Measure frame-by-frame changes in sample region
1183	    cap.set(cv2.CAP_PROP_POS_FRAMES, start_frame)
1184	    prev_gray = None
1185	    motions = []
1186	
1187	    for _ in range(start_frame, end_frame):
1188	        ret, frame = cap.read()
1189	        if not ret:
1190	            break
1191	        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
1192	        gray = cv2.GaussianBlur(gray, (21, 21), 0)
1193	
1194	        if prev_gray is not None:
1195	            diff = cv2.absdiff(prev_gray, gray)
1196	            _, thresh = cv2.threshold(diff, 25, 255, cv2.THRESH_BINARY)
1197	            motion_ratio = cv2.countNonZero(thresh) / (
1198	                thresh.shape[0] * thresh.shape[1]
1199	            )
1200	            motions.append(motion_ratio)
1201	        prev_gray = gray
1202	
1203	    cap.release()
1204	
1205	    if motions:
1206	        avg_motion = sum(motions) / len(motions)
1207	        # Support both legacy dict (from wrapper) and future direct model
1208	        if hasattr(cfg, "indexing"):
1209	            threshold = getattr(cfg.indexing, "motion_threshold", 0.08)
1210	        else:
1211	            threshold = cfg.get("indexing", {}).get("motion_threshold", 0.08)  # type: ignore[attr-defined]
1212	        print(f"  Sample Frame Count: {len(motions)}")
1213	        print(
1214	            f"  Average motion index around timestamp: {round(avg_motion, 4)} (threshold: {threshold})"
1215	        )
1216	        if avg_motion > threshold:
1217	            print(
1218	                "  Result: [ACTIVE PLAY] High motion detected. Segment classifies as active play."
1219	            )
1220	        else:
1221	            print(
1222	                "  Result: [DEAD TIME] Idle movement. Segment classified as dead time."
1223	            )
1224	    else:
1225	        print("  [ERROR] No visual frames were extracted in the sample range.")
1226	
1227	    print("=" * 60)
