82	            ret, frame = cap.read()

84	                break
85	
86	            # Process frame
87	            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
88	            # Apply Gaussian Blur to smooth noise
89	            gray = cv2.GaussianBlur(gray, (21, 21), 0)

91	            t = frame_idx / fps
92	            timestamps.append(t)
93	
94	            if prev_gray is not None:
95	                # Frame absolute difference (highly optimized motion tracking)
96	                diff = cv2.absdiff(prev_gray, gray)
97	                _, thresh = cv2.threshold(diff, 25, 255, cv2.THRESH_BINARY)

99	                # Dilate to fill gaps
100	                thresh = cv2.dilate(thresh, None, iterations=2)
101	
102	                # Motion ratio
103	                motion_ratio = cv2.countNonZero(thresh) / (thresh.shape[0] * thresh.shape[1])
104	                motion_signals.append(motion_ratio)
105	            else:
106	                motion_signals.append(0.0)
107	
108	            prev_gray = gray
109	            frame_idx += 1
110	            processed_count += 1

112	            if progress_step and frame_idx >= next_progress:
113	                pct = 100.0 * frame_idx / total_frames

116	                logger.info(f"[motion] {pct:5.1f}% — {frame_idx}/{total_frames} frames, "
117	                            f"{processed_count} analyzed, {elapsed:.0f}s elapsed, ~{eta:.0f}s left")
118	                next_progress += progress_step
119	
120	            if max_frames is not None and processed_count >= max_frames:
121	                logger.debug(f"Reached max_frames limit of {max_frames}. Halting indexer frame parsing.")
122	                break
123	
124	            if frame_idx >= total_frames:
125	                break
126	
127	        cap.release()
128	
129	        # Smooth signal using a moving average
130	        window_size = 5
131	        smoothed = []
132	        for i in range(len(motion_signals)):
133	            start = max(0, i - window_size // 2)
134	            end = min(len(motion_signals), i + window_size // 2 + 1)
135	            smoothed.append(sum(motion_signals[start:end]) / (end - start))
136	
137	        # Detect active play boundaries
138	        in_segment = False
139	        segment_start = None
140	        raw_segments: List[Tuple[float, float]] = []
141	
142	        for i, t in enumerate(timestamps):
143	            signal = smoothed[i]
144	            if signal > motion_threshold:
145	                if not in_segment:
146	                    in_segment = True
147	                    segment_start = t
148	            else:
149	                if in_segment:

151	                    raw_segments.append((segment_start, t))
152	
153	        # Handle video ending while in active segment
154	        if in_segment:
155	            raw_segments.append((segment_start, timestamps[-1]))
156	
157	        # Post-Processing: 1. Merge segments separated by less than dead_time_merge_gap
158	        merged_segments: List[Tuple[float, float]] = []
159	        if raw_segments:
160	            curr_start, curr_end = raw_segments[0]
161	            for next_start, next_end in raw_segments[1:]:
162	                if next_start - curr_end < dead_time_merge_gap:
163	                    # Merge
164	                    curr_end = next_end
165	                else:
166	                    merged_segments.append((curr_start, curr_end))
167	                    curr_start, curr_end = next_start, next_end
168	            merged_segments.append((curr_start, curr_end))
169	
170	        # Post-Processing: 2. Filter segments shorter than min_segment_duration
171	        final_segments: List[Tuple[float, float]] = [
172	            r for r in merged_segments if (r[1] - r[0]) >= min_segment_duration
173	        ]

## 2. Bucket ingest — can the engine process a GCS video today?

**Yes, but only via the worker CLI — not via the HTTP API.**

- **GCS is a first-class, fully-implemented `StorageBackend`** (`backend/storage/gcs.py`): `fetch_to_local` downloads a blob (`:30-39`), `put` uploads (`:41-43`), plus `list`/`exists`/`stat`/`signed_url`. It lazy-imports `google-cloud-storage` (`:24`) and uses **Application Default Credentials** (`GOOGLE_APPLICATION_CREDENTIALS` SA-JSON or workload identity — never config; `gcs.py` module docstring `:1-6`).
- **URI routing:** `parse_uri` (`backend/storage/ref.py:48-68`) recognizes `gcs://bucket/key` and normalizes `gs://` → `gcs://` (`:60-61`). `__init__.py::fetch/put/list_uris` (`:56-77`) dispatch on the scheme. Config `StorageConfig` (`models.py:295-319`) supports `backend: Literal["local","s3","gcs","gdrive"]` (default `"local"`) with a per-backend `gcs: dict` settings slot (`:318`).
- **The worker `infer` path does the pull → pipeline → push** (`worker/__main__.py:cmd_infer`):
  - `:101` — `storage.fetch(args.video_uri, scratch/in.mp4)` downloads the GCS blob to local scratch.
  - `:105` — computes the `video_id` (MD5 of the pulled bytes).
  - `:130` — runs the configured segmenter (same registry as the API).
  - `:160-175` — serializes `outputs/<video_id>/{intervals.csv,report.json}` and **pushes them back** to `--out-uri` (which can itself be `gcs://`).

  CLI: `python -m backend.worker infer --video-uri gcs://bucket/match.mp4 --out-uri gcs://bucket/out [--segmenter ...] [--set k=v]`.

**Critical gap for the CUJ:** the worker writes `intervals.csv` + `report.json` to a bucket — it does **NOT** write to the API's SQLite `play_segments` table, and it does **NOT** create a `videos` row that `/api/query` / `/api/compile` can read. So a worker `infer` run alone does **not** give the SPA a queryable `video_id`. Conversely, `POST /api/process` only accepts a **local** `video_path` (`ProcessRequest.video_path`, validated by `validate_input_path` + `os.path.exists` at `main.py:264-269`) — it has **no bucket-URI ingest**. There is no HTTP "ingest from `gcs://`" endpoint. This is exactly the "ingest-from-bucket wiring is **net-new**" gap the appliance doc flags (`LOCAL_APPLIANCE_ARCHITECTURE.md` §4.2 row "Bucket URI" = **design**, and the reserved "bucket-worker bridge" T14).

**Other backends:** `s3` (AWS/R2/DO-Spaces/MinIO via `endpoint_url`) is fully implemented (`s3.py`); `gdrive` (`gdrive.py`) reads a **locally-mounted** Google Drive as plain files — usable but not a remote API client.

## 3. Detectors / compute — what a real dry-run needs

Segmenters registered (`segmenters/__init__.py`, `list_available()`): `motion`, `gemini`, `yolo_hybrid`, `tracknet_hybrid`, `wasb_hybrid`, `fusion_hybrid`.

- **`config.json` default = `gemini`** (root `config.json` `indexing.default_segmenter: "gemini"`). (Note: the Pydantic *code* default is `yolo_hybrid` at `models.py:126`, but the shipped `config.json` overrides it to `gemini`.)
- **Gemini path (cloud, real):** `gemini.py:86-93` **requires `GEMINI_API_KEY`** in the environment and raises a clear error if absent. Runs on CPU (uploads downsized chunks to Google). **Needs: a Gemini API key + network. No GPU, no WSL.** This is the realistic CPU dry-run detector.
- **Native WASB GPU path:** `wasb_hybrid` → `TrajectoryHybridSegmenter._resolve_runner` (`trajectory_hybrid.py:65-86`). `detector_impl` ∈ `{auto=WSL, native=Linux-native GPU, stub=CPU mock}` (default `"auto"`, `models.py:133`; `RALLY_DETECTOR_IMPL` env overrides). **Needs a GPU host** (+ WSL for `auto`, or a native Linux GPU box for `native`). `trajectory_hybrid._build_native_runner` still **raises NotImplementedError** (`:57-63`) — only `wasb_hybrid` overrides it with a real `NativeWasbRunner` (`wasb_hybrid.py:40-46`). Native is **owner-byte-parity-gated, not production-verified** (per `LOCAL_APPLIANCE_ARCHITECTURE.md` §3, risk table).
- **`motion` (CPU, real but low-quality):** `motion.py` — pure OpenCV frame-differencing, **no key, no GPU**. BUT it **fabricates** `server`/`receiver` (`random.sample`), `shot_count`, and `events` (`motion.py:191-239`) — which the SPA is explicitly told never to surface (`types.ts:20-26`, `PER_RALLY_SELECTION.md` §3 "Fabricated — do NOT surface"). It produces real *segment boundaries* though.
- **`stub` (CPU mock):** `detectors/stub_runner.py` — deterministic synthetic trajectories, **no GPU/WSL/weights**. Combined with `--set indexing.skip_ai_handoff=true` it is the validated **secret-free, GPU-free** path (`worker/__main__.py:9-11`). Output is **not real detection** ("mock — not real detection" per the doc's honesty matrix).

**CPU-only paths that produce real-ish rallies:** `gemini` (real, needs key) and `motion` (real boundaries, fabricated metadata). `stub` is mock. **For a true "real rallies, no GPU" dry-run, Gemini + a key is the path.** A GPU is only required for native/WSL WASB.

## 4. Reel download to the local machine

**Fully working today.** Two-step, browser-reachable:

1. `POST /api/compile` writes the MP4 into `output_dir` and returns BOTH a server-local `filepath` **and** `download_url` (`main.py:410-411`): `download_url = f"/api/highlights/{quote(video_id)}/{quote(out_name)}"`. **The `download_url` field exists** (the "M1" addition — `PER_RALLY_SELECTION.md` P6 lists it as `☐`/TODO, but it is in the code now).
2. `GET /api/highlights/{video_id}/{filename}` (`main.py:321-342`) streams the file via `FileResponse` with the right media type — the browser-hittable endpoint.

**SPA side:** the SPA does **not** consume `download_url`. Its `CompileResponse` type (`types.ts:33-37`) only declares `status` + `filepath`; after compile it **hand-builds** the URL itself via `highlightUrl(videoId, filename)` (`client.ts:86-88`, called in `App.tsx:133`), then renders an inline `<video src>` preview + an `<a href download>` link (`CompileResult.tsx:48-61`). So download-to-local **works end to end already** for any video that's in the engine's DB — the only redundancy is the SPA ignoring the server-provided `download_url`.

## 5. The SPA gap — what's missing for this CUJ

The SPA is **strictly job-tethered and read-only on ingest**. Hard evidence: a grep of `web/src` for `process|upload|jobs|capabilities|videos|reels|health|download_url|job_id` returned **zero matches**. The SPA's entire API surface is `client.ts`: `listRallies` (`/query`), `compileReel` (`/compile`), `highlightUrl` (`/highlights`), `getConfig` (`/config`). That's it.

Concretely missing:

- **No ingest/process UI at all.** The SPA cannot point at a video (bucket or local), cannot call `/api/process`, cannot call the `/api/upload/*` chunked endpoints, cannot pick a segmenter. It assumes a video is **already processed**.
- **How it gets a `video_id`:** purely from the URL query string — `?video=<id>` (`App.tsx:48`: `new URLSearchParams(...).get("video")`). With no `?video=`, it shows hardcoded `DEMO_RALLIES` (`App.tsx:30-37`). There is **no in-UI path** to obtain a `video_id` — no job submission, no poll of `/api/jobs/{id}`, no `/api/videos` list.
- **No progress/poll UI** (no `/api/jobs` polling — that's appliance doc screen S2, unbuilt).
- **No history/reload survival** (no `/api/videos` / `/api/reels` — S4, unbuilt; the endpoints don't even exist).

What the SPA **does** do well (M0 complete): load `?video=<id>` → `POST /api/query` (default all-selected) → deterministic query/select/bulk → `POST /api/compile` → `<video>` preview + download, with `min_shot_count` warning from `/api/config`, structured `cid`-traced logs, and full i18n (en/hi/kn). So the **back half** of the CUJ (review → select → compile → download) is built; the **front half** (point at a bucket video → process → get a `video_id`) is entirely missing from the UI.

## 6. Cheapest path to a dry run

The minimum to demonstrate "GCS video → UI → downloaded reel" **without building in-UI ingest** is to pre-seed the engine DB out-of-band, then drive the existing SPA back-half against it. The blocker is that **neither** the worker `infer` (writes CSV to bucket, not the DB) **nor** `/api/process` (local path only) bridges "GCS video → a queryable `video_id` in the API's SQLite". So the cheap path is:

**Steps (which already work vs must be run/built):**

1. **Get the bucket video onto the engine host as a local file** — `python -m backend.worker infer --video-uri gcs://bucket/match.mp4 --out-uri gcs://bucket/out --set indexing.skip_ai_handoff=true` *fetches* it to scratch, OR simpler: `gsutil cp gcs://bucket/match.mp4 ./incoming/` (the `gcs.py` fetch already proves GCS read works). — *fetch works; trivial to run.*
2. **Process it through the HTTP API so it lands in the API's DB** — `POST /api/process {video_path: "<abs local path>", segmenter: "gemini"}` (needs `GEMINI_API_KEY`) → poll `GET /api/jobs/{job_id}` until `result.video_id` appears. This populates `videos` + `play_segments` that `/api/query` and `/api/compile` read. — *endpoint works; just needs a key + a real video.* (Or `segmenter: "motion"` for a key-free real-boundaries run, accepting fabricated-metadata caveats.)
3. **Open the SPA pointed at that video** — `https://<host>/?video=<video_id>` (dev: `npm run dev`, Vite proxies `/api` → `:8000`). The SPA auto-loads rallies via `POST /api/query`. — *works today.*
4. **Select → Build → download** — pick rallies, click Build → `POST /api/compile` → `<video>` preview + `<a download>` saves the MP4 locally. — *works today.*
5. **Repeat in Hindi** — click the हिन्दी button in `LanguageSwitcher` (or append `?lang=hi`, or set localStorage), re-run select→compile→download. — *works today* (Hindi catalog fully populated, `i18n/index.ts` + `locales/hi/common.json`).

**So the only genuinely missing glue is step 2's "point at GCS" being a CLI/manual step, not in the UI.** If you accept a 1-command pre-process to mint the `video_id`, the dry run needs **zero new code** — only a running engine with a Gemini key, the GCS video copied/fetched local, and the SPA opened with `?video=<id>`.

**The alternative (full in-UI ingest)** would require building appliance-doc screens S1/S2 + the **net-new bucket-ingest HTTP endpoint** (T12/T14) — much more work and explicitly out of the current MVP scope.

## 7. Dependencies / unknowns to decide or stand up

- **Engine host & OS-runnability:** The engine binds `127.0.0.1:8000` (`main.py:655`) and needs `ffmpeg` + OpenCV. It's pure-Python FastAPI and *should* run on Windows for the **Gemini/motion** CPU paths (no WSL needed). **WSL/Linux-GPU is required only for `wasb` native/auto detection.** The validated GPU-free e2e is documented on the **CPU DigitalOcean droplet** (`168.144.126.176`, blr1, CPU-only). Decide: run the dry-run engine on **local Windows** (Gemini-CPU, simplest), the **CPU droplet**, or a **GPU host** (only if you want real WASB). For the bucket dry-run, local Windows + Gemini is the fewest moving parts.
- **Compute choice:** Gemini-CPU (real, needs key, cloud upload — consent copy) vs `motion` (key-free, fabricated metadata hidden) vs GPU/WASB (owner-byte-parity-gated, not production-verified). For "real-ish rallies, cheapest" → **Gemini + key**.
- **Which bucket video:** must pass the engine's validators (`config.json` validation: ≥1280×720, ≥29.9 fps, blur/scene-cut/court-relevance checks) or `/api/process` returns `status:"failed"`. Pick a clean ≥720p badminton clip.
- **GCS auth:** `GOOGLE_APPLICATION_CREDENTIALS` → SA-JSON (or `gcloud` ADC). Never in config (`gcs.py` docstring). Only needed if you use the worker fetch rather than `gsutil`.
- **CORS / SPA→engine connection:** engine sets CORS `allow_origins=["*"]`, `allow_credentials=False` (`main.py:41-47`) — so cross-origin browser calls are permitted. In dev, Vite **proxies** `/api` → `localhost:8000` (`vite.config.ts`), so it's same-origin and CORS is moot. In prod the appliance design is a single reverse-proxy origin (the gate, not the app, is the boundary). For a quick dry-run, `npm run dev` against a local engine needs no CORS/auth work.
- **Edge auth:** none in the engine by design (`LOCAL_APPLIANCE_ARCHITECTURE.md` §4.4 — Caddy basicauth or cloudflared+Access). Not required for a localhost dry-run; **required** before any non-localhost exposure (and **rotate the chat-shared `GEMINI_API_KEY`**).
- **The DB-bridge unknown (the real blocker):** there is **no supported in-API way to ingest a `gcs://` URI into a queryable `video_id`**. For the dry-run this is sidestepped by the local-copy + `/api/process` pre-step (Q6); productizing "bucket video from the UI" needs the net-new bucket-ingest endpoint (T12/T14) — an owner-gated decision.
