15	def interval_iou(a: Interval, b: Interval) -> float:
16	    """Temporal intersection-over-union of two intervals.
17	
18	    Returns 0.0 when they don't overlap (or either is degenerate). IoU is
19	    overlap / union, both measured as durations on the timeline.
20	    """

22	    b_start, b_end = b

26	    if inter <= 0.0:

28	    union = (a_end - a_start) + (b_end - b_start) - inter
29	    return inter / union if union > 0 else 0.0

@dataclass(frozen=True)
class RallySpec:
    """One in-flight rally segment: the shuttle oscillates across the net midline.

    ``start_frame`` — first frame index; ``n_frames`` — length; ``period`` — oscillation
    period in frames (governs net-crossing count); ``amplitude`` — px excursion from the
    net midline along the play axis. Deterministic sine motion → fixed velocity/crossings.
    """

    start_frame: int
    n_frames: int
    period: float = 30.0
    amplitude: float = 400.0


@dataclass(frozen=True)
class GapSpec:
    """Dead-time between rallies — what separates one window from the next.

    ``kind="still"`` — shuttle visible but resting (velocity ≈ 0 → no active frames);
    ``kind="occluded"`` — shuttle not visible (Visibility=0 → trajectory broken). Both
    end an active run so the next rally forms a distinct window.
    """

    start_frame: int
    n_frames: int
    kind: str = "still"  # "still" | "occluded"


@dataclass(frozen=True)
class FixtureSpec:
    """A full synthetic scenario: a frame-indexed sequence of rallies and gaps + its GT.

    ``gts`` are the hand-authored ground-truth rally intervals (seconds) — the synthetic
    "labels.csv". For a synthetic fixture they line up with the rallies by construction,
    which is exactly what makes the quality-floor a *plumbing-correctness* check, not a
    field-quality one (§6).
    """

    slug: str
    description: str
    fps: float
    frame_width: float
    frame_height: float
    segments: Tuple[Any, ...]  # ordered (RallySpec | GapSpec)
    gts: Tuple[Interval, ...]
    net_midline: float = 640.0


def _rally_points(spec: RallySpec, net_midline: float, frame_height: float
                  ) -> List[Tuple[int, int, float, float]]:
    """Deterministic in-flight shuttle: sine oscillation across the net midline (play axis = x)."""
    rows: List[Tuple[int, int, float, float]] = []
    y_mid = frame_height / 2.0
    for i in range(spec.n_frames):
        x = net_midline + spec.amplitude * math.sin(2.0 * math.pi * i / spec.period)
        # A small orthogonal wobble keeps y-spread < x-spread so the play axis is unambiguously x.
        y = y_mid + 30.0 * math.sin(2.0 * math.pi * i / (spec.period / 2.0))
        rows.append((spec.start_frame + i, 1, x, y))
    return rows


def _gap_points(spec: GapSpec, net_midline: float, frame_height: float
                ) -> List[Tuple[int, int, float, float]]:
    """Dead-time rows: resting-visible (still) or absent (occluded)."""
    rows: List[Tuple[int, int, float, float]] = []
    y_mid = frame_height / 2.0
    for i in range(spec.n_frames):
        if spec.kind == "occluded":
            rows.append((spec.start_frame + i, 0, 0.0, 0.0))
        elif spec.kind == "still":
            rows.append((spec.start_frame + i, 1, net_midline, y_mid))
        else:  # pragma: no cover - guarded by builder
            raise ValueError(f"unknown gap kind {spec.kind!r} (use 'still'|'occluded')")
    return rows


def make_synthetic_trajectory(spec: FixtureSpec) -> List[Tuple[int, int, float, float]]:
    """Build the deterministic ``(Frame, Visibility, X, Y)`` rows for a synthetic scenario."""
    rows: List[Tuple[int, int, float, float]] = []
    for seg in spec.segments:
        if isinstance(seg, RallySpec):
            rows.extend(_rally_points(seg, spec.net_midline, spec.frame_height))
        elif isinstance(seg, GapSpec):
            rows.extend(_gap_points(seg, spec.net_midline, spec.frame_height))
        else:  # pragma: no cover - guarded by dataclass shape
            raise TypeError(f"unknown segment type {type(seg).__name__}")
    rows.sort(key=lambda r: r[0])
    return rows


def write_trajectory_csv(rows: Sequence[Tuple[int, int, float, float]], path: Path) -> None:
    """Write trajectory rows as the TrackNet CSV (``Frame,Visibility,X,Y``), LF + 6dp floats."""
    lines = ["Frame,Visibility,X,Y"]
    for frame, vis, x, y in rows:
        lines.append(f"{int(frame)},{int(vis)},{_fmt(x)},{_fmt(y)}")
    path.write_text("\n".join(lines) + "\n", encoding="utf-8", newline="\n")


def write_labels_csv(gts: Sequence[Interval], path: Path) -> None:
    """Write GT intervals as a ``start,end`` labels CSV (the strict gt_loader format), LF."""
    lines = ["start,end"]
    for s, e in gts:
        lines.append(f"{_fmt(s)},{_fmt(e)}")
    path.write_text("\n".join(lines) + "\n", encoding="utf-8", newline="\n")

def _fmt(x: float) -> str:
    """Render a float with up to 6dp, no scientific notation, ``-0.0`` normalised to ``0.0``."""
    r = round(float(x) + 0.0, _FLOAT_ROUND_DP)
    if r == 0.0:
        r = 0.0
    return f"{r:.6f}"


def _round(x: float) -> float:
    r = round(float(x) + 0.0, _FLOAT_ROUND_DP)
    return 0.0 if r == 0.0 else r

    def exact(path: str, a: Any, b: Any) -> None:
        if a != b:
            out.append(f"{path}: {a!r} != {b!r} (exact)")

    def approx(path: str, a: Any, b: Any) -> None:
        try:
            if abs(float(a) - float(b)) > _FLOAT_TOL:
                out.append(f"{path}: {a!r} != {b!r} (|Δ| > {_FLOAT_TOL})")
        except (TypeError, ValueError):
            out.append(f"{path}: non-numeric float field {a!r} vs {b!r}")

    exact("slug", recomputed.get("slug"), committed.get("slug"))
    exact("schema_version", recomputed.get("schema_version"), committed.get("schema_version"))
    exact("gt_hash", recomputed.get("gt_hash"), committed.get("gt_hash"))
    approx("iou", recomputed.get("iou"), committed.get("iou"))

    rc, cc = recomputed.get("candidates", []), committed.get("candidates", [])
    if len(rc) != len(cc):

    else:

# `eval_baselines/fixtures/` — Golden Regression Fixtures (Tier-0, synthetic)

Video-free, **CONTROL-only** characterization fixtures for the rally-segmentation pipeline.
A contributor on **any machine — no video, GPU, WSL, or DB** — replays these committed
structured files through the already-shipped eval stack and gets a red/green regression
signal. See `docs/GOLDEN_REGRESSION_FIXTURES.md` (deliverable **M1**; §4, §6, §7).

## What's here

Each fixture is one directory:

| File | Purpose |
|------|---------|
| `trajectory.csv` | Synthetic shuttle trajectory (`Frame,Visibility,X,Y`) — the freeze seam input. |
| `labels.csv` | Synthetic ground-truth rally intervals (`start,end`). |
| `expected.json` | Frozen replay snapshot (candidates + 5 shuttle features + CONTROL segments + score). |
| `provenance.json` | Pinned windowing params + privacy/licensing tags + description. |

`manifest.json` lists every fixture; `backend/eval/golden_fixtures.py` is the shared replay
module + freeze/refresh/check CLI.

## Guardrails (non-negotiable)

- **Synthetic / owned only.** Every fixture is generated by deterministic formulae (NO random)
  — zero third-party footage, zero provenance/licensing risk. Tagged `kind: "synthetic"`,
  `license: "synthetic"`, `minors_free: true`.
- **No biometric / identity streams.** Tier-0 carries the **5 shuttle feature columns only**
  (`duration, peak_velocity, mean_velocity, active_ratio, net_crossings`). There is, by design,
  **no person / pose / gait / face / audio / player field anywhere in the schema** — a test
  asserts this positively (deletion-at-source, §4c). Minors and biometrics are excluded.
- **CONTROL / pure-stdlib only.** The replay uses `experiment.CONTROL` (`is_learned() == False`)
  — no numpy / logistic-regression. Learned variants' near-threshold decisions flip across CPUs,
  so they can **never** anchor a cross-machine snapshot (§4d / D5).
- **Parse-compare, never byte-compare.** The gate `json.load`s both sides: exact equality on
  int/structural fields, abs-tolerance `1e-6` on floats (CRLF/float-format safe).
- **Windowing is pinned per fixture.** The `(vt, mg, mwd, max_win)` params live in each
  `provenance.json` and are passed explicitly to `build_candidates` — the replay does **not**
  depend on the named `windowing.SERVED` constant, so a future SERVED retune can't silently
  churn fixtures.

## Fixtures are GENERATED, never hand-edited

Do **not** hand-edit `expected.json` / `provenance.json` / the CSVs. They are produced
correct-by-construction by the CLI.
